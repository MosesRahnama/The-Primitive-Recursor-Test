def lpo(s, t):
    # s and t are tuples or strings (representing variables)
    if isinstance(s, str) and isinstance(t, str):
        return s == t
    if isinstance(t, str):
        # s is compound, t is variable
        # check if t is in s
        return any(has_var(si, t) for si in s[1:])
    if isinstance(s, str):
        return False
    
    # s and t are compound terms: s = (f, s1, ...), t = (g, t1, ...)
    f, g = s[0], t[0]
    s_args, t_args = s[1:], t[1:]
    
    # Condition 2: there exists si such that si >=_lpo t

                break

                break
