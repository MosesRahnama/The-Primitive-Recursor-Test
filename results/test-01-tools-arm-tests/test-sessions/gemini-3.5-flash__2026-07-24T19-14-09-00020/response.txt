def lpo(s, t, precedence, status):
    # s and t are terms.
    # precedence: dict mapping constructor to an integer (higher means larger in precedence)
    # status: dict mapping constructor to a permutation (tuple of indices)
    
    if s == 0: # void is 0
        s_head = 'void'
        s_args = []
    else:
        s_head = s[0]
        s_args = list(s[1:])
        
    if t == 0:
        t_head = 'void'
        t_args = []
    else:
        t_head = t[0]
        t_args = list(t[1:])
        
    # LPO 1: s_i >= t for some i
    for s_i in s_args:
        if s_i == t or lpo(s_i, t, precedence, status):
            return True
            
    # LPO 2: s_head > t_head and s > t_i for all i
    if precedence[s_head] > precedence[t_head]:
        for t_i in t_args:
            if not (s == t_i or lpo(s, t_i, precedence, status)):
                return False
        return True
        
    # LPO 3: s_head == t_head and s > t_i for all i, and (s_1, ..., s_n) >_lex (t_1, ..., t_n)
    if s_head == t_head:
        for t_i in t_args:
            if not (s == t_i or lpo(s, t_i, precedence, status)):
                return False
        # compare lexicographically using the permutation in status
        perm = status.get(s_head, tuple(range(len(s_args))))
        for idx in perm:
            if idx >= len(s_args):
                break
            s_arg = s_args[idx]
            t_arg = t_args[idx]
            if s_arg == t_arg:
                continue
            elif lpo(s_arg, t_arg, precedence, status):
                return True
            else:
                return False
        return False
        
    return False
