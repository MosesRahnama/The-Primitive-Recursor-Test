# Let's define the terms and the rewrite rules.
# Since we want to search for loops, we can represent terms as tuples or custom classes.

class Term:
    def __repr__(self):
        return self.to_str()

class Void(Term):
    def to_str(self): return "void"
    def __eq__(self, other): return isinstance(other, Void)
    def __hash__(self): return hash("void")

class Delta(Term):
    def __init__(self, t): self.t = t
    def to_str(self): return f"delta({self.t})"
    def __eq__(self, other): return isinstance(other, Delta) and self.t == other.t
    def __hash__(self): return hash(("delta", self.t))

class Integrate(Term):
    def __init__(self, t): self.t = t
    def to_str(self): return f"integrate({self.t})"
    def __eq__(self, other): return isinstance(other, Integrate) and self.t == other.t
    def __hash__(self): return hash(("integrate", self.t))

class Merge(Term):
    def __init__(self, t1, t2): self.t1, self.t2 = t1, t2
    def to_str(self): return f"merge({self.t1}, {self.t2})"
    def __eq__(self, other): return isinstance(other, Merge) and self.t1 == other.t1 and self.t2 == other.t2
    def __hash__(self): return hash(("merge", self.t1, self.t2))

class App(Term):
    def __init__(self, t1, t2): self.t1, self.t2 = t1, t2
    def to_str(self): return f"app({self.t1}, {self.t2})"
    def __eq__(self, other): return isinstance(other, App) and self.t1 == other.t1 and self.t2 == other.t2
    def __hash__(self): return hash(("app", self.t1, self.t2))

class Rec(Term):
    def __init__(self, b, s, n): self.b, self.s, self.n = b, s, n
    def to_str(self): return f"rec({self.b}, {self.s}, {self.n})"
    def __eq__(self, other): return isinstance(other, Rec) and self.b == other.b and self.s == other.s and self.n == other.n
    def __hash__(self): return hash(("rec", self.b, self.s, self.n))

class EqW(Term):
    def __init__(self, a, b): self.a, self.b = a, b
    def to_str(self): return f"eqW({self.a}, {self.b})"
    def __eq__(self, other): return isinstance(other, EqW) and self.a == other.a and self.b == other.b
    def __hash__(self): return hash(("eqW", self.a, self.b))

# We want to find all one-step reductions of a term under congruence rules.
def step(t):
    # Returns a set of terms that t can reduce to in one step
    res = set()
    
    # 1. Top-level rules
    if isinstance(t, Integrate) and isinstance(t.t, Delta):
        res.add(Void())
    if isinstance(t, Merge):
        if isinstance(t.t1, Void):
            res.add(t.t2)
        if isinstance(t.t2, Void):
            res.add(t.t1)
        if t.t1 == t.t2:
            res.add(t.t1)
    if isinstance(t, Rec):
        if isinstance(t.n, Void):
            res.add(t.b)
        if isinstance(t.n, Delta):
            res.add(App(t.s, Rec(t.b, t.s, t.n.t)))
    if isinstance(t, EqW):
        if t.a == t.b:
            res.add(Void())
        res.add(Integrate(Merge(t.a, t.b)))
        
    # 2. Congruence rules
    if isinstance(t, Delta):
        for s in step(t.t):
            res.add(Delta(s))
    elif isinstance(t, Integrate):
        for s in step(t.t):
            res.add(Integrate(s))
    elif isinstance(t, Merge):
        for s in step(t.t1):
            res.add(Merge(s, t.t2))
        for s in step(t.t2):
            res.add(Merge(t.t1, s))
    elif isinstance(t, App):
        for s in step(t.t1):
            res.add(App(s, t.t2))
        for s in step(t.t2):
            res.add(App(t.t1, s))
    elif isinstance(t, Rec):
        for s in step(t.b):
            res.add(Rec(s, t.s, t.n))
        for s in step(t.s):
            res.add(Rec(t.b, s, t.n))
        for s in step(t.n):
            res.add(Rec(t.b, t.s, s))
    elif isinstance(t, EqW):
        for s in step(t.a):
            res.add(EqW(s, t.b))
        for s in step(t.b):
            res.add(EqW(t.a, s))
            
    return res
