def get_vars(t):
    if isinstance(t, str):
        return {t}
    f, args = t
    vars = set()
    for arg in args:
        vars |= get_vars(arg)
    return vars

def multiset_greater(M, N, prec):
    # remove common elements
    M = list(M)
    N = list(N)
    for x in M[:]:
        if x in N:
            M.remove(x)
            N.remove(x)
    if not M:
        return False
    # for every y in N, there must be x in M such that x > y
    for y in N:
        if not any(mpo(x, y, prec) for x in M):
            return False
    return True
