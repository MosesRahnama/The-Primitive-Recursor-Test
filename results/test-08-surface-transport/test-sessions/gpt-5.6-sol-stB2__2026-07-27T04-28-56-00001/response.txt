\[
\mathsf{exec\_walk\_weight} \succ \mathsf{add} \succ \mathsf{app} \succ
\mathsf{cons} \succ \mathsf{zero} \succ \mathsf{nil},

\[

\]

   \[
   s \succ

   s \succ \mathsf{app}(\mathsf{app}(w,x),y).

   s \succ
