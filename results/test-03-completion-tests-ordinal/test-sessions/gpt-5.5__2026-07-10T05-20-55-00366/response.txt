A good way to finish the three `sorry`s is to factor the hard ordinal arithmetic into two local comparison lemmas, then keep the rule cases themselves short. I would structure the completion like this.

```lean
-- Useful helper: two terms below an additively-principal omega-power
-- have their ordinal sum, and successor of that sum, still below it.
--
-- In Mathlib this is usually proved using that `omega0 ^ E` is additively
-- principal when `0 < E`.
private lemma add_add_one_lt_omega_opow
    {E x y : Ordinal}
    (hE : 0 < E)
    (hx : x < omega0 ^ E)
    (hy : y < omega0 ^ E) :
    x + y + 1 < omega0 ^ E := by
  -- Skeleton:
  -- 1. show `omega0 ^ E` is additively principal / closed under addition
  -- 2. from `hx hy`, get `x + y < omega0 ^ E`
  -- 3. since `omega0 ^ E` is a limit ordinal, get `(x + y) + 1 < omega0 ^ E`
  sorry
```

For the recursion successor case, prove a specialized bound saying the whole `app` target is below the leading omega-power of the source.

```lean
private lemma app_recDelta_bound
    (B S N : Ordinal) :
    omega0 ^ (3 : Ordinal) * (S + 1)
      + omega0 ^ (2 : Ordinal)
          * (omega0 ^ (N + S + (6 : Ordinal))
              + omega0 * (B + 1) + 1 + 1)
      + 1
    <
    omega0 ^
      (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)) := by
  let E :=
    omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)

  have hEpos : 0 < E := by
    -- immediate from the final `+ 6`
    -- e.g. prove `0 < (6 : Ordinal)` and use `add_pos_right`
    sorry

  -- First app component is small.
  have hleft :
      omega0 ^ (3 : Ordinal) * (S + 1) < omega0 ^ E := by
    -- Show its exponent/degree is below `E`.
    -- A standard route:
    --   `omega0 ^ 3 * (S + 1) < omega0 ^ (S + 4)`
    -- and then
    --   `S + 4 ≤ E`
    -- because `E` has the huge leading part
    --   `omega0 ^ 5 * (N + 1) + 1`.
    sorry

  -- The recursive argument appearing in the second app component is also small.
  have hrec :
      omega0 ^ (2 : Ordinal)
          * (omega0 ^ (N + S + (6 : Ordinal))
              + omega0 * (B + 1) + 1 + 1)
      < omega0 ^ E := by
    -- Bound the parenthesized term by an omega-power:
    --
    --   omega0 ^ (N + S + 6)
    --     + omega0 * (B + 1) + 1 + 1
    --   < omega0 ^ (N + S + 7)
    --
    -- Then multiply on the left by `omega0 ^ 2`:
    --
    --   omega0 ^ 2 * (...) < omega0 ^ (2 + (N + S + 7))
    --
    -- Finally show
    --
    --   2 + (N + S + 7) ≤ E
    --
    -- since
    --
    --   E = omega0 ^ 5 * (N + 1) + 1 + S + 6
    --
    -- and the leading term `omega0 ^ 5 * (N + 1)` dominates `N`.
    sorry

  exact add_add_one_lt_omega_opow hEpos hleft hrec
```

Then the `R_rec_succ` case becomes:

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    let B := mu b
    let S := mu s
    let N := mu n

    have hmain :
        omega0 ^ (3 : Ordinal) * (S + 1)
          + omega0 ^ (2 : Ordinal)
              * (omega0 ^ (N + S + (6 : Ordinal))
                  + omega0 * (B + 1) + 1 + 1)
          + 1
        <
        omega0 ^
          (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)) :=
      app_recDelta_bound B S N

    -- The source measure is the same leading omega-power, plus positive tail.
    calc
      omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal)
              * (omega0 ^ (mu n + mu s + (6 : Ordinal))
                  + omega0 * (mu b + 1) + 1 + 1)
          + 1
          <
        omega0 ^
          (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) := by
            simpa [B, S, N] using hmain
      _ ≤
        omega0 ^
          (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal))
          + omega0 * (mu b + 1) :=
            le_add_right _ _
      _ ≤
        omega0 ^
          (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal))
          + omega0 * (mu b + 1) + 1 :=
            le_add_right _ _
```

For `eq_diff`, use a second specialized bound.

```lean
private lemma integrate_merge_bound
    (A B : Ordinal) :
    omega0 ^ (4 : Ordinal)
        * (omega0 ^ (3 : Ordinal) * (A + 1)
            + omega0 ^ (2 : Ordinal) * (B + 1) + 1 + 1)
      + 1
    <
    omega0 ^ (A + B + (9 : Ordinal)) := by
  let E := A + B + (9 : Ordinal)

  have hEpos : 0 < E := by
    -- from the final `+ 9`
    sorry

  have hmerge :
      omega0 ^ (3 : Ordinal) * (A + 1)
          + omega0 ^ (2 : Ordinal) * (B + 1) + 1 + 1
      <
      omega0 ^ (A + B + (5 : Ordinal)) := by
    -- Bound the two merge components separately:
    --
    --   omega0 ^ 3 * (A + 1) < omega0 ^ (A + B + 5)
    --   omega0 ^ 2 * (B + 1) < omega0 ^ (A + B + 5)
    --
    -- The first uses `A + 4 ≤ A + B + 5`.
    -- The second uses `B + 3 ≤ A + B + 5`.
    -- Then close under addition/successor using additivity-principality.
    sorry

  have hmul :
      omega0 ^ (4 : Ordinal)
        * (omega0 ^ (3 : Ordinal) * (A + 1)
            + omega0 ^ (2 : Ordinal) * (B + 1) + 1 + 1)
      <
      omega0 ^ (A + B + (9 : Ordinal)) := by
    -- From `hmerge`:
    --
    --   inner < omega0 ^ (A + B + 5)
    --
    -- hence
    --
    --   omega0 ^ 4 * inner < omega0 ^ (4 + (A + B + 5)).
    --
    -- Then show
    --
    --   4 + (A + B + 5) ≤ A + B + 9.
    sorry

  -- Since `omega0 ^ E` is a limit ordinal, adding one stays below it.
  have hsucc :
      omega0 ^ (4 : Ordinal)
        * (omega0 ^ (3 : Ordinal) * (A + 1)
            + omega0 ^ (2 : Ordinal) * (B + 1) + 1 + 1)
        + 1
      <
      omega0 ^ E := by
    -- use limit-ness of `omega0 ^ E` and `hmul`
    sorry

  simpa [E] using hsucc
```

Then the `R_eq_diff` case is just:

```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    have hmain :
        omega0 ^ (4 : Ordinal)
            * (omega0 ^ (3 : Ordinal) * (mu a + 1)
                + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
          + 1
        <
        omega0 ^ (mu a + mu b + (9 : Ordinal)) :=
      integrate_merge_bound (mu a) (mu b)

    calc
      omega0 ^ (4 : Ordinal)
            * (omega0 ^ (3 : Ordinal) * (mu a + 1)
                + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
          + 1
          <
        omega0 ^ (mu a + mu b + (9 : Ordinal)) := hmain
      _ <
        omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 :=
          lt_add_one _
```

So the high-level completion is:

1. Finish `R_eq_refl` directly with positivity.
2. Prove one absorption lemma for the recursive successor target.
3. Prove one absorption lemma for the `integrate (merge a b)` target.
4. In both hard cases, compare the target to the leading omega-power of the source, then use `le_add_right` / `lt_add_one` to account for the source’s positive tail.
