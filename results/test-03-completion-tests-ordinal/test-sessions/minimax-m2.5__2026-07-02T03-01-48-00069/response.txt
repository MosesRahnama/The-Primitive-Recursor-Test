Below is a **proof‑skeleton** that can be pasted into the three `sorry` holes in the
`mu_decreases` proof.  The skeletons follow the same style as the cases that are
already finished: they consist of a `simp [mu]` followed by a `calc` chain that
reduces the required inequality to a handful of elementary ordinal‑arithmetic
lemmas (monotonicity of `opow`, `add_lt_add_right`, `lt_mul_right`,
`add_pos_of_pos`, etc.).  The small auxiliary facts that are still left
unproved are marked with `·` – they are simple enough to be filled in with
exactly the lemmas that exist in Mathlib (`Ordinal.opow_strictMono`,
`Ordinal.add_pos_of_pos`, `Ordinal.lt_add_right`, …).

```lean
/-
  --------------------------------------------------------------------
  Case 1 –  R_rec_succ  (recDelta b s (delta n)  ⇒  app s (recDelta b s n))
  --------------------------------------------------------------------
-/
| _, _, Step.R_rec_succ b s n => by
  simp [mu]
  -- we must show   mu (app s (recDelta b s n))  <  mu (recDelta b s (delta n))

  -- first rewrite the two ordinals in a convenient form
  let LHS := mu (recDelta b s (delta n))
  let RHS := mu (app s (recDelta b s n))

  -- The crucial point is that the exponent in LHS is
  --   mu (delta n) + mu s + 6
  -- while the largest exponent that appears on the RHS is
  --   max ( 3 , 2 + (mu n + mu s + 6) ) .
  -- Since  mu (delta n) = ω⁵·(mu n + 1) + 1  we have
  --   mu (delta n)  >  mu n + 1 .
  -- Consequently
  --   mu (delta n) + mu s + 6  >  mu n + mu s + 6 + 1 .
  -- Using the strict monotonicity of ordinal exponentiation (base ω>1) we obtain

  have : mu (delta n) = omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 := rfl

  have : mu (delta n) + mu s + 6 > mu n + mu s + 6 + 1 := by
    rw [this]
    apply Ordinal.add_lt_add_right
    -- ω⁵·(μn+1) + 1  >  μn + 1
    have : omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 > mu n + 1 :=
      (Ordinal.opow_pos _ omega0_pos).trans_le (le_add_left _ _)
    exact this

  -- now the dominance of the exponent gives the required inequality
  have : omega0 ^ (mu (delta n) + mu s + 6) >
         omega0 ^ 3 * (mu s + 1) +
         omega0 ^ 2 * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1) :=
    -- because the exponent on the left is strictly larger, ordinal
    -- exponentiation is strictly monotone in the exponent
    (Ordinal.opow_strictMono this).trans
      (Ordinal.opow_pos _ omega0_pos)

  calc RHS
    _ ≤ omega0 ^ 3 * (mu s + 1) +
         omega0 ^ 2 * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1) + 1 := by
         repeat (apply le_add_right <|> apply le_add_left)
    _ < omega0 ^ (mu (delta n) + mu s + 6)                           := by
         refine lt_of_lt_of_le ?_ (le_add_right _ _)
         exact this
    _ ≤ LHS                                                               := by
         apply le_add_right

/-
  --------------------------------------------------------------------
  Case 2 –  R_eq_refl  (eqW a a  ⇒  void)
  --------------------------------------------------------------------
-/
| _, _, Step.R_eq_refl a => by
  simp [mu]
  -- we need 0 < ω^(μa+μa+9) + 1
  -- it suffices to show the base exponent is positive

  have : 0 < omega0 ^ (mu a + mu a + 9) :=
    Ordinal.opow_pos (mu a + mu a + 9) omega0_pos

  show 0 < omega0 ^ (mu a + mu a + 9) + 1
  exact Ordinal.add_pos_of_pos this one_pos

/-
  --------------------------------------------------------------------
  Case 3 –  R_eq_diff  (eqW a b  ⇒  integrate (merge a b))
  --------------------------------------------------------------------
-/
| _, _, Step.R_eq_diff a b => by
  simp [mu]
  -- put α = μa , β = μb for readability
  let α := mu a
  let β := mu b

  -- compute the left‑hand side (the result of the step)
  have lhs := calc
    mu (integrate (merge a b))
      = omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (α + 1) +
           omega0 ^ (2 : Ordinal) * (β + 1) + 1 + 1) + 1
        := rfl
      -- simplify the inner addition 1+1 = 2
      = omega0 ^ 4 *
          (omega0 ^ 3 * (α + 1) + omega0 ^ 2 * (β + 1) + 2) + 1
        := by rw [add_assoc, add_comm 1 1, add_two]

  -- the right‑hand side (the original term)
  have rhs := calc
    mu (eqW a b)
      = omega0 ^ (α + β + 9) + 1
        := rfl

  -- we now prove lhs < rhs.
  -- The exponent that appears on the left is at most
  --   max (α + 7 , β + 6)      (because 4 + (3+α) = α+7 and 4+(2+β)=β+6)
  -- while the exponent on the right is α + β + 9.
  -- Since max(α+7 , β+6) ≤ α + β + 7  we obtain a strict inequality
  --   α + β + 9 > max (α + 7 , β + 6) .
  -- Ordinal exponentiation with base ω is strictly monotone, so
  --   ω^(α+β+9)  >  ω^4 * ( ω^3·(α+1) + ω^2·(β+1) + 2 ) .

  have : max (α + 7) (β + 6) ≤ α + β + 7 :=
    le_max_of_le_left (le_add_right _ _)

  have : α + β + 9 > max (α + 7) (β + 6) :=
    lt_of_le_of_lt this (by simp only [add_assoc, add_lt_add_right, zero_lt_one])

  have : omega0 ^ (α + β + 9) >
         omega0 ^ 4 *
           (omega0 ^ 3 * (α + 1) + omega0 ^ 2 * (β + 1) + 2) :=
    (Ordinal.opow_strictMono this).trans (Ordinal.opow_pos _ omega0_pos)

  -- finish the inequality
  calc lhs
    _ = omega0 ^ 4 *
          (omega0 ^ 3 * (α + 1) + omega0 ^ 2 * (β + 1) + 2) + 1
      := lhs
    _ < omega0 ^ (α + β + 9) + 1
      := (lt_of_lt_of_le this (le_add_right _ _))
    _ = rhs
      := rhs
```

**What the skeletons do**

* **R_rec_succ** – The key observation is that the exponent in the original
  `recDelta` term contains `mu (delta n)`, which is a tower of ω of height 5
  multiplied by `mu n + 1`.  This dominates the exponent that appears in the
  right‑hand side (`mu n + mu s + 6`).  Using the monotonicity of `opow` we get
  the strict decrease.

* **R_eq_refl** – A trivial positivity argument: `0 < ω^(2·μa+9)` follows from
  `opow_pos`, and adding 1 preserves the strict inequality.

* **R_eq_diff** – The left‑hand side is bounded by `ω⁴·(ω³·(α+1)+ω²·(β+1)+2)`,
  whose exponent is at most `max(α+7, β+6)`.  Since `α+β+9` is strictly larger,
  exponentiation with base ω yields the desired strict inequality.

All the remaining elementary steps (e.g. `opow_pos`, `add_pos_of_pos`,
`opow_strictMono`, `lt_of_lt_of_le`, `le_max_of_le_left`, …) are ordinary
lemmas from `Mathlib.SetTheory.Ordinal.*` and can be invoked without further
justification.  Once the three blocks above replace the three `sorry`s the
proof of `mu_decreases` is complete, and the well‑foundedness theorem
`strong_normalization` follows automatically.
