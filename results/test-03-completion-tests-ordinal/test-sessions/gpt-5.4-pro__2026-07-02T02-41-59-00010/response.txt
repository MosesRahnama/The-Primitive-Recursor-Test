Below are in-place proof skeletons for the three remaining branches. I have **not** compiled them, so the only thing you may need to adjust is the exact names of the normality / additive-principal lemmas for `ω^(-)` in your Mathlib snapshot.

### 1) `eqW a a --> void`

This one is just the same positivity pattern as `R_int_delta`.

```lean
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr <| by
      apply ne_of_gt
      exact add_pos_right _ one_pos
```

### 2) `eqW a b --> integrate (merge a b)`

A clean way to do this is:

- let `E := mu a + mu b + 9`
- rewrite the left-hand side as a sum of four pieces
- show each piece is `< ω^E`
- combine them using the fact that `ω^E` is additively principal
- then add the trailing `+ 1` on both sides

```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    let E : Ordinal := mu a + mu b + 9

    /- Helper: a finite ω-prefix is below one higher exponent.
       You can reuse the same helper in the rec_succ branch. -/
    have hprefix :
        ∀ k : Nat, ∀ x : Ordinal,
          omega0 ^ (k : Ordinal) * (x + 1) < omega0 ^ (x + (k + 1 : Ordinal)) := by
      intro k x
      -- skeleton:
      -- 1. prove `x + 1 ≤ omega0 ^ (x + 1)`
      -- 2. multiply on the left by `omega0 ^ k`
      -- 3. rewrite with `opow_add`
      -- 4. use monotonicity/normality of `fun y => omega0 ^ y`
      --
      -- Typical shape:
      -- have hx : x + 1 ≤ omega0 ^ (x + 1) := ...
      -- have hm :
      --   omega0 ^ (k : Ordinal) * (x + 1)
      --     ≤ omega0 ^ (k : Ordinal) * omega0 ^ (x + 1) := ...
      -- calc
      --   omega0 ^ (k : Ordinal) * (x + 1)
      --       ≤ omega0 ^ (k : Ordinal) * omega0 ^ (x + 1) := hm
      --   _ = omega0 ^ ((k : Ordinal) + (x + 1)) := by
      --         simpa [opow_add, Ordinal.mul_assoc]
      --   _ ≤ omega0 ^ (x + (k + 1 : Ordinal)) := by
      --         -- finite-left-addition estimate on exponents
      --         ...
      --   _ < omega0 ^ (x + (k + 1 : Ordinal)) := by exact lt_of_le_of_ne ...
      sorry

    have ha0 : omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + 8) := by
      simpa using hprefix 7 (mu a)

    have hb0 : omega0 ^ (6 : Ordinal) * (mu b + 1) < omega0 ^ (mu b + 7) := by
      simpa using hprefix 6 (mu b)

    have ha : omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ E := by
      refine lt_of_lt_of_le ha0 ?_
      -- exponent monotonicity, since `mu a + 8 ≤ mu a + mu b + 9`
      -- use normality/monotonicity of `omega0 ^ ·`
      have hE : mu a + 8 ≤ E := by
        -- `8 ≤ mu b + 9`
        -- then add `mu a` on the left
        ...
      exact
        -- something like:
        -- (Ordinal.isNormal_opow ?).monotone hE
        ...
    
    have hb : omega0 ^ (6 : Ordinal) * (mu b + 1) < omega0 ^ E := by
      refine lt_of_lt_of_le hb0 ?_
      have hE : mu b + 7 ≤ E := by
        -- `mu b + 7 ≤ mu a + mu b + 9`
        ...
      exact
        -- monotonicity of `omega0 ^ ·`
        ...

    have hω4 : omega0 ^ (4 : Ordinal) < omega0 ^ E := by
      -- enough to show `4 < E`; since `E = mu a + mu b + 9`, this is immediate
      have hE : (4 : Ordinal) < E := by
        -- `4 < 9 ≤ mu a + mu b + 9`
        ...
      exact
        -- strict monotonicity of `omega0 ^ ·`
        ...

    have hsum :
        omega0 ^ (7 : Ordinal) * (mu a + 1) +
          omega0 ^ (6 : Ordinal) * (mu b + 1) +
          omega0 ^ (4 : Ordinal) +
          omega0 ^ (4 : Ordinal) < omega0 ^ E := by
      -- combine `ha`, `hb`, `hω4`, `hω4`
      -- using additive-principal closure of `omega0 ^ E`
      --
      -- In Mathlib this is usually done with lemmas from `Ordinal.Principal`
      -- saying that if `x < ω^E` and `y < ω^E`, then `x + y < ω^E`.
      ...
    
    have hrewrite :
        omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (mu a + 1) +
            omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
        =
        omega0 ^ (7 : Ordinal) * (mu a + 1) +
          omega0 ^ (6 : Ordinal) * (mu b + 1) +
          omega0 ^ (4 : Ordinal) +
          omega0 ^ (4 : Ordinal) := by
      -- distribute left multiplication, then rewrite powers
      simp [mul_add, Ordinal.mul_assoc, opow_add, add_assoc, add_left_comm, add_comm]

    have hmain :
        omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (mu a + 1) +
            omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
        < omega0 ^ E := by
      rw [hrewrite]
      exact hsum

    simpa [E] using add_lt_add_right hmain 1
```

### 3) `recDelta b s (delta n) --> app s (recDelta b s n)`

This is the delicate branch. The structure I would use is:

- set `E := mu (delta n) + mu s + 6`
- show the `ω^3 * (mu s + 1)` part is `< ω^E`
- expand `ω^2 * (mu (recDelta b s n) + 1)` by distributing over the definition of `mu (recDelta b s n)`
- handle its four pieces separately:
  1. the head `ω^2 * ω^(mu n + mu s + 6)` is `< ω^E`
  2. the two `ω^2` constants are `< ω^E`
  3. the `ω^3 * (mu b + 1)` term is the only non-routine one; isolate it as a helper claim
- combine everything, then add the final `+ 1`

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    let E : Ordinal := mu (delta n) + mu s + 6

    have hprefix :
        ∀ k : Nat, ∀ x : Ordinal,
          omega0 ^ (k : Ordinal) * (x + 1) < omega0 ^ (x + (k + 1 : Ordinal)) := by
      intro k x
      -- same helper as in the eq_diff branch
      ...
    
    have hs0 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu s + 4) := by
      simpa using hprefix 3 (mu s)

    have hs : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ E := by
      refine lt_of_lt_of_le hs0 ?_
      have hE : mu s + 4 ≤ E := by
        -- because `mu (delta n)` is huge and positive
        -- enough: `4 ≤ mu (delta n) + 6`
        ...
      exact
        -- monotonicity of `omega0 ^ ·`
        ...

    have hhead :
        omega0 ^ (2 : Ordinal) * omega0 ^ (mu n + mu s + 6) < omega0 ^ E := by
      -- rewrite to `omega0 ^ ((2 : Ordinal) + (mu n + mu s + 6))`
      -- then compare exponents:
      -- `(2 : Ordinal) + (mu n + mu s + 6) < mu (delta n) + mu s + 6`
      -- because `mu (delta n) = omega0 ^ 5 * (mu n + 1) + 1` dominates `mu n + 8`
      ...
    
    have hω2 : omega0 ^ (2 : Ordinal) < omega0 ^ E := by
      have hE : (2 : Ordinal) < E := by
        -- again because `E = mu (delta n) + mu s + 6`
        ...
      exact
        -- strict monotonicity of `omega0 ^ ·`
        ...

    /-
      This is the key local helper.
      After distributing `ω^2 * (mu (recDelta b s n) + 1)`,
      you get a term `ω^3 * (mu b + 1)`.
      You need to compare it with the right-hand side
        `ω^E + ω * (mu b + 1)`.
      
      If your current Mathlib lemmas are sufficient, prove this directly.
      If not, factor it out as a separate structural lemma about the image of `mu`.
    -/
    have htail :
        omega0 ^ (3 : Ordinal) * (mu b + 1)
          ≤ omega0 ^ E + omega0 * (mu b + 1) := by
      -- proof idea:
      --   * either the `b`-part is still below the big head `ω^E`,
      --     in which case this is immediate by `le_add_right`;
      --   * or its leading exponent is already ≥ E, and then the finite left factor
      --     is absorbed by `ω * (mu b + 1)`.
      --
      -- In practice, this is the only place where you may want a separate lemma
      -- about the shape of ordinals in the range of `mu`.
      ...
    
    have hrec :
        omega0 ^ (2 : Ordinal) *
          (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1 + 1)
          < omega0 ^ E + omega0 * (mu b + 1) := by
      -- distribute the product:
      --   ω^2 * ω^(mu n + mu s + 6)
      -- + ω^2 * (ω * (mu b + 1))
      -- + ω^2 * 1
      -- + ω^2 * 1
      --
      -- rewrite the second term as `ω^3 * (mu b + 1)`,
      -- use `hhead`, `htail`, `hω2`, `hω2`,
      -- then combine with additive-principal closure of `ω^E`.
      have hrewrite :
          omega0 ^ (2 : Ordinal) *
            (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1 + 1)
          =
          omega0 ^ (2 : Ordinal) * omega0 ^ (mu n + mu s + 6) +
            omega0 ^ (3 : Ordinal) * (mu b + 1) +
            omega0 ^ (2 : Ordinal) +
            omega0 ^ (2 : Ordinal) := by
        simp [mul_add, Ordinal.mul_assoc, opow_add, add_assoc, add_left_comm, add_comm]
      rw [hrewrite]
      -- now combine `hhead`, `htail`, `hω2`, `hω2`
      ...
    
    have hmain :
        omega0 ^ (3 : Ordinal) * (mu s + 1) +
          omega0 ^ (2 : Ordinal) *
            (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1 + 1) + 1
        < omega0 ^ E + omega0 * (mu b + 1) + 1 := by
      -- combine `hs` with `hrec`, then add the trailing `+ 1`
      ...
    
    simpa [E] using hmain
```

## Practical note

The **easy** missing branch is `R_eq_refl`.  
`R_eq_diff` should go through once you set up the finite-prefix helper and use additive-principal closure of `ω^E`.  
`R_rec_succ` is the only branch where a genuinely nontrivial auxiliary lemma may be needed; the precise sticking point is the comparison of

```lean
omega0 ^ (3 : Ordinal) * (mu b + 1)
```

with

```lean
omega0 ^ E + omega0 * (mu b + 1)
```

inside the expansion of the recursive argument. If that exact subgoal resists, that is the place to isolate a separate lemma about the shape of ordinals in the image of `mu`.
