I'll provide proof skeletons for the three remaining cases in the `mu_decreases` theorem. These proofs rely on ordinal arithmetic properties, particularly the dominance of higher ω-powers.

### 1. `R_merge_cancel` case
```lean
| _, _, Step.R_merge_cancel t => by
    simp [mu]
    have h1 : mu t < mu t + 1 := lt_add_one (mu t)
    have h2 : mu t + 1 ≤ omega0 ^ (2 : Ordinal) * (mu t + 1) :=
      le_mul_right _ (opow_pos 2 omega0_pos)
    have h3 : mu t < omega0 ^ (2 : Ordinal) * (mu t + 1) := lt_of_lt_of_le h1 h2
    have h4 : omega0 ^ (2 : Ordinal) * (mu t + 1) ≤
              omega0 ^ (3 : Ordinal) * (mu t + 1) :=
      le_mul_right _ (opow_pos 1 omega0_pos)
    calc mu t
      _ < omega0 ^ (2 : Ordinal) * (mu t + 1) := h3
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) := h4
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) :=
          le_add_right _ _
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) + 1 :=
          le_add_right _ _
```

### 2. `R_rec_succ` case
```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- The key is that ω^(ω^5 * (μn + 1) + μs + 7) dominates all terms in the RHS
    have h1 : omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7 > mu s + 6 := by
      apply add_lt_add_right
      apply lt_of_lt_of_le
      · exact lt_add_one (mu s + 6)
      · apply le_add_right
        apply le_mul_right
        apply opow_pos 5 omega0_pos
    have h2 : omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) >
              omega0 ^ (mu s + 6) := opow_lt_opow_left omega0_pos h1
    have h3 : omega0 ^ (mu s + 6) ≥ omega0 * (mu b + 1) := by
      apply opow_le_opow_of_le (by simp [Ordinal.pos_iff_ne_zero])
      apply le_add_right
    have h4 : omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) >
              omega0 * (mu b + 1) := lt_of_lt_of_le h2 h3
    have h5 : omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) >
              omega0 ^ (3 : Ordinal) * (mu s + 1) := by
      apply opow_lt_opow_left omega0_pos
      apply lt_of_lt_of_le
      · exact lt_add_one (mu s)
      · apply le_mul_right
        apply opow_pos 3 omega0_pos
    have h6 : omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) >
              omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) := by
      apply opow_lt_opow_left omega0_pos
      apply lt_of_lt_of_le
      · apply lt_add_of_pos_right
        apply opow_pos 2 omega0_pos
      · apply le_mul_right
        apply opow_pos 5 omega0_pos
    calc mu (app s (recDelta b s n))
      _ = omega0 ^ (3 : Ordinal) * (mu s + 1) +
          omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := rfl
      _ < omega0 ^ (3 : Ordinal) * (mu s + 1) +
          omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) + 1 :=
          add_lt_add_left (mul_lt_mul_right (opow_pos 2 omega0_pos)
            (add_lt_add_right (mu_lt_recDelta b s n) 1)) _
      _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) := by
          apply lt_of_lt_of_le
          · apply add_lt_add_right h5
          · apply le_add_right
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) +
          omega0 * (mu b + 1) + 1 := le_add_right _ _
      _ = mu (recDelta b s (delta n)) := rfl
```

### 4. `R_eq_diff` case
```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- The key is that ω^(μa + μb + 9) dominates ω^4 * (ω^3 * (μa + 1) + ω^2 * (μb + 1) + 2)
    have h1 : mu a + mu b + 9 > 4 := by
      apply lt_of_lt_of_le
      · exact lt_add_of_pos_right (by apply add_pos_right; apply opow_pos 4 omega0_pos)
      · apply le_add_right
    have h2 : omega0 ^ (mu a + mu b + 9) > omega0 ^ 4 := opow_lt_opow_left omega0_pos h1
    have h3 : omega0 ^ 4 ≥ omega0 ^ 4 * (omega0 ^ 3 * (mu a + 1) + omega0 ^ 2 * (mu b + 1) + 2) := by
      apply opow_le_opow_of_le (by simp [Ordinal.pos_iff_ne_zero])
      apply le_mul_right
      apply le_add_right
    have h4 : omega0 ^ (mu a + mu b + 9) >
              omega0 ^ 4 * (omega0 ^ 3 * (mu a + 1) + omega0 ^ 2 * (mu b + 1) + 2) :=
      lt_of_lt_of_le h2 h3
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := rfl
      _ = omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) +
          omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1 := rfl
      _ < omega0 ^ (mu a + mu b + 9) := by
          apply lt_of_lt_of_le
          · apply add_lt_add_right h4
          · apply le_add_right
      _ ≤ omega0 ^ (mu a + mu b + 9) + 1 := le_add_right _ _
      _ = mu (eqW a b) := rfl
```

These proofs rely on the fact that higher ω-powers dominate lower ones, and that ordinal addition preserves order. The most complex case is `R_rec_succ` where we need to show that the exponent containing ω^5 dominates all terms in the right-hand side.
