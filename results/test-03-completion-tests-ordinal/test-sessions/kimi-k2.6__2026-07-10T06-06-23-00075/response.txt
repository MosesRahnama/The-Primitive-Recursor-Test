Here is a proof skeleton for the three remaining `sorry` cases.  The two easy openings (`R_eq_refl` and the transitivity step in the other cases) are filled in directly; the genuinely hard arithmetic is isolated in named `have` blocks so that you (or `nlinarith` / custom lemmas) can discharge them.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- Step 1: the reduct is bounded by the leading tower of the source
    have h1 : omega0 ^ (3 : Ordinal) * (mu s + 1)
            + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1)
            + 1
            < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
      have h2 : mu (recDelta b s n) + 1
              = omega0 ^ (mu n + mu s + (6 : Ordinal))
                + omega0 * (mu b + 1) + 2 := by
        simp [mu]
        <;> rw [add_assoc]
      rw [h2]
      have h3 : omega0 ^ (2 : Ordinal)
                * (omega0 ^ (mu n + mu s + (6 : Ordinal))
                   + omega0 * (mu b + 1) + 2)
              = omega0 ^ (2 + mu n + mu s + (6 : Ordinal))
                + omega0 ^ (3 : Ordinal) * (mu b + 1)
                + omega0 ^ (2 : Ordinal) * 2 := by
        rw [mul_add, mul_add]
        congr
        · rw [← opow_add]
        · rw [← mul_assoc]
          congr 1
          rw [← opow_add]
          simp
        · rfl
      rw [h3]
      -- Each summand on the left is dominated by omega^(mu(delta n) + mu s + 6)
      -- because mu(delta n) = omega^5*(mu n+1)+1 dwarfs every exponent
      -- appearing in the expansion of the reduct.
      sorry
    -- Step 2: the source is that tower plus two positive tail terms
    have h2 : omega0 ^ (mu (delta n) + mu s + (6 : Ordinal))
            < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal))
              + omega0 * (mu b + 1) + 1 := by
      have pos : 0 < omega0 * (mu b + 1) + 1 := by
        apply add_pos_right
        exact one_pos
      have step : omega0 ^ (mu (delta n) + mu s + (6 : Ordinal))
                < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + (omega0 * (mu b + 1) + 1) :=
        lt_add_of_pos_right _ pos
      rwa [add_assoc] at step ⊢
    exact lt_trans h1 h2

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- It suffices to show the integrate/merge tower is below the eqW tower.
    have h1 : omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
            < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      have h2 : mu (merge a b) + 1
              = omega0 ^ (3 : Ordinal) * (mu a + 1)
                + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2 := by
        simp [mu]
        <;> rw [add_assoc]
      rw [h2]
      have h3 : omega0 ^ (4 : Ordinal)
                * (omega0 ^ (3 : Ordinal) * (mu a + 1)
                   + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2)
              = omega0 ^ (7 : Ordinal) * (mu a + 1)
                + omega0 ^ (6 : Ordinal) * (mu b + 1)
                + omega0 ^ (4 : Ordinal) * 2 := by
        rw [mul_add, mul_add]
        congr
        · rw [← mul_assoc]; congr 1; rw [← opow_add]; simp
        · rw [← mul_assoc]; congr 1; rw [← opow_add]; simp
        · rfl
      rw [h3]
      -- Each term on the left is < omega^(mu a + mu b + 9):
      --   omega^7*(mu a+1) < omega^(mu a+8) ≤ omega^(mu a+mu b+9)
      --   omega^6*(mu b+1) < omega^(mu b+7) ≤ omega^(mu a+mu b+9)
      --   omega^4*2       < omega^5         ≤ omega^(mu a+mu b+9)
      sorry
    exact add_lt_add_right h1 1
```
