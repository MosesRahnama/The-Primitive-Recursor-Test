Here is a proof skeleton for the three remaining cases.  The main ideas are:

* **`R_rec_succ`** – show that `mu (delta n) > 2 + mu n`, hence the exponent `mu (delta n) + mu s + 6` on the RHS is *strictly* larger than the largest exponent `2 + mu n + mu s + 6` appearing in the LHS.  Since `omega ^ (large exponent)` is additively principal, it swallows the entire LHS.
* **`R_eq_refl`** – a routine positivity argument, identical in spirit to the `R_int_delta` case.
* **`R_eq_diff`** – expand `mu (integrate (merge a b))` using left-distributivity.  Each of the three resulting summands has an exponent (at most `7`, `6`, or `4`) that is dominated by `mu a + mu b + 9`.  Because `omega ^ (mu a + mu b + 9)` is additively principal, the whole sum sits strictly below it.

```lean
-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    have h_delta_n : mu (delta n) > 2 + mu n := by
      simp [mu]
      by_cases h : mu n < omega0
      · -- mu n is finite, so RHS is infinite
        sorry
      · -- mu n is infinite: 2 + mu n = mu n, and omega^5*(mu n+1)+1 > mu n
        sorry
    have h_exp : 2 + mu n + mu s + 6 < mu (delta n) + mu s + 6 := by
      have h1 : 2 + mu n < mu (delta n) := h_delta_n
      have h2 : (2 + mu n) + mu s < mu (delta n) + mu s :=
        add_lt_add_right h1 (mu s)
      linarith
    have h_principal : AdditivelyPrincipal (omega0 ^ (mu (delta n) + mu s + 6)) := by
      apply principal_opow_omega0
      sorry
    have h_lhs_lt : omega0 ^ (3 : Ordinal) * (mu s + 1)
                    + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1)
                    + 1
                    < omega0 ^ (mu (delta n) + mu s + 6) := by
      -- The dominant LHS term is
      --   omega^2 * omega^(mu n + mu s + 6) = omega^(2 + mu n + mu s + 6).
      -- Because h_exp gives a strict inequality of exponents,
      -- this dominant term is already < omega^(mu (delta n) + mu s + 6).
      -- The remaining summands are even smaller, so additive principality closes the goal.
      sorry
    calc mu (app s (recDelta b s n))
      _ = omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := by simp [mu]
      _ < omega0 ^ (mu (delta n) + mu s + 6) := h_lhs_lt
      _ ≤ omega0 ^ (mu (delta n) + mu s + 6) + omega0 * (mu b + 1) + 1 := le_add_left _ _

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    have h_principal : AdditivelyPrincipal (omega0 ^ (mu a + mu b + 9)) := by
      apply principal_opow_omega0
      sorry
    have h1 : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1))
              < omega0 ^ (mu a + mu b + 9) := by
      -- omega^7 * (mu a + 1) < omega^(mu a + mu b + 9)
      -- because mu a + 1 < mu a + mu b + 9 (as ordinals) and omega^7 * x ≤ omega^(7 + ...)
      sorry
    have h2 : omega0 ^ (4 : Ordinal) * (omega0 ^ (2 : Ordinal) * (mu b + 1))
              < omega0 ^ (mu a + mu b + 9) := by
      -- omega^6 * (mu b + 1) < omega^(mu a + mu b + 9)
      sorry
    have h3 : omega0 ^ (4 : Ordinal) * (2 : Ordinal)
              < omega0 ^ (mu a + mu b + 9) := by
      -- omega^4 * 2 < omega^5 ≤ omega^(mu a + mu b + 9)
      sorry
    have h4 : omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
              < omega0 ^ (mu a + mu b + 9) := by
      rw [mul_add, mul_add]
      -- h_principal lets us add the three strict inequalities h1, h2, h3
      sorry
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := by simp [mu]
      _ < omega0 ^ (mu a + mu b + 9) := h4
      _ < omega0 ^ (mu a + mu b + 9) + 1 := lt_add_one _
```
