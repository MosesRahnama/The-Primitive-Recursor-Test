I'll provide proof skeletons for the three remaining cases in the `mu_decreases` theorem. These proofs rely on ordinal arithmetic properties, particularly the dominance of higher ω-powers.

### 1. `R_merge_cancel` case
```lean
| _, _, Step.R_merge_cancel t => by
    simp [mu]
    have h1 : mu t < mu t + 1 := lt_add_one (mu t)
    have h2 : mu t + 1 ≤ omega0 ^ (2 : Ordinal) * (mu t + 1) :=
      le_mul_right _ (opow_pos 2 omega0_pos)
    have h3 : mu t < omega0 ^ (2 : Ordinal) * (mu t + 1) := lt_of_lt_of_le h1 h2
    have h4 : omega0 ^ (2 : Ordinal) * (mu t + 1) ≤
              omega0 ^ (3 : Ordinal) * (mu t + 1) :=
      le_mul_right _ (opow_pos 1 omega0_pos)
    calc mu t
      _ < omega0 ^ (2 : Ordinal) * (mu t + 1) := h3
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) := h4
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) :=
          le_add_right _ _
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) + 1 :=
          le_add_right _ _
```

### 3. `R_rec_succ` case
```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- The key is that ω^(ω^5 * (μ n + 1) + μ s + 7) dominates all terms on the RHS
    have h1 : omega0 ^ (mu n + mu s + 6) <
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) :=
      begin
        apply opow_lt_opow_left omega0_pos
        apply add_lt_add_right
        apply add_lt_add_right
        apply opow_lt_opow_left (by linarith [omega0_pos])
        linarith
      end
    have h2 : omega0 * (mu b + 1) <
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) :=
      begin
        apply mul_lt_opow
        exact omega0_pos
        exact lt_add_one (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 6)
      end
    calc mu (app s (recDelta b s n))
      _ = omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := rfl
      _ ≤ omega0 ^ (3 : Ordinal) * (mu s + 1) +
         omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) + 1 := by
          { apply add_le_add_left
            apply mul_le_mul_of_nonneg_right
            { apply add_le_add_right
              apply le_add_right }
            { exact opow_nonneg 2 omega0_pos } }
      _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) :=
          begin
            apply add_lt_add_right
            apply add_lt_add_right
            apply opow_lt_opow_left omega0_pos
            apply add_lt_add_of_lt_of_le
            { apply add_lt_add_right
              apply opow_lt_opow_left (by linarith [omega0_pos])
              linarith }
            { apply le_add_right }
          end
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7) +
           omega0 * (mu b + 1) + 1 := le_add_right _ _
```

### 4. `R_eq_diff` case
```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- The key is that ω^(μ a + μ b + 9) dominates ω^4 * (ω^3 * (μ a + 1) + ω^2 * (μ b + 1) + 2)
    have h1 : omega0 ^ (3 : Ordinal) * (mu a + 1) <
              omega0 ^ (mu a + mu b + 9) :=
      begin
        apply mul_lt_opow
        exact omega0_pos
        apply add_lt_add_right
        apply opow_lt_opow_left (by linarith [omega0_pos])
        linarith
      end
    have h2 : omega0 ^ (2 : Ordinal) * (mu b + 1) <
              omega0 ^ (mu a + mu b + 9) :=
      begin
        apply mul_lt_opow
        exact omega0_pos
        apply add_lt_add_left
        apply opow_lt_opow_left (by linarith [omega0_pos])
        linarith
      end
    have h3 : 2 < omega0 ^ (mu a + mu b + 9) :=
      begin
        apply lt_opow_of_lt (by linarith [omega0_pos])
        linarith
      end
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := rfl
      _ = omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) +
                                    omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1 := rfl
      _ < omega0 ^ (4 : Ordinal) * (omega0 ^ (mu a + mu b + 9)) :=
          begin
            apply mul_lt_mul_of_pos_left
            { apply add_lt_add
              { apply add_lt_add
                { exact h1 }
                { exact h2 } }
              { exact lt_add_one _ } }
            { exact opow_pos 4 omega0_pos }
          end
      _ < omega0 ^ (mu a + mu b + 9) :=
          begin
            apply opow_lt_opow_left omega0_pos
            apply add_lt_add_right
            apply opow_lt_opow_left (by linarith [omega0_pos])
            linarith
          end
      _ ≤ omega0 ^ (mu a + mu b + 9) + 1 := le_add_right _ _
```

These proofs rely on several key ordinal arithmetic properties:
1. Higher ω-powers dominate lower ones (ω^(α+1) > ω^α)
2. Multiplication by ω increases the ordinal (ω * α > α for α < ω^ω)
3. Addition of positive terms preserves order
4. The exponential function is strictly increasing in its exponent

The most complex cases are `R_rec_succ` and `R_eq_diff` where we need to show that the exponent on the left side is sufficiently large to dominate all terms on the right side.
