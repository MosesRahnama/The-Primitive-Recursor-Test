### 1) `R_rec_succ` skeleton

  -- ω^3*(mu s+1) + ω^2*( (ω^(mu n + mu s + 6) + ω*(mu b+1) + 1) + 1 ) + 1
  --   < ω^(mu (delta n) + mu s + 6) + ω*(mu b+1) + 1

  -- 1) exponent growth from n to delta n
  have hn_lt_delta : mu n < mu (delta n) := by

    have h1 : mu n < mu n + 1 := lt_add_one (mu n)
    have h2 : mu n + 1 ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) :=
      le_mul_right _ (opow_pos 5 omega0_pos)
    -- now pass through +1 in mu(delta n)
    -- mu(delta n) = ω^5*(mu n+1)+1
    exact lt_of_lt_of_le h1 (le_trans h2 (le_add_right _ _))

  have hexp :
      mu n + mu s + (6 : Ordinal) < mu (delta n) + mu s + (6 : Ordinal) := by
    exact add_lt_add_right (add_lt_add_right hn_lt_delta (mu s)) (6 : Ordinal)

  --  exponents are < the bigger principal term)
