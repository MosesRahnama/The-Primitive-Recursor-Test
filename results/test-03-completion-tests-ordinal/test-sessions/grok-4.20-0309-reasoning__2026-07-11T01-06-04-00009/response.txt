-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- Bound RHS by a single power slightly larger than its dominant term:
    --   ω³·(μs+1) + ω²·(μ(rec..n)+2) + 2 < ω^(μn + μs + 9)
    have h_rhs : mu (app s (recDelta b s n)) < omega0 ^ (mu n + mu s + 9) := by
      calc
        _ ≤ omega0 ^ 3 * (mu s + 1) + omega0 ^ 2 * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 2)) + 2 :=
            by simp [mu]; apply add_le_add; apply add_le_add_right; exact le_rfl
        _ < omega0 ^ (mu n + mu s + 7) + omega0 ^ (mu n + mu s + 7) :=
            by
              apply add_lt_add
              · exact lt_of_le_of_lt (mul_le_opow ..) (opow_lt_opow_left omega0_pos (by simp_arith))
              · refine lt_of_le_of_lt (mul_le_opow ..) ?_
                apply opow_lt_opow_left omega0_pos
                apply add_lt_add_right
                exact lt_add_of_le_of_pos (le_add_right ..) one_pos
        _ ≤ omega0 ^ (mu n + mu s + 9) := le_opow_add omega0_pos (by decide)
    -- The LHS exponent strictly exceeds the RHS exponent:
    --   μ(δn) + μs + 6 = ω⁵·(μn+1) + (μs+7) > μn + μs + 9
    have h_exp : mu n + mu s + 9 < mu (delta n) + mu s + 6 := by
      simp [mu]
      have h5 : mu n + 1 ≤ omega0 ^ 5 * (mu n + 1) := le_mul_left _ (opow_pos 5 omega0_pos)
      calc
        _ < omega0 ^ 5 * (mu n + 1) + (mu s + 7) := lt_add_of_le_of_pos h5 one_pos
        _ ≤ mu (delta n) + mu s + 6 := by simp_arith
    have h_pow : omega0 ^ (mu n + mu s + 9) < omega0 ^ (mu (delta n) + mu s + 6) :=
      opow_lt_opow_left omega0_pos h_exp
    calc
      mu (app s (recDelta b s n)) < omega0 ^ (mu n + mu s + 9) := h_rhs
      _ < omega0 ^ (mu (delta n) + mu s + 6) := h_pow
      _ ≤ omega0 ^ (mu (delta n) + mu s + 6) + omega0 * (mu b + 1) + 1 := by
            apply le_add_right; apply le_add_right; exact le_rfl

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- First bound the merged argument:
    have h_merge : mu (merge a b) + 2 < omega0 ^ (max (mu a) (mu b) + 4) := by
      simp [mu]
      have h3 : omega0 ^ 3 * (mu a + 1) ≤ omega0 ^ (max (mu a) (mu b) + 3) :=
        le_opow_of_le_max (by decide) (le_max_left ..)
      have h2 : omega0 ^ 2 * (mu b + 1) ≤ omega0 ^ (max (mu a) (mu b) + 3) :=
        le_opow_of_le_max (by decide) (le_max_right ..)

        _ ≤ omega0 ^ (max (mu a) (mu b) + 3) + omega0 ^ (max (mu a) (mu b) + 3) + 2 :=
            add_le_add (add_le_add h3 h2) (by decide)
