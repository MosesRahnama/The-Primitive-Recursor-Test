The three remaining cases can be organized by reducing each goal to a positivity or domination lemma. The `eqW` reflexive case is immediate; the other two are best handled by introducing abbreviations for the relevant exponents and proving the main ordinal bound separately.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    let E : Ordinal :=
      mu (delta n) + mu s + (6 : Ordinal)

    let R : Ordinal :=
      omega0 ^ (mu n + mu s + (6 : Ordinal))
        + omega0 * (mu b + 1) + 1

    /*
      After unfolding, the goal has the form

        omega0 ^ 3 * (mu s + 1)
          + omega0 ^ 2 * (R + 1) + 1
        <
        omega0 ^ E + omega0 * (mu b + 1) + 1.

      First show that the measure of the `app` term is below the
      leading omega-power on the right.
    */
    have h_app :
        omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) * (R + 1) + 1
        < omega0 ^ E := by
      have h_exp :
          (mu n + mu s + (6 : Ordinal)) + 2 < E := by
        dsimp [E]
        simp [mu]
        /*
          Use that

            mu (delta n) = omega0 ^ 5 * (mu n + 1) + 1

          and that the omega-power term strictly dominates
          `mu n + mu s + 8`.
        */
        sorry

      have h_rec :
          R + 1 < omega0 ^ (mu n + mu s + (6 : Ordinal)) + 2 := by
        dsimp [R]
        /*
          Bound the lower-order terms of `R + 1` by an omega-power.
          This is the usual Cantor-normal-form estimate.
        */
        sorry

      /*
        Apply monotonicity of ordinal exponentiation and multiplication,
        then combine the two summands using the fact that both have
        exponent strictly below `E`.
      */
      sorry

    calc
      omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) * (R + 1) + 1
          < omega0 ^ E := h_app
      _ ≤ omega0 ^ E + omega0 * (mu b + 1) + 1 :=
        le_add_right _ _


-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)


-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    let A : Ordinal := mu a
    let B : Ordinal := mu b
    let E : Ordinal := A + B + (9 : Ordinal)

    /*
      The unfolded goal is

        omega0 ^ 4 *
            (omega0 ^ 3 * (A + 1)
              + omega0 ^ 2 * (B + 1) + 2) + 1
        <
        omega0 ^ E + 1.
    */

    have h_core :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (A + 1)
              + omega0 ^ (2 : Ordinal) * (B + 1) + 2)
        < omega0 ^ E := by
      have hA :
          omega0 ^ (4 : Ordinal) *
              (omega0 ^ (3 : Ordinal) * (A + 1))
            < omega0 ^ E := by
        dsimp [E]
        /*
          Normalize the left side to an omega-power term, then use
          `A + 7 < A + B + 9` after accounting for the lower-order
          factors.  In particular, the constants 3 and 4 contribute
          only a finite shift in the Cantor-normal-form estimate.
        */
        sorry

      have hB :
          omega0 ^ (4 : Ordinal) *
              (omega0 ^ (2 : Ordinal) * (B + 1))
            < omega0 ^ E := by
        dsimp [E]
        /*
          Same estimate for the second merge component.
        */
        sorry

      have h_two :
          omega0 ^ (4 : Ordinal) * 2 < omega0 ^ E := by
        dsimp [E]
        /*
          Use positivity of `A + B + 9` and the fact that every finite
          omega-power below it is dominated by `omega0 ^ E`.
        */
        sorry

      /*
        Combine the three strict bounds using ordinal addition
        monotonicity and the fact that all summands have exponent
        strictly below `E`.
      */
      sorry

    calc
      omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (A + 1)
            + omega0 ^ (2 : Ordinal) * (B + 1) + 2) + 1
          < omega0 ^ E + 1 :=
        add_lt_add_right h_core _
      _ = omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
        rfl
```

The important correction to the comments in the successor case is that the left-hand side is the measure of

```lean
app s (recDelta b s n)
```

namely

```lean
omega0 ^ 3 * (mu s + 1)
  + omega0 ^ 2 * (mu (recDelta b s n) + 1) + 1
```

rather than an omega-power whose exponent is `mu (delta n) + mu s + 6`. The latter is the leading term of the right-hand side, `mu (recDelta b s (delta n))`.
