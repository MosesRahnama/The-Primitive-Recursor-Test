Here are the three remaining cases, written so that they slot directly into the `mu_decreases` proof.  Each case reduces to showing that the recursive arguments of the reduct are bounded by a single positive power of `omega0`; finite sums of smaller ordinals stay below such a power, and the final `+ 1` is preserved by strict monotonicity.

```tactics
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- Both sides end in `+ 1`; strict monotonicity of successor lets us drop it.
    have h1 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu (delta n) + mu s + 6) := by
      have h2 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu s + 4) := by
        have hs : mu s + 1 < omega0 ^ (mu s + 1) := by
          -- Every successor lies strictly below the next power of omega.
          sorry
        have h3 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (3 : Ordinal) * omega0 ^ (mu s + 1) := by
          apply mul_lt_mul_of_pos_left hs (opow_pos 3 omega0_pos)
        have h4 : omega0 ^ (3 : Ordinal) * omega0 ^ (mu s + 1) = omega0 ^ (3 + mu s + 1) := by
          rw [← opow_add]
        have h5 : (3 : Ordinal) + mu s + 1 ≤ mu s + 4 := by
          -- 3 + α ≤ α + 3 for all ordinals, hence 3 + α + 1 ≤ α + 4
          sorry
        exact lt_of_lt_of_le (h3.trans_le (le_of_eq h4)) (opow_le_opow_right omega0_pos h5)
      have h6 : omega0 ^ (mu s + 4) ≤ omega0 ^ (mu (delta n) + mu s + 6) := by
        apply opow_le_opow_right omega0_pos
        -- μ s + 4 ≤ μ(δ n) + μ s + 6 because 4 ≤ μ(δ n) + 6
        sorry
      exact lt_of_lt_of_le h2 h6
    have h7 : omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) < omega0 ^ (mu (delta n) + mu s + 6) := by
      have h8 : mu (recDelta b s n) + 1 < omega0 ^ (mu (delta n) + mu s + 6) := by
        simp [mu]
        have h9 : omega0 ^ (mu n + mu s + 6) < omega0 ^ (mu (delta n) + mu s + 6) := by
          apply opow_lt_opow_right omega0_pos
          -- μ n < μ(δ n) implies μ n + μ s + 6 < μ(δ n) + μ s + 6
          sorry
        have h10 : omega0 * (mu b + 1) < omega0 ^ (mu (delta n) + mu s + 6) := by
          sorry
        have h11 : (2 : Ordinal) < omega0 ^ (mu (delta n) + mu s + 6) := by
          sorry
        -- A finite sum of ordinals below a positive power of omega stays below it.
        sorry
      have h12 : omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) < omega0 ^ (2 : Ordinal) * omega0 ^ (mu (delta n) + mu s + 6) := by
        apply mul_lt_mul_of_pos_left h8 (opow_pos 2 omega0_pos)
      have h13 : omega0 ^ (2 : Ordinal) * omega0 ^ (mu (delta n) + mu s + 6) = omega0 ^ (mu (delta n) + mu s + 6) := by
        rw [← opow_add]
        -- 2 + X = X because X ≥ ω is infinite
        sorry
      rwa [h13] at h12
    have h14 : omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1)
               < omega0 ^ (mu (delta n) + mu s + 6) := by
      -- Sum of two ordinals below a positive power of omega stays below it.
      sorry
    calc mu (app s (recDelta b s n))
        = omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := by simp [mu]
      _ < omega0 ^ (mu (delta n) + mu s + 6) + 1 := by exact add_lt_add_right h14 1
      _ ≤ omega0 ^ (mu (delta n) + mu s + 6) + omega0 * (mu b + 1) + 1 := by
          apply add_le_add_right
          apply le_add_right

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact add_pos_right _ (opow_pos (mu a + mu a + 9) omega0_pos)

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- It suffices to show ω⁴·(μ(merge a b)+1) < ω^(μ a + μ b + 9); adding 1 preserves the strict inequality.
    have h1 : mu (merge a b) + 1 < omega0 ^ (mu a + mu b + 5) := by
      simp [mu]
      have ha : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + mu b + 5) := by
        have ha1 : mu a + 1 < omega0 ^ (mu a + 1) := by
          -- Every successor lies strictly below the next power of omega.
          sorry
        have ha2 : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ (3 : Ordinal) * omega0 ^ (mu a + 1) := by
          apply mul_lt_mul_of_pos_left ha1 (opow_pos 3 omega0_pos)
        have ha3 : omega0 ^ (3 : Ordinal) * omega0 ^ (mu a + 1) = omega0 ^ (3 + mu a + 1) := by
          rw [← opow_add]
        have ha4 : (3 : Ordinal) + mu a + 1 ≤ mu a + 4 := by
          -- 3 + α ≤ α + 3, hence 3 + α + 1 ≤ α + 4
          sorry
        have ha5 : omega0 ^ (mu a + 4) < omega0 ^ (mu a + mu b + 5) := by
          apply opow_lt_opow_right omega0_pos
          -- α + 4 < α + β + 5 because 4 < β + 5
          sorry
        exact lt_of_lt_of_le (ha2.trans_le (le_of_eq ha3)) (opow_le_opow_right omega0_pos ha4).trans ha5
      have hb : omega0 ^ (2 : Ordinal) * (mu b + 1) < omega0 ^ (mu a + mu b + 5) := by
        -- Symmetric to ha
        sorry
      have h2 : (2 : Ordinal) < omega0 ^ (mu a + mu b + 5) := by
        -- 2 < ω ≤ ω^(positive)
        sorry
      -- A finite sum of ordinals below a positive power of omega stays below it.
      sorry
    have h3 : omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) < omega0 ^ (mu a + mu b + 9) := by
      calc omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
          < omega0 ^ (4 : Ordinal) * omega0 ^ (mu a + mu b + 5) := by
            apply mul_lt_mul_of_pos_left h1 (opow_pos 4 omega0_pos)
        _ = omega0 ^ (4 + (mu a + mu b + 5)) := by rw [← opow_add]
        _ ≤ omega0 ^ (mu a + mu b + 9) := by
            apply opow_le_opow_right omega0_pos
            -- 4 + α ≤ α + 4, so 4 + (μ a + μ b + 5) ≤ μ a + μ b + 9
            sorry
    exact add_lt_add_right h3 1
```
