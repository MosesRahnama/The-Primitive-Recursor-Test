-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact add_pos_right _ (opow_pos _ omega0_pos)

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- Distribute omega^4 over the expanded merge measure
    have h_distr : omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
        = omega0 ^ (7 : Ordinal) * (mu a + 1) + omega0 ^ (6 : Ordinal) * (mu b + 1)
          + omega0 ^ (4 : Ordinal) * (2 : Ordinal) := by
      simp [mu, mul_add, ← opow_add]
      <;> norm_num
      <;> ring_nf <;> norm_num
    rw [h_distr]
    -- Each distributed term is bounded below omega^(mu a + mu b + 9)
    have h1 : omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      have h_a : mu a + 1 < omega0 ^ (mu a + 1) := by
        -- A standard fact: every ordinal is strictly below omega raised to its
        -- own successor.  This can be shown by induction on the ordinal.
        sorry
      have h7 : omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ (7 : Ordinal) * omega0 ^ (mu a + 1) :=
        mul_lt_mul_of_pos_left h_a (opow_pos 7 omega0_pos)
      rw [← opow_add] at h7
      norm_num at h7
      have h8 : (7 : Ordinal) + (mu a + 1) = mu a + (8 : Ordinal) := by
        rw [add_comm (7 : Ordinal), add_assoc]; norm_num
      rw [h8] at h7
      have h9 : mu a + (8 : Ordinal) ≤ mu a + mu b + (9 : Ordinal) := by
        apply add_le_add_left
        -- 8 ≤ mu b + 9 because mu b ≥ 0
        sorry
      exact lt_of_lt_of_le h7 (opow_le_opow_right omega0_pos h9)
    have h2 : omega0 ^ (6 : Ordinal) * (mu b + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      -- Symmetric to h1
      sorry
    have h3 : omega0 ^ (4 : Ordinal) * (2 : Ordinal) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      have h4 : 4 < mu a + mu b + (9 : Ordinal) := by
        -- The right-hand exponent is at least 9
        sorry
      have h5 : omega0 ^ (4 : Ordinal) * (2 : Ordinal) < omega0 ^ (4 : Ordinal) * omega0 := by
        apply mul_lt_mul_of_pos_left
        · norm_num
        · exact opow_pos 4 omega0_pos
      have h6 : omega0 ^ (4 : Ordinal) * omega0 = omega0 ^ (5 : Ordinal) := by
        rw [← opow_add]; norm_num
      rw [h6] at h5
      have h7 : omega0 ^ (5 : Ordinal) ≤ omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
        apply opow_le_opow_right omega0_pos
        -- 5 ≤ mu a + mu b + 9
        sorry
      exact lt_of_lt_of_le h5 h7
    -- The target ordinal is a power of omega (> 0), hence additively principal
    have h_principal : Principal Add (omega0 ^ (mu a + mu b + (9 : Ordinal))) := by
      apply principal_add_omega0_opow
      have : 0 < mu a + mu b + (9 : Ordinal) := by
        -- The exponent is at least 9
        sorry
      exact this
    have h_sum : omega0 ^ (7 : Ordinal) * (mu a + 1) + (omega0 ^ (6 : Ordinal) * (mu b + 1)
                   + omega0 ^ (4 : Ordinal) * (2 : Ordinal)) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      apply h_principal
      · apply h_principal
        · exact h1
        · exact h2
      · exact h3
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := by simp [mu]
      _ < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
          apply Order.succ_lt_succ h_sum
      _ = mu (eqW a b) := by simp [mu]

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu, mul_add, ← opow_add]
    norm_num
    -- The RHS leading term omega^(mu(delta n) + mu s + 6) dominates the
    -- entire LHS because mu(delta n) = omega^5 * (mu n + 1) + 1 is built
    -- to swallow the mu n + mu s + 8 exponent appearing on the left.
    have h_dom : omega0 ^ (3 : Ordinal) * (mu s + 1) + (omega0 ^ (mu n + mu s + (8 : Ordinal))
        + omega0 ^ (3 : Ordinal) * (mu b + 1) + omega0 ^ (2 : Ordinal) * (2 : Ordinal))
        < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
      have h_exp : mu n + mu s + (8 : Ordinal) < mu (delta n) + mu s + (6 : Ordinal) := by
        -- Key ordinal inequality: the delta-wrapper exponent is large enough
        -- to exceed the unwrapped exponent even after adding mu s.
        sorry
      have h_lead : omega0 ^ (mu n + mu s + (8 : Ordinal)) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
        apply opow_lt_opow_right omega0_pos h_exp
      have h_t1 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
        -- omega^3 * (mu s + 1) < omega^(mu s + 4) ≤ omega^(mu(delta n) + mu s + 6)
        sorry
      have h_t2 : omega0 ^ (3 : Ordinal) * (mu b + 1) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
        -- Bounded by the principal term since mu(delta n) + mu s + 6 > mu b + 3
        sorry
      have h_t3 : omega0 ^ (2 : Ordinal) * (2 : Ordinal) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
        -- 2 < 6, so omega^2 * 2 < omega^3 ≤ ...
        sorry
      have h_pr : Principal Add (omega0 ^ (mu (delta n) + mu s + (6 : Ordinal))) := by
        apply principal_add_omega0_opow
        have : 0 < mu (delta n) + mu s + (6 : Ordinal) := by
          -- The exponent is at least 6
          sorry
        exact this
      apply h_pr
      · apply h_pr
        · exact h_t1
        · exact h_lead
      · apply h_pr
        · exact h_t2
        · exact h_t3
    calc mu (app s (recDelta b s n))
      _ = omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (mu n + mu s + (8 : Ordinal))
          + omega0 ^ (3 : Ordinal) * (mu b + 1) + omega0 ^ (2 : Ordinal) * (2 : Ordinal) + 1 := by
          simp [mu, mul_add, ← opow_add]
          <;> norm_num
          <;> ring_nf <;> norm_num
      _ < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + 1 := by
          apply Order.succ_lt_succ h_dom
      _ ≤ omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := by
          apply le_add_left
          apply le_add_left
          exact le_mul_right _ omega0_pos
      _ = mu (recDelta b s (delta n)) := by simp [mu]
