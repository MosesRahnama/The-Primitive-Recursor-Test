-- merge t t --> t
| _, _, Step.R_merge_cancel t => by
    simp [mu]
    -- Goal: mu t < ω³·(μ t + 1) + ω²·(μ t + 1) + 1
    have h₁ : μ t < μ t + 1 := lt_add_one _
    have h₂ : μ t + 1 ≤ ω³·(μ t + 1) := le_mul_right _ (opow_pos 3 omega0_pos)
    have h₃ : μ t < ω³·(μ t + 1) := lt_of_lt_of_le h₁ h₂
    calc μ t < ω³·(μ t + 1) := h₃
         _ ≤ ω³·(μ t + 1) + ω²·(μ t + 1) := le_add_right _ _
         _ ≤ ω³·(μ t + 1) + ω²·(μ t + 1) + 1 := le_add_right _ _

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    simp [mu]
    -- Goal: μ b < ω^(0 + μ s + 6) + ω·(μ b + 1) + 1
    have h₁ : μ b < μ b + 1 := lt_add_one _
    have h₂ : μ b + 1 ≤ ω·(μ b + 1) := le_mul_right _ omega0_pos
    have h₃ : μ b < ω·(μ b + 1) := lt_of_lt_of_le h₁ h₂
    calc μ b < ω·(μ b + 1) := h₃
         _ ≤ ω^(μ s + 6) + ω·(μ b + 1) := le_add_left _ _
         _ ≤ ω^(μ s + 6) + ω·(μ b + 1) + 1 := le_add_right _ _

    -- i.e. ω⁴·(ω³·(μ a + 1) + ω²·(μ b + 1) + 2) + 1
