A reasonable way to finish the file is to isolate the nontrivial ordinal-arithmetic estimates as local helper facts. The rule cases then become short applications of those estimates.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    /*
      After simplification, the goal has the shape

        ω^3 * (mu s + 1)
          + ω^2 * (ω^(mu n + mu s + 6)
                    + ω * (mu b + 1) + 1 + 1)
          + 1
        <
        ω^(mu (delta n) + mu s + 6)
          + ω * (mu b + 1) + 1

      It is enough to show that the left-hand side is strictly below
      the leading ω-power on the right.
    */

    have hδ :
        mu n + mu s + 6 <
          mu (delta n) + mu s + 6 := by
      simp [mu]
      /*
        The essential fact is

          mu n < omega0 ^ 5 * (mu n + 1) + 1 = mu (delta n).

        The remaining additions are monotone.
      */
      have hn :
          mu n < omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 := by
        calc
          mu n < mu n + 1 := lt_add_one _
          _ ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) := by
            exact le_mul_right _ (opow_pos 5 omega0_pos)
          _ ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 := le_add_right _ _
      exact add_lt_add_right (add_lt_add_right hn (mu s)) 6

    have hpow :
        omega0 ^ (mu n + mu s + 6) <
          omega0 ^ (mu (delta n) + mu s + 6) := by
      exact opow_lt_opow_right' omega0_pos hδ

    /*
      The following is the main “tower dominates lower constructors”
      estimate.  It can be proved by repeatedly using ordinal
      multiplication/addition monotonicity together with hpow.
    */
    have hdom :
        omega0 ^ (3 : Ordinal) * (mu s + 1)
            + omega0 ^ (2 : Ordinal) *
                (omega0 ^ (mu n + mu s + 6)
                  + omega0 * (mu b + 1) + 1 + 1)
            + 1
          <
        omega0 ^ (mu (delta n) + mu s + 6) := by
      /*
        Arithmetic subproof:
        - prove each lower-order summand is below the right-hand
          ω-power;
        - combine them using `add_lt_of_lt` / `lt_of_lt_of_le`;
        - use hpow for the recursive summand.
      */
      sorry

    exact lt_of_lt_of_le hdom (le_add_right _ _)
```

The reflexivity case only needs positivity:

```lean
-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]

    have hpos :
        (0 : Ordinal) < omega0 ^ (mu a + mu a + 9) := by
      exact opow_pos _ omega0_pos

    exact lt_of_lt_of_le hpos (le_add_right _ _)
```

For the non-reflexive equality rule, first bound the measure of the expanded term by the leading power in the source measure:

```lean
-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    /*
      The simplified goal is essentially

        ω^4 * (
          ω^3 * (mu a + 1)
            + ω^2 * (mu b + 1) + 1 + 1
        ) + 1
        <
        ω^(mu a + mu b + 9) + 1.
    */

    have hdom :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1)
              + 1 + 1)
          <
        omega0 ^ (mu a + mu b + 9) := by
      /*
        Arithmetic subproof:
        - establish that the exponent `mu a + mu b + 9`
          dominates the finite powers 4, 3, and 2;
        - distribute/associate ordinal multiplication as needed;
        - use monotonicity of `omega0 ^ ·`;
        - absorb the finite trailing summands into the larger
          ω-power.
      */
      sorry

    exact add_lt_add_right hdom 1
```

A useful organization is to move the two `sorry` blocks into standalone arithmetic lemmas before `mu_decreases`, for example:

```lean
lemma rec_succ_bound (b s n : Trace) :
    omega0 ^ (3 : Ordinal) * (mu s + 1)
        + omega0 ^ (2 : Ordinal) *
            (mu (recDelta b s n) + 1)
        + 1
      <
    omega0 ^ (mu (delta n) + mu s + 6) := by
  simp [mu]
  -- ordinal tower comparison
  sorry

lemma eq_diff_bound (a b : Trace) :
    omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
      <
    omega0 ^ (mu a + mu b + 9) := by
  simp [mu]
  -- ordinal tower comparison
  sorry
```

Then the corresponding cases reduce to:

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    have h := rec_succ_bound b s n
    exact lt_of_lt_of_le h (le_add_right _ _)

| _, _, Step.R_eq_diff a b => by
    simp [mu]
    have h := eq_diff_bound a b
    exact add_lt_add_right h 1
```

The exact names of the ordinal monotonicity lemmas may need adjustment, but this is the intended proof structure: prove the leading-`ω`-power domination separately, then finish each rule by monotonicity of ordinal addition.
