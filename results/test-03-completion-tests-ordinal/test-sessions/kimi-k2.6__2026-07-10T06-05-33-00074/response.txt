Here is a proof skeleton for the three remaining cases.  The overall strategy follows the pattern already used for the solved cases: identify the dominant term on the right‑hand side, show that it is additively principal, and then bound every summand of the left‑hand side strictly below it.

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact add_pos_right _ (opow_pos (mu a + mu a + 9) omega0_pos)

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- The exponent mu(delta n) + mu s + 6 is strictly larger than
    -- mu n + mu s + 6 because mu(delta n) > mu n.  Hence the tower
    -- omega^(mu(delta n) + mu s + 6) is additively principal and
    -- dominates the inner recursive call.
    have h1 : mu (recDelta b s n) + 1 < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
      sorry
    have h2 : omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
      sorry
    have h3 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
      sorry
    calc mu (app s (recDelta b s n))
        = omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := by simp [mu]
      _ < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
          -- combine h2, h3 and +1 using the fact that omega^... is additively principal
          sorry
      _ < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) := by
          apply lt_add_right
          exact mul_pos omega0_pos (add_pos_right (mu b) one_pos)
      _ < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := by
          apply lt_add_right
          exact one_pos

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu, mul_add]
    -- mu(merge a b) + 1 expands to a finite sum.  Each summand is
    -- strictly below the additively principal ordinal omega^(mu a + mu b + 9).
    have h1 : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      sorry
    have h2 : omega0 ^ (2 : Ordinal) * (mu b + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      sorry
    have h3 : (2 : Ordinal) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      sorry
    have h4 : omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal)
              < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      -- omega^(mu a + mu b + 9) is additively principal, so a finite sum of smaller ordinals is smaller
      sorry
    have h5 : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal))
              < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      -- Multiply the bound by omega^4.  For infinite exponents omega^4 * omega^alpha = omega^alpha,
      -- while for finite exponents the inequality follows by a direct computation.
      sorry
    calc mu (integrate (merge a b))
        = omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal)) + 1 := by simp [mu, mul_add]
      _ < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
          apply add_lt_add_right
          exact h5
      _ = mu (eqW a b) := by simp [mu]
