The two genuinely difficult cases can be isolated into ordinal-arithmetic inequalities. A proof skeleton is:

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    /*
      After unfolding, the essential inequality is

        ω^3 * (μ s + 1)
          + ω^2 * (ω^(μ n + μ s + 6)
                    + ω * (μ b + 1) + 1 + 1) + 1
        <
        ω^(ω^5 * (μ n + 1) + 1 + μ s + 6).

      This is the ordinal-arithmetic part: the exponent on the right
      dominates every exponent occurring in the polynomial expression
      on the left.
    */
    have hmain :
        omega0 ^ (3 : Ordinal) * (mu s + 1)
            + omega0 ^ (2 : Ordinal) *
                (omega0 ^ (mu n + mu s + 6)
                  + omega0 * (mu b + 1) + 1 + 1) + 1
          <
        omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1)
                  + 1 + mu s + 6) := by
      -- Prove this using:
      --   * monotonicity of ordinal addition and multiplication;
      --   * `opow_lt_opow` after comparing exponents;
      --   * finite powers of `omega0` being absorbed by
      --     `omega0 ^ (omega0 ^ 5 * (mu n + 1) + 1 + mu s + 6)`.
      sorry

    calc
      omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (mu n + mu s + 6)
                + omega0 * (mu b + 1) + 1 + 1) + 1
          <
        omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1)
                  + 1 + mu s + 6) := hmain
      _ ≤
        omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1)
                  + 1 + mu s + 6)
          + omega0 * (mu b + 1) + 1 :=
        le_add_right _ _

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]

    have hpos :
        0 < omega0 ^ (mu a + mu a + 9) + 1 := by
      exact lt_of_lt_of_le zero_lt_one (le_add_left _ _)

    exact hpos

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    /*
      After unfolding, it remains to prove

        ω^4 *
          (ω^3 * (μ a + 1)
            + ω^2 * (μ b + 1) + 1 + 1) + 1
        <
        ω^(μ a + μ b + 9) + 1.

      The arithmetic lemma needed here is that a finite ω-polynomial
      in x and y is strictly below ω^(x + y + 9).
    */
    have hmain :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1
          <
        omega0 ^ (mu a + mu b + 9) + 1 := by
      -- Prove the polynomial-versus-exponential estimate using
      -- ordinal exponent monotonicity and `opow_add`.
      sorry

    exact hmain
```

A useful way to organize the omitted arithmetic is to prove separate auxiliary lemmas before `mu_decreases`, for example:

```lean
lemma finite_poly_lt_opow
    (x y : Ordinal) :
    omega0 ^ (4 : Ordinal) *
        (omega0 ^ (3 : Ordinal) * (x + 1)
          + omega0 ^ (2 : Ordinal) * (y + 1) + 2) + 1
      <
    omega0 ^ (x + y + 9) + 1 := by
  sorry

lemma rec_app_lt_tower
    (b s n : Trace) :
    omega0 ^ (3 : Ordinal) * (mu s + 1)
        + omega0 ^ (2 : Ordinal) *
            (omega0 ^ (mu n + mu s + 6)
              + omega0 * (mu b + 1) + 2) + 1
      <
    omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1)
              + 1 + mu s + 6) := by
  sorry
```

Then the two difficult cases reduce to:

```lean
    exact lt_of_lt_of_le
      (rec_app_lt_tower b s n)
      (le_add_right _ _)
```

and

```lean
    exact finite_poly_lt_opow (mu a) (mu b)
```

respectively. The `R_eq_refl` case needs no substantial ordinal arithmetic beyond positivity of `ω^α + 1`.
