I'll provide proof skeletons for the three remaining cases in the `mu_decreases` theorem. These proofs rely on ordinal arithmetic properties, particularly the dominance of higher ω-powers.

### 1. `R_merge_cancel` case
```lean
| _, _, Step.R_merge_cancel t => by
    simp [mu]
    have h1 : mu t < mu t + 1 := lt_add_one (mu t)
    have h2 : mu t + 1 ≤ omega0 ^ (2 : Ordinal) * (mu t + 1) :=
      le_mul_right _ (opow_pos 2 omega0_pos)
    have h3 : mu t < omega0 ^ (2 : Ordinal) * (mu t + 1) := lt_of_lt_of_le h1 h2
    have h4 : omega0 ^ (2 : Ordinal) * (mu t + 1) ≤
              omega0 ^ (3 : Ordinal) * (mu t + 1) :=
      mul_le_mul_right _ (opow_le_opow_right omega0_pos (by norm_num))
    calc mu t
      _ < omega0 ^ (2 : Ordinal) * (mu t + 1) := h3
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) := h4
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) :=
          le_add_right _ _
      _ ≤ omega0 ^ (3 : Ordinal) * (mu t + 1) + omega0 ^ (2 : Ordinal) * (mu t + 1) + 1 :=
          le_add_right _ _
```

### 2. `R_rec_succ` case
```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- The key is that ω^(ω^5 * (μn + 1) + ...) dominates all terms in the RHS
    have h1 : mu n + 1 ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) :=
      le_mul_right _ (opow_pos 5 omega0_pos)
    have h2 : omega0 ^ (5 : Ordinal) * (mu n + 1) <
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6)) :=
      opow_lt_opow_left omega0_pos (by
        calc 5 < omega0 := by norm_num
          _ ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) :=
              le_self_mul (opow_pos 5 omega0_pos) (mu n + 1)
          _ ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6) :=
              le_add_right _ _)
    have h3 : omega0 ^ (mu n + mu s + 6) <
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6)) :=
      opow_lt_opow_left omega0_pos (by
        calc mu n + mu s + 6 ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6) :=
              add_le_add_right (le_mul_right _ (opow_pos 5 omega0_pos)) _
          _ < omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6) :=
              lt_add_of_pos_right _ (by simp [add_pos_iff, or_true]))
    have h4 : omega0 * (mu b + 1) <
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6)) :=
      mul_lt_opow omega0_pos (by
        calc 1 < omega0 ^ (5 : Ordinal) * (mu n + 1) :=
              opow_pos 5 omega0_pos
          _ ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6) :=
              le_add_right _ _)
    calc mu (app s (recDelta b s n))
      _ = omega0 ^ (3 : Ordinal) * (mu s + 1) +
          omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := rfl
      _ ≤ omega0 ^ (3 : Ordinal) * (mu s + 1) +
          omega0 ^ (2 : Ordinal) *
          (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) + 1 := by
          apply add_le_add_left
          apply mul_le_mul_right
          apply add_le_add_right
          apply le_of_lt
          simp [mu]
      _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + (mu s + 6)) := by
          apply lt_of_le_of_lt (add_le_add_right (le_refl _) _)
          apply add_lt_add_left
          apply mul_lt_opow omega0_pos
          apply lt_of_lt_of_le (add_lt_add_of_lt_of_le h3 h4)
          apply le_self_add
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + 6) +
          omega0 * (mu b + 1) + 1 := by
          apply le_add_right
      _ = mu (recDelta b s (delta n)) := rfl
```

### 4. `R_eq_diff` case
```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- The key is that ω^(μa + μb + 9) dominates all terms in the LHS
    have h1 : omega0 ^ (3 : Ordinal) * (mu a + 1) <
              omega0 ^ (mu a + mu b + 9) :=
      mul_lt_opow omega0_pos (by
        calc 3 < mu a + mu b + 9 := by
              apply lt_add_of_pos_right
              apply add_pos_of_pos_of_nonneg
              exact Ordinal.pos_iff_ne_zero.mpr (ne_of_gt (mu_pos a))
              exact le_add_of_nonneg_right (Ordinal.zero_le (mu b)))
    have h2 : omega0 ^ (2 : Ordinal) * (mu b + 1) <
              omega0 ^ (mu a + mu b + 9) :=
      mul_lt_opow omega0_pos (by
        calc 2 < mu a + mu b + 9 := by
              apply lt_add_of_pos_right
              apply add_pos_of_pos_of_nonneg
              exact Ordinal.pos_iff_ne_zero.mpr (ne_of_gt (mu_pos a))
              exact le_add_of_nonneg_right (Ordinal.zero_le (mu b)))
    have h3 : 2 < omega0 ^ (mu a + mu b + 9) :=
      lt_opow_self omega0_pos (by
        calc 2 < mu a + mu b + 9 := by
              apply lt_add_of_pos_right
              apply add_pos_of_pos_of_nonneg
              exact Ordinal.pos_iff_ne_zero.mpr (ne_of_gt (mu_pos a))
              exact le_add_of_nonneg_right (Ordinal.zero_le (mu b)))
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := rfl
      _ ≤ omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (mu a + 1) +
           omega0 ^ (2 : Ordinal) * (mu b + 1) + 2) + 1 := by
          apply add_le_add_right
          apply mul_le_mul_right
          apply add_le_add_right
          apply le_of_lt
          simp [mu]
      _ < omega0 ^ (4 : Ordinal) *
          (omega0 ^ (mu a + mu b + 9) +
           omega0 ^ (mu a + mu b + 9) +
           omega0 ^ (mu a + mu b + 9)) + 1 := by
          apply add_lt_add_right
          apply mul_lt_mul_right (opow_pos 4 omega0_pos)
          apply add_lt_add
          · exact h1
          · apply add_lt_add_right h2
          · exact h3
      _ < omega0 ^ (mu a + mu b + 9) := by
          apply lt_opow_self omega0_pos
          apply lt_of_lt_of_le (by norm_num)
          apply le_add_right
      _ ≤ omega0 ^ (mu a + mu b + 9) + 1 := le_add_right _ _
```

These proofs rely on the key ordinal arithmetic properties:
1. Higher ω-powers dominate lower ones (ω^(k+1) > ω^k)
2. Multiplication by ω increases the ordinal's magnitude
3. Addition of positive terms preserves order
4. The exponential function is strictly increasing for ordinals

The most complex cases (`R_rec_succ` and `R_eq_diff`) require showing that the exponent in the reduced term is large enough to dominate all terms in the original measure.
