I'll provide proof skeletons for the three remaining cases in the `mu_decreases` theorem. These proofs rely on ordinal arithmetic properties, particularly the dominance of higher ω-powers.

### 1. `R_merge_cancel` case
```lean
| _, _, Step.R_merge_cancel t => by
    simp [mu]
    -- We need to show: μ t < ω³*(μ t + 1) + ω²*(μ t + 1) + 1
    have h1 : μ t < μ t + 1 := lt_add_one (μ t)
    have h2 : μ t + 1 ≤ ω² * (μ t + 1) := le_mul_right _ (opow_pos 2 omega0_pos)
    have h3 : μ t < ω² * (μ t + 1) := lt_of_lt_of_le h1 h2
    have h4 : ω² * (μ t + 1) ≤ ω³ * (μ t + 1) :=
      le_mul_right _ (opow_pos (1 : Ordinal) omega0_pos)
    calc μ t
      _ < ω² * (μ t + 1) := h3
      _ ≤ ω³ * (μ t + 1) := h4
      _ ≤ ω³ * (μ t + 1) + ω² * (μ t + 1) := le_add_right _ _
      _ ≤ ω³ * (μ t + 1) + ω² * (μ t + 1) + 1 := le_add_right _ _
```

### 2. `R_rec_succ` case
```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- Key insight: ω^(ω⁵*(μ n + 1) + μ s + 7) dominates all terms in RHS
    have h1 : ω⁵ * (μ n + 1) + μ s + 7 > μ n + μ s + 6 := by
      apply add_lt_add_right
      apply add_lt_add_right
      apply mul_lt_mul_of_pos_left _ (opow_pos 5 omega0_pos)
      exact lt_add_one (μ n)
    have h2 : ω^(μ n + μ s + 6) < ω^(ω⁵ * (μ n + 1) + μ s + 7) :=
      opow_lt_opow_of_lt (by simp [Ordinal.pos_iff_ne_zero]; exact one_pos) h1
    have h3 : ω * (μ (recDelta b s n) + 1) < ω^(ω⁵ * (μ n + 1) + μ s + 7) := by
      apply mul_lt_opow _ (opow_pos _ omega0_pos)
      apply add_lt_add_right
      apply lt_of_le_of_lt (le_add_right _ _)
      exact h2
    have h4 : ω³ * (μ s + 1) < ω^(ω⁵ * (μ n + 1) + μ s + 7) := by
      apply mul_lt_opow _ (opow_pos _ omega0_pos)
      apply lt_of_le_of_lt (le_add_right _ _)
      exact h2
    calc μ (app s (recDelta b s n))
      _ = ω³ * (μ s + 1) + ω² * (μ (recDelta b s n) + 1) + 1 := by rfl
      _ ≤ ω³ * (μ s + 1) + ω * (μ (recDelta b s n) + 1) + 1 := by
          apply add_le_add_left
          apply mul_le_mul_right _ (opow_pos 1 omega0_pos)
          apply le_add_right
      _ < ω^(ω⁵ * (μ n + 1) + μ s + 7) := by
          apply lt_of_lt_of_le (add_lt_add h4 h3)
          apply le_add_right
      _ ≤ ω^(ω⁵ * (μ n + 1) + μ s + 7) + ω * (μ b + 1) + 1 := le_add_right _ _
      _ = μ (recDelta b s (delta n)) := by rfl
```

### 4. `R_eq_diff` case
```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- Show ω⁴ * (ω³*(μ a + 1) + ω²*(μ b + 1) + 2) + 1 < ω^(μ a + μ b + 9) + 1
    have h1 : ω³ * (μ a + 1) < ω^(μ a + μ b + 9) := by
      apply mul_lt_opow _ (opow_pos _ omega0_pos)
      apply add_lt_add_right
      apply add_lt_add_left
      apply add_lt_add_right
      apply lt_of_le_of_lt (le_add_right _ _)
      apply opow_pos
    have h2 : ω² * (μ b + 1) < ω^(μ a + μ b + 9) := by
      apply mul_lt_opow _ (opow_pos _ omega0_pos)
      apply add_lt_add_right
      apply lt_of_le_of_lt (le_add_right _ _)
      apply opow_pos
    have h3 : ω⁴ * (ω³ * (μ a + 1) + ω² * (μ b + 1) + 2) < ω^(μ a + μ b + 9) := by
      apply mul_lt_opow _ (opow_pos _ omega0_pos)
      apply add_lt_add_right
      apply add_lt_add_right
      apply lt_of_le_of_lt (le_add_right _ _)
      exact lt_add_one 1
      exact lt_of_le_of_lt (le_add_right _ _) (add_lt_add h1 h2)
    calc μ (integrate (merge a b))
      _ = ω⁴ * (μ (merge a b) + 1) + 1 := by rfl
      _ = ω⁴ * (ω³ * (μ a + 1) + ω² * (μ b + 1) + 2) + 1 := by rfl
      _ < ω^(μ a + μ b + 9) := h3
      _ ≤ ω^(μ a + μ b + 9) + 1 := le_add_right _ _
      _ = μ (eqW a b) := by rfl
```

These proofs rely on the key properties of ordinal arithmetic:
1. Higher ω-powers dominate lower ones
2. Multiplication by ω increases the exponent
3. Addition of positive terms preserves order
4. The exponential function is strictly increasing

The most complex case is `R_rec_succ` where we need to show that the ω⁵ term in the exponent dominates all other terms. The other cases are more straightforward applications of ordinal arithmetic properties.
