-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    set m := mu n
    set σ := mu s
    set β := mu b
    -- dominant term on RHS is ω raised to a huge exponent
    have exp_big : m + σ + 6 < ω0 ^ (5 : Ordinal) * (m + 1) + 1 + σ + 6 := by
      have h5 : ω0 ^ (5 : Ordinal) > 0 := opow_pos 5 omega0_pos
      calc
        m + σ + 6
          ≤ m + σ + 6 + 1 := le_add_right _ _
          _ < ω0 ^ (5 : Ordinal) := lt_opow_of_lt_log (by simp [h5]) (by decide)
          _ ≤ ω0 ^ (5 : Ordinal) * (m + 1) := le_mul_right _ h5
          _ ≤ ω0 ^ (5 : Ordinal) * (m + 1) + 1 + σ + 6 := by
               apply le_add_right; apply le_add_right; rfl
    have rec_lt : mu (recDelta b s n) < ω0 ^ (m + σ + 6) :=
      lt_add_of_le_of_pos (le_add_left _ _) (opow_pos _ omega0_pos)
    have app_bound : mu (app s (recDelta b s n)) < ω0 ^ (m + σ + 8) := by
      calc
        mu (app s (recDelta b s n))
          ≤ ω0 ^ (3 : Ordinal) * (σ + 1)
            + ω0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1 := by simp [mu]; rfl
          _ < ω0 ^ (3 : Ordinal) * (σ + 1)
            + ω0 ^ (2 : Ordinal) * (ω0 ^ (m + σ + 6) + 1) + 1 :=
              add_lt_add_right (add_lt_add_left (mul_lt_mul_right rec_lt _) _) _
          _ ≤ ω0 ^ (m + σ + 8) := by
               -- ω²·ωᵏ = ω^{k+2}, ω³·α < ω^{k+8} for k = m+σ+6
               apply lt_of_le_of_lt _ (opow_le_opow_right (le_add_right (le_refl _)) _)
               simp only [opow_add, mul_add]
               apply add_le_add
               · apply le_add_right
                 apply le_trans (mul_le_mul_left (opow_le_opow_right (by decide) _))
                 rw [← opow_add]; apply opow_le_opow_right; omega
               · apply le_add_right; apply le_trans _ (opow_le_opow_right (by decide) _)
                 simp [opow_add]; apply le_add_right; rfl
    -- the huge exponent on the RHS absorbs the entire app term
    have main : mu (app s (recDelta b s n)) < ω0 ^ (mu (delta n) + μ s + 6) :=
      lt_of_lt_of_le app_bound (opow_le_opow_right exp_big _)
    calc
      mu (app s (recDelta b s n))
        < ω0 ^ (mu (delta n) + μ s + 6) := main
        _ ≤ ω0 ^ (mu (delta n) + μ s + 6)
          + ω0 * (β + 1) + 1 := le_add_right _ _
      _ = mu (recDelta b s (delta n)) := by simp [mu]; rfl

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ (opow_pos _ omega0_pos))

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    set α := mu a
    set β := mu b
    -- merge(a,b) < ω³·(α+1) + ω²·(β+1) + 2
    have merge_bound : mu (merge a b) + 1
        < ω0 ^ (3 : Ordinal) * (α + 1) + ω0 ^ (2 : Ordinal) * (β + 1) + 2 := by
      simp only [mu]; apply add_lt_add_right; apply add_lt_add
      · exact mul_lt_mul_right (lt_add_one _) _
      · exact mul_lt_mul_right (lt_add_one _) _
    have integ_bound : mu (integrate (merge a b))
        < ω0 ^ (4 : Ordinal) * (ω0 ^ (3 : Ordinal) * (α + 1)
            + ω0 ^ (2 : Ordinal) * (β + 1) + 2) + 1 := by
      simp only [mu]; apply add_lt_add_right; apply mul_lt_mul_right merge_bound _
    -- LHS is < ω^7·(something); RHS exponent is α+β+9 ≥ 9 > 7
    have pow7_lt : ω0 ^ (4 : Ordinal) * (ω0 ^ (3 : Ordinal) * (α + 1)
          + ω0 ^ (2 : Ordinal) * (β + 1) + 2) + 1
        < ω0 ^ (α + β + 9) := by
      have h7 : (7 : Ordinal) < α + β + 9 := by
        have h := add_pos_right (add_le_add (Ordinal.zero_le α) (Ordinal.zero_le β)) (by decide : 9 > 0)
        apply lt_of_le_of_lt _ (add_lt_add_right h 7); decide
      rw [← opow_add]; apply lt_opow_of_lt_log
      · exact opow_pos _ omega0_pos
      · apply lt_of_le_of_lt (le_add_right (le_refl _)) h7
    calc
      mu (integrate (merge a b))
        < ω0 ^ (α + β + 9) := by
            apply lt_of_le_of_lt (le_of_lt integ_bound) pow7_lt
        _ ≤ ω0 ^ (α + β + 9) + 1 := le_add_right _ _
      _ = mu (eqW a b) := by simp [mu]; rfl

These skeletons follow the style and helper lemmas already used in the file (`lt_add_of_le_of_pos`, `opow_pos`, `le_add_right`, `calc`, `Ordinal.pos_iff_ne_zero`). The key ordinal-arithmetic facts exploited are:

- `ω^5·(μn+1)` dominates `μn + μs + k` (used for `rec_succ`).
- `ω²·ω^k = ω^{k+2}` and absorption of lower powers (used for bounding the `app` term).
- `ω^4·(ω^3·…)` yields a power `ω^7·…` which is `< ω^(μa+μb+9)` whenever the exponent `μa+μb+9 ≥ 9` (used for `eq_diff`).
- Trivial positivity for the reflexive `eqW` case.
