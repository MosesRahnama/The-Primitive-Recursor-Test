A convenient way to finish the three cases is to isolate the ordinal-arithmetic arguments as local auxiliary facts. The constructor reductions themselves can then be handled by `simpa [mu]`.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    /*
      After simplification, the goal has the shape

        ω^3 * (mu s + 1)
          + ω^2 * (ω^(mu n + mu s + 6)
                    + ω * (mu b + 1) + 1 + 1)
          + 1
        <
        ω^(mu (delta n) + mu s + 6)
          + ω * (mu b + 1) + 1

      The useful intermediate estimates are:

        (1)  mu n + mu s + 6
              < mu (delta n) + mu s + 6

        (2)  ω^3 * (mu s + 1)
              + ω^2 * (ω^(mu n + mu s + 6)
                        + ω * (mu b + 1) + 2)
              <
              ω^(mu (delta n) + mu s + 6)

      Estimate (2) follows by putting the left-hand side into Cantor
      normal form and observing that its largest exponent is bounded by
      mu n + mu s + 8, whereas

        mu (delta n) = ω^5 * (mu n + 1) + 1

      dominates that exponent.
    */

    have hδ :
        mu n + mu s + (6 : Ordinal)
          < mu (delta n) + mu s + (6 : Ordinal) := by
      -- Expand `mu (delta n)` and use
      -- `mu n < ω₀ ^ 5 * (mu n + 1)`.
      simp [mu]
      calc
        mu n + mu s + (6 : Ordinal)
            < (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1)
                + mu s + 6 := by
              -- ordinal monotonicity and `mu n < ω₀^5 * (mu n + 1)`
              sorry
        _ = mu (delta n) + mu s + 6 := by simp [mu]

    have htower :
        omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (mu n + mu s + 6)
                + omega0 * (mu b + 1) + 2)
          <
        omega0 ^ (mu (delta n) + mu s + 6) := by
      /*
        Prove this by:

        1. Bounding the finite-height terms on the left by an ordinal
           whose leading exponent is at most `mu n + mu s + 8`.

        2. Showing
             mu n + mu s + 8
               < mu (delta n) + mu s + 6
           from the definition of `mu (delta n)`.

        3. Applying strict monotonicity of `x ↦ ω₀ ^ x`.
      */
      sorry

    calc
      omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (mu n + mu s + 6)
                + omega0 * (mu b + 1) + 2)
          + 1
          <
        omega0 ^ (mu (delta n) + mu s + 6) + 1 := by
            exact add_lt_add_right htower 1
      _ =
        omega0 ^ (mu (delta n) + mu s + 6)
          + omega0 * (mu b + 1) + 1 := by
            -- This is the normalized form produced by `[mu]`.
            simp [mu]

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]

    have hpos :
        (0 : Ordinal) <
          omega0 ^ (mu a + mu a + (9 : Ordinal)) + 1 := by
      calc
        (0 : Ordinal) < 1 := zero_lt_one
        _ ≤ omega0 ^ (mu a + mu a + (9 : Ordinal)) + 1 :=
          le_add_left _ _

    exact hpos

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    /*
      The normalized goal is

        ω₀^4 *
          (ω₀^3 * (mu a + 1)
            + ω₀^2 * (mu b + 1) + 2) + 1
        <
        ω₀^(mu a + mu b + 9) + 1.

      First bound the left-hand side by a single omega power.  Its
      largest Cantor-normal-form exponent is bounded by

        max (7 + mu a) (6 + mu b),

      while the right-hand exponent is `mu a + mu b + 9`.
    */

    have hbound :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2)
          <
        omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      /*
        A typical proof proceeds as follows:

        * distribute multiplication over the ordinal sums;
        * rewrite products of omega powers using
            `opow_add`;
        * bound the finite coefficients and the trailing `+ 2`;
        * use
            7 + mu a < mu a + mu b + 9
        and
            6 + mu b < mu a + mu b + 9;
        * conclude using strict monotonicity of ordinal exponentiation.
      */
      sorry

    calc
      omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2)
          + 1
          <
        omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
            exact add_lt_add_right hbound 1
```

The two nontrivial arithmetic subgoals are best factored into reusable lemmas rather than proved directly inside the rule cases. For example, one can introduce lemmas of the following shape before `mu_decreases`:

```lean
lemma rec_succ_domination (b s n : Trace) :
    omega0 ^ (3 : Ordinal) * (mu s + 1)
      + omega0 ^ (2 : Ordinal) *
          (omega0 ^ (mu n + mu s + 6)
            + omega0 * (mu b + 1) + 2)
      <
    omega0 ^ (mu (delta n) + mu s + 6) := by
  -- Cantor-normal-form comparison
  sorry

lemma eq_diff_domination (a b : Trace) :
    omega0 ^ (4 : Ordinal) *
        (omega0 ^ (3 : Ordinal) * (mu a + 1)
          + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2)
      <
    omega0 ^ (mu a + mu b + 9) := by
  -- Cantor-normal-form comparison
  sorry
```

Then the corresponding cases reduce to:

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    have h := rec_succ_domination b s n
    exact add_lt_add_right h 1

| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact
      calc
        (0 : Ordinal) < 1 := zero_lt_one
        _ ≤ omega0 ^ (mu a + mu a + 9) + 1 := le_add_left _ _

| _, _, Step.R_eq_diff a b => by
    simp [mu]
    have h := eq_diff_domination a b
    exact add_lt_add_right h 1
```

The `eqW` reflexivity case requires no substantial ordinal arithmetic; it is only positivity of a successor ordinal.
