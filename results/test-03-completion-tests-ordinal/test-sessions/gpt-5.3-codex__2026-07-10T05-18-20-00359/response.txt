### 1) `R_rec_succ` skeleton

```lean
| _, _, Step.R_rec_succ b s n => by
  simp [mu]
  -- Goal shape after simp:
  -- ω^3 * (mu s + 1) + ω^2 * (ω^(mu n + mu s + 6) + ω * (mu b + 1) + 1 + 1) + 1
  --   <
  -- ω^(mu (delta n) + mu s + 6) + ω * (mu b + 1) + 1
  --
  -- with mu (delta n) = ω^5 * (mu n + 1) + 1.

  -- Strategy:
  -- (A) Bound the whole LHS by some term < ω^(mu (delta n) + mu s + 6).
  -- (B) Then conclude < RHS since RHS = ω^(...) + lower terms.

  -- 1) Show exponent growth:
  have hexp :
      mu n + mu s + (6 : Ordinal) < mu (delta n) + mu s + (6 : Ordinal) := by
    -- enough: mu n < mu (delta n)
    have hn1 : mu n < mu n + 1 := lt_add_one (mu n)
    have hn2 : mu n + 1 ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) :=
      le_mul_right _ (opow_pos 5 omega0_pos)
    have hn3 : mu n < omega0 ^ (5 : Ordinal) * (mu n + 1) := lt_of_lt_of_le hn1 hn2
    -- then mu n < mu(delta n) by unfolding mu(delta n)=...+1
    have hn4 : mu n < mu (delta n) := by
      simpa [mu] using lt_of_lt_of_le hn3 (le_add_right _ _)
    exact add_lt_add_right (add_lt_add_right hn4 _) _

  -- 2) Convert exponent inequality to omega-power inequality.
  have hpow :
      omega0 ^ (mu n + mu s + (6 : Ordinal))
        < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
    exact opow_lt_opow_right omega0 one_lt_omega0 hexp

  -- 3) Dominate LHS by something below ω^(mu(delta n)+mu s+6).
  --    (Use standard lemmas: each summand is ≤ that big principal term,
  --     then finite/additive closure below principal bound.)
  have hleft :
      omega0 ^ (3 : Ordinal) * (mu s + 1) +
        omega0 ^ (2 : Ordinal) *
          (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 + 1) + 1
      < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := by
    -- skeleton placeholder:
    --   * bound ω^3*(...) < ω^(mu(delta n)+...)
    --   * bound ω^2*(...) < ω^(mu(delta n)+...)
    --   * sum +1 still < ω^(mu(delta n)+...) (principal closure)
    sorry

  -- 4) Finish: big principal term is first summand of RHS.
  calc
    omega0 ^ (3 : Ordinal) * (mu s + 1) +
      omega0 ^ (2 : Ordinal) *
        (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 + 1) + 1
        < omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) := hleft
    _ ≤ omega0 ^ (mu (delta n) + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := by
          exact le_add_right _ _
```
