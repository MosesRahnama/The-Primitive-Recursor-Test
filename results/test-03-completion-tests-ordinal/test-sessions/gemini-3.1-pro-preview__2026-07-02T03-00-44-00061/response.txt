Here is the proof skeleton for the three remaining cases. 

For `R_eq_refl`, the structure is identical to `R_int_delta` because the goal simplifies to proving `0 < X + 1`, which is solved by the exact same tactic sequence. 

For the other two rules, the proof is broken down using `calc` blocks that isolate the core ordinal arithmetic dominations (which are left as `sorry`) from the structural additions, establishing the strict decrease.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- The exponential term from delta (omega^5 * (mu n + 1)) strictly dominates the polynomial terms from app and merge.
    have h_dom : omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 + 1) + 1 < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) := by
      sorry
    
    -- Using the dominant term, we show it bounds the entire RHS structure.
    calc omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 + 1) + 1
      _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) := h_dom
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) := le_add_right _ _
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := le_add_right _ _

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    -- Since mu void = 0, we just need to show 0 < omega^(2 * mu a + 9) + 1. 
    -- We can reuse the exact same positivity argument from R_int_delta.
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- The RHS exponent (mu a + mu b + 9) dominates the nested multiplication from integrate + merge.
    have h_dom : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      sorry
      
    -- We then append the + 1 to both sides of the strict inequality.
    calc omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1
      _ < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
        sorry
```
