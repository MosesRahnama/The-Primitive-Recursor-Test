Here is a proof skeleton for the three remaining cases. The key insights are: (1) for `R_rec_succ`, the ordinal tower height `omega^5 * (mu n + 1)` in the successor case dominates the recursive call's exponent `mu n`; (2) for `R_eq_diff`, the exponent `mu a + mu b + 9` is designed to absorb the `integrate-merge` expansion which is bounded by `omega^7`; and (3) for `R_eq_refl`, the measure is a successor ordinal and hence positive.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    -- LHS dominant term: omega^(mu n + mu s + 8) 
    -- RHS dominant term: omega^(omega^5 * (mu n + 1) + mu s + 7)
    -- Since mu n + 1 < omega^5 * (mu n + 1), we have mu n + mu s + 8 < omega^5 * (mu n + 1) + mu s + 7
    have h1 : mu n + 1 < omega0 ^ (5 : Ordinal) * (mu n + 1) := by
      apply lt_mul_of_one_lt_of_le
      · exact one_lt_omega0.trans_le (opow_le_opow_right omega0_pos (by norm_num))
      · rfl
    have h2 : mu n + mu s + (8 : Ordinal) < omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (7 : Ordinal) := by
      calc mu n + mu s + (8 : Ordinal)
          = (mu n + 1) + mu s + (7 : Ordinal) := by simp [add_assoc, add_comm]
      _ < omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (7 : Ordinal) := by
        apply add_lt_add_right
        apply add_lt_add_right
        exact h1
    have h3 : omega0 ^ (mu n + mu s + (8 : Ordinal)) < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (7 : Ordinal)) := by
      apply opow_lt_opow_right
      · exact omega0_pos
      · exact h2
    -- Bound the full LHS by its dominant term plus smaller terms
    have h4 : omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + (2 : Ordinal)) < 
              omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (7 : Ordinal)) := by
      -- The term omega^2 * omega^(mu n + mu s + 6) = omega^(mu n + mu s + 8) dominates the LHS
      -- and is less than the RHS by h3. Other terms are smaller than this dominant term.
      sorry
    calc 
      omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + (2 : Ordinal)) + 1
          < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (7 : Ordinal)) + 1 := add_lt_add_right h4 1
      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := by
        simp [add_assoc]

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    -- Show 0 < omega^(2*mu a + 9) + 1
    have h : 0 < omega0 ^ (mu a + mu a + (9 : Ordinal)) + 1 := by
      apply lt_of_lt_of_le (Ordinal.zero_lt_one)
      exact le_add_right _ _
    exact h

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- LHS (without +1): omega^4 * (omega^3 * (mu a + 1) + omega^2 * (mu b + 1) + 2)
    --                  = omega^7 * (mu a + 1) + omega^6 * (mu b + 1) + omega^4 * 2
    -- RHS (without +1): omega^(mu a + mu b + 9)
    -- Since mu a + 1 < omega^(mu a) for mu a >= 1 (and similar for b), 
    -- we have omega^7 * (mu a + 1) < omega^(7 + mu a) <= omega^(mu a + mu b + 9)
    have h : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal)) < 
             omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
      have h1 : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal)) ≤
                omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1)) + omega0 ^ (4 : Ordinal) * (omega0 ^ (2 : Ordinal) * (mu b + 1)) + omega0 ^ (4 : Ordinal) * (2 : Ordinal) := by
        rw [mul_add, mul_add, add_assoc]
      have h2 : omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1)) = omega0 ^ (7 : Ordinal) * (mu a + 1) := by
        rw [← mul_assoc]; congr 1; rw [← opow_add]; norm_num
      have h3 : omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
        -- Since mu a + 1 ≤ omega^(mu a), we get omega^7 * (mu a + 1) ≤ omega^(7 + mu a) < omega^(mu a + 9) ≤ ...
        sorry
      -- Similarly bound the other terms omega^6*(mu b+1) and omega^4*2
      sorry
    calc 
      omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (2 : Ordinal)) + 1
          < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := add_lt_add_right h 1
      _ ≤ omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by rfl
```
