The three open cases are `R_rec_succ`, `R_eq_refl` and `R_eq_diff`. Below is a proof skeleton for each (comments only; no file edits).

### 2. `R_eq_diff` — dominate `ω⁴ · (…)` by `ω^(μa+μb+9)`

```lean
| _, _, Step.R_eq_diff a b => by
  simp [mu]
  -- Goal:
  --   ω⁴ * (ω³*(μa+1) + ω²*(μb+1) + 2) + 1
  --     < ω^(μa + μb + 9) + 1
  -- Equivalent (cancel +1 on both sides) to the body inequality below.
  apply add_lt_add_right
  -- Expand left body by left-distributivity:
  --   ω⁴ * (ω³*(μa+1) + ω²*(μb+1) + 2)
  --     = ω⁷*(μa+1) + ω⁶*(μb+1) + ω⁴*2
  --
  -- Claim: each of the three summands is < ω^(μa+μb+9).
  -- Then, since ω^γ (γ ≥ 1) is additively principal, their sum is too.
  have hγ : (1 : Ordinal) ≤ mu a + mu b + 9 := by
    -- 9 ≤ μa+μb+9
    exact le_add_left _ _
  have princ : IsAddPrincipal (omega0 ^ (mu a + mu b + 9)) :=
    omega0_opow_isAddPrincipal (ne_of_gt (lt_of_lt_of_le one_pos hγ))
    -- (or whatever the Mathlib name is under Principal)

  -- Term 1: ω⁷ * (μa+1) < ω^(μa+μb+9)
  have t1 : omega0 ^ (7 : Ordinal) * (mu a + 1)
              < omega0 ^ (mu a + mu b + 9) := by
    -- Use α ≤ ω^α always, so
    --   ω⁷*(μa+1) ≤ ω⁷ * ω^(μa+1) = ω^(μa+8)
    -- and μa+8 < μa+μb+9  (since 8 < μb+9 always: μb ≥ 0 ⇒ μb+9 ≥ 9 > 8)
    -- hence ω^(μa+8) < ω^(μa+μb+9)  (base ω > 1).
    have h_factor : mu a + 1 ≤ omega0 ^ (mu a + 1) :=
      (le_opow_self omega0_pos _).trans (opow_le_opow_right omega0_pos (le_add_right _ _))
      -- or simply: any ordinal β satisfies β ≤ ω^β (use le_of_lt (lt_omega0_opow …)
      --   / self-le lemmas available from Exponential)
    have h_mul :
        omega0 ^ (7 : Ordinal) * (mu a + 1)
          ≤ omega0 ^ (7 : Ordinal) * (omega0 ^ (mu a + 1)) :=
      mul_le_mul_left' h_factor _
    have h_exp :
        omega0 ^ (7 : Ordinal) * (omega0 ^ (mu a + 1))
          = omega0 ^ (mu a + 8) := by
      -- ω⁷ * ω^(μa+1) = ω^(7+μa+1) = ω^(μa+8)
      rw [← opow_add]; congr 1
      -- 7 + (μa+1) = μa+8  (commute / reassoc naturals-as-ordinals)
      ring_nf  -- or calc with add_comm / add_assoc
    have h_lt_exp : (mu a + 8 : Ordinal) < mu a + mu b + 9 := by
      -- μa+8 < μa+(μb+9)  ⇔  8 < μb+9  ⇔  0 < μb+1
      have : (8 : Ordinal) < mu b + 9 :=
        lt_of_lt_of_le (by decide : (8:Ordinal) < 9) (le_add_left _ _)
      -- rewrite goal as μa+8 < μa+(μb+9) and cancel μa
      simpa [add_assoc, add_comm 8, add_left_comm] using
        (add_lt_add_left this (mu a))
    have h_opow_lt :
        omega0 ^ (mu a + 8) < omega0 ^ (mu a + mu b + 9) :=
      (opow_lt_opow_iff_right one_lt_omega0).2 h_lt_exp
    exact lt_of_le_of_lt (h_mul.trans (le_of_eq h_exp)) h_opow_lt

  -- Term 2: ω⁶ * (μb+1) < ω^(μa+μb+9)   (symmetric argument)
  have t2 : omega0 ^ (6 : Ordinal) * (mu b + 1)
              < omega0 ^ (mu a + mu b + 9) := by
    -- ω⁶*(μb+1) ≤ ω⁶*ω^(μb+1) = ω^(μb+7) ≤ ω^(μa+μb+7) < ω^(μa+μb+9)
    sorry  -- same shape as t1; exponents 6/7 vs +9 give slack

  -- Term 3: ω⁴ * 2 < ω^(μa+μb+9)
  have t3 : omega0 ^ (4 : Ordinal) * 2
              < omega0 ^ (mu a + mu b + 9) := by
    -- ω⁴*2 < ω⁵ ≤ ω⁹ ≤ ω^(μa+μb+9)
    have : omega0 ^ (4 : Ordinal) * 2 < omega0 ^ (5 : Ordinal) := by
      -- finite:  ω⁴·2 < ω⁴·ω = ω⁵
      exact mul_lt_mul_of_pos_left (by decide : (2:Ordinal) < omega0)
        (opow_pos _ omega0_pos)
    have h5 : omega0 ^ (5 : Ordinal) ≤ omega0 ^ (mu a + mu b + 9) :=
      opow_le_opow_right omega0_pos (by
        -- 5 ≤ μa+μb+9
        exact le_trans (by decide : (5:Ordinal) ≤ 9) (le_add_left _ _))
    exact lt_of_lt_of_le this h5

  -- Sum of three principal-small terms is still < ω^(…)
  have sum_lt :
      omega0 ^ (7 : Ordinal) * (mu a + 1)
        + omega0 ^ (6 : Ordinal) * (mu b + 1)
        + omega0 ^ (4 : Ordinal) * 2
        < omega0 ^ (mu a + mu b + 9) := by
    -- Use IsAddPrincipal repeatedly (or add_lt_of_lt / principal.add_lt)
    exact princ.add_lt (princ.add_lt t1 t2) t3

  -- Reassociate the expanded form back to ω⁴ * (…):
  --   ω⁷*(μa+1) + ω⁶*(μb+1) + ω⁴*2
  --     = ω⁴ * (ω³*(μa+1) + ω²*(μb+1) + 2)
  have expand :
      omega0 ^ (4 : Ordinal)
        * (omega0 ^ (3 : Ordinal) * (mu a + 1)
            + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2)
        = omega0 ^ (7 : Ordinal) * (mu a + 1)
            + omega0 ^ (6 : Ordinal) * (mu b + 1)
            + omega0 ^ (4 : Ordinal) * 2 := by
    -- left-distrib + opow_add: ω⁴*ω³ = ω⁷, ω⁴*ω² = ω⁶
    simp only [mul_add, ← opow_add]
    -- then 4+3=7, 4+2=6; refl
    ring_nf  -- or manual congr

  exact expand ▸ sum_lt
```

Key design point already noted in the file: the `+ 9` in `μ(eqW)` is exactly large enough to absorb the finite tower height 7 coming from `ω⁴ · ω³`.

### 3. `R_rec_succ` — `δ` inflates the exponent past every RHS subterm

```lean
| _, _, Step.R_rec_succ b s n => by
  simp [mu]
  -- After simp the goal is essentially
  --   μ(app s (recDelta b s n))
  --     < μ(recDelta b s (delta n))
  -- i.e.
  --   ω³*(μs+1) + ω²*(μ(recDelta b s n)+1) + 1
  --     < ω^(μ(δ n) + μs + 6) + ω*(μb+1) + 1
  --
  -- Expand μ(δ n) = ω⁵*(μn+1) + 1, so the big LHS-of-goal exponent is
  --   μ(δ n) + μs + 6 = ω⁵*(μn+1) + μs + 7.
  -- Expand μ(recDelta b s n) = ω^(μn+μs+6) + ω*(μb+1) + 1, so the
  -- right-hand side of the comparison (the “small” side) is
  --   ω³*(μs+1)
  --     + ω² * (ω^(μn+μs+6) + ω*(μb+1) + 2) + 1
  --   = ω³*(μs+1) + ω^(μn+μs+8) + ω³*(μb+1) + ω²*2 + 1
  --     (using ω²*ω^e = ω^(e+2), ω²*ω = ω³).

  -- Strategy: show the entire small side is < ω^(ω⁵*(μn+1) + μs + 7),
  -- which is ≤ the full large side (large side = that power + positive terms).

  set E : Ordinal :=
    omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + 7
  -- E = μ(δ n) + μs + 6

  have big_le :
      omega0 ^ E ≤ mu (recDelta b s (delta n)) := by
    -- μ(recDelta … (δ n)) = ω^E + ω*(μb+1) + 1 ≥ ω^E
    simp [mu, E]
    exact le_add_right _ _

  -- Now show small side < ω^E.
  have small_lt : mu (app s (recDelta b s n)) < omega0 ^ E := by
    simp [mu]
    -- Goal: ω³*(μs+1) + ω²*(ω^(μn+μs+6) + ω*(μb+1) + 2) + 1 < ω^E
    -- Expand the middle product:
    have expand :
        omega0 ^ (2 : Ordinal)
          * (omega0 ^ (mu n + mu s + 6)
              + omega0 * (mu b + 1) + 2)
        = omega0 ^ (mu n + mu s + 8)
            + omega0 ^ (3 : Ordinal) * (mu b + 1)
            + omega0 ^ (2 : Ordinal) * 2 := by
      simp only [mul_add, ← opow_add]
      -- 2 + (μn+μs+6) = μn+μs+8;  2+1 = 3
      ring_nf

    -- Four summands to bound, each < ω^E; then principal-sum.
    have princE : IsAddPrincipal (omega0 ^ E) :=
      omega0_opow_isAddPrincipal (by
        -- E ≥ 7 > 0
        exact ne_of_gt (lt_of_lt_of_le (by decide : (0:Ordinal) < 7)
          (le_add_left _ _)))

    -- (i) ω^(μn+μs+8) < ω^E
    --     because μn+μs+8 < E = ω⁵*(μn+1) + μs + 7
    --     ⇔ μn + 1 < ω⁵*(μn+1)   (cancel μs, compare +8 vs +7)
    have exp_lt : mu n + mu s + 8 < E := by
      -- μn+8 < ω⁵*(μn+1) + 7
      -- enough: μn+1 ≤ μn+7 < ω*(μn+1) ≤ ω⁵*(μn+1)
      have h1 : mu n + 1 < omega0 * (mu n + 1) := by
        have : (1 : Ordinal) < omega0 := one_lt_omega0
        exact (lt_mul_of_one_lt_left (succ_pos _) this)
        -- more elementary: μn+1 = 1*(μn+1) < ω*(μn+1)
      have h2 : omega0 * (mu n + 1)
                  ≤ omega0 ^ (5 : Ordinal) * (mu n + 1) := by
        -- ω ≤ ω⁵ ⇒ ω·c ≤ ω⁵·c
        exact mul_le_mul_right' (opow_le_opow_right omega0_pos (by decide)) _
      have h3 : mu n + 1 < omega0 ^ (5 : Ordinal) * (mu n + 1) :=
        lt_of_lt_of_le h1 h2
      -- lift by + (μs+7) on both sides carefully:
      --   (μn+1) + (μs+7) < ω⁵*(μn+1) + (μs+7)
      -- and left = μn+μs+8
      simpa [E, add_assoc, add_comm, add_left_comm] using
        add_lt_add_right h3 (mu s + 7)

    have t_pow : omega0 ^ (mu n + mu s + 8) < omega0 ^ E :=
      (opow_lt_opow_iff_right one_lt_omega0).2 exp_lt

    -- (ii) ω³*(μs+1) < ω^E
    have t_s : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ E := by
      -- ω³*(μs+1) ≤ ω³*ω^(μs+1) = ω^(μs+4) ≤ ω^(μs+7) ≤ ω^E
      -- (E = ω⁵*(μn+1) + μs + 7 ≥ μs + 7)
      sorry  -- same ω^k·(α+1) ≤ ω^(α+k+…) pattern as in R_eq_diff

    -- (iii) ω³*(μb+1) < ω^E   (identical shape, exponent slack ≥ 7)
    have t_b : omega0 ^ (3 : Ordinal) * (mu b + 1) < omega0 ^ E := by
      sorry

    -- (iv) ω²*2 < ω^E   (finite, trivial)
    have t_fin : omega0 ^ (2 : Ordinal) * 2 < omega0 ^ E := by
      -- ω²*2 < ω³ ≤ ω^7 ≤ ω^E
      sorry

    -- (v) the final +1 < ω^E
    have t_one : (1 : Ordinal) < omega0 ^ E :=
      one_lt_omega0.trans_le (le_opow_self omega0_pos
        (by -- E ≥ 7 ≥ 1
            exact le_trans (by decide : (1:Ordinal) ≤ 7) (le_add_left _ _)))
      -- or simply: opow_pos _ _ ▸ …

    -- Assemble: small = t_s + (t_pow + t_b + t_fin) + 1
    -- Apply princE.add_lt repeatedly.
    sorry  -- bookkeeping with expand / reassoc

  exact lt_of_lt_of_le small_lt big_le
```

### Summary of the ordinal facts you need (all in the imported Mathlib modules)

| Fact | Use |
|---|---|
| `ω^α` additively principal for `α ≠ 0` | sum of `< ω^α` terms stays `< ω^α` |
| `opow_lt_opow_iff_right (1 < ω)` | compare pure powers by comparing exponents |
| `ω^a * ω^b = ω^(a+b)` (`opow_add`) | expand `ω^k * ω^e` |
| left distributivity `a*(b+c) = a*b + a*c` | expand measure expressions |
| `β ≤ ω^β` / `β < ω · (β+1)` | absorb a coefficient into a slightly larger exponent |
| finite slack (`+6`, `+7`, `+9` built into `μ`) | guarantee strict exponent inequality even after the absorptions |

The two non-trivial skeletons (`R_eq_diff`, `R_rec_succ`) follow the same template: expand → bound each summand by a pure power of `ω` with a strictly smaller exponent → reassemble with principality → cancel the trailing `+ 1`.
