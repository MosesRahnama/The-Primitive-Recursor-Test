### 4. `R_eq_diff` case
```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- The key is that μa + μb + 9 dominates all exponents in the LHS
    have h1 : mu a + 1 ≤ omega0 ^ (3 : Ordinal) * (mu a + 1) :=
      le_mul_right _ (opow_pos 3 omega0_pos)
    have h2 : mu b + 1 ≤ omega0 ^ (2 : Ordinal) * (mu b + 1) :=
      le_mul_right _ (opow_pos 2 omega0_pos)
    have h3 : mu a + μb + 9 < omega0 ^ (mu a + μb + 9) :=
      opow_lt_opow_of_lt (by apply Nat.one_le_of_lt; exact omega0_pos) (by linarith)
    have h4 : omega0 ^ (3 : Ordinal) * (mu a + 1) +
              omega0 ^ (2 : Ordinal) * (mu b + 1) + 2 <
              omega0 ^ (mu a + μb + 9) := by
      apply add_lt_add
      · apply opow_lt_opow_of_lt (by apply Nat.one_le_of_lt; exact omega0_pos)
        calc mu a + 1
          _ ≤ mu a + μb + 9 := by linarith
      · apply opow_lt_opow_of_lt (by apply Nat.one_le_of_lt; exact omega0_pos)
        calc mu b + 1
          _ ≤ mu a + μb + 9 := by linarith
      · exact lt_add_of_pos_right _ (by linarith)
    calc omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) +
                                   omega0 ^ (2 : Ordinal) * (mu b + 1) + 2) + 1
      _ < omega0 ^ (4 : Ordinal) * omega0 ^ (mu a + μb + 9) := by
          apply add_lt_add_right
          apply mul_lt_mul_of_pos_left h4 (opow_pos 4 _)
      _ ≤ omega0 ^ (mu a + μb + 9) := by
          apply opow_le_opow_of_le (by apply Nat.one_le_of_lt; exact omega0_pos)
          linarith
      _ ≤ omega0 ^ (mu a + μb + 9) + 1 := le_add_right _ _
```

These proofs rely on the fact that higher ω-powers dominate lower ones, and that ordinal addition and multiplication preserve order in specific ways. The key insight is that the exponents in the right-hand sides are large enough to dominate all terms in the left-hand sides.
