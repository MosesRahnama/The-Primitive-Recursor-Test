Below is a Lean-style proof skeleton for the three `sorry` cases. I would not try to inline all ordinal arithmetic immediately; instead, isolate the two hard comparisons as local/global helper lemmas about `omega0`-towers.

One small note: in the comment for `R_rec_succ`, the “LHS/RHS” names are swapped: the source measure is the large `recDelta ... (delta n)` tower, and the target measure is the `app ...` expression.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    /-
    After `simp [mu]`, the goal is essentially:

      omega0 ^ 3 * (mu s + 1)
        + omega0 ^ 2 *
            (omega0 ^ (mu n + mu s + 6)
              + omega0 * (mu b + 1) + 1 + 1)
        + 1
      <
      omega0 ^
        (omega0 ^ 5 * (mu n + 1) + 1 + mu s + 6)
        + omega0 * (mu b + 1) + 1

    It is enough to prove the stronger bound that the whole `app` measure is
    below the leading tower of the source.
    -/

    set B : Ordinal := mu b
    set S : Ordinal := mu s
    set N : Ordinal := mu n

    have hmain :
        omega0 ^ (3 : Ordinal) * (S + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (N + S + (6 : Ordinal))
                + omega0 * (B + 1) + 1 + 1)
          + 1
        <
        omega0 ^
          (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)) := by
      /-
      Skeleton for this helper proof:

      1. Let

           Esmall := N + S + 6
           Ebig   := omega0^5 * (N + 1) + 1 + S + 6

         Show

           Esmall < Ebig.

         This follows because

           N < N + 1 ≤ omega0^5 * (N + 1),

         hence

           N + S + 6 < omega0^5 * (N + 1) + 1 + S + 6.

      2. From monotonicity of `omega0 ^ ·`, get

           omega0 ^ Esmall < omega0 ^ Ebig.

      3. Show each lower-order summand of the target `app` measure is also
         below `omega0 ^ Ebig`:

           omega0^3 * (S + 1) < omega0 ^ Ebig,
           omega0^2 * (omega0^Esmall + omega0*(B+1) + 1 + 1)
             < omega0 ^ Ebig,
           1 < omega0 ^ Ebig.

         The important comparison is the second one. Use the usual ordinal
         estimate

           omega0^2 * omega0^Esmall = omega0 ^ (2 + Esmall)

         and then show

           2 + Esmall < Ebig.

      4. Use additively-principal closure of `omega0 ^ Ebig` to combine the
         summand bounds. Since `omega0 ^ Ebig` is additively principal and
         positive, a finite ordinal sum of terms below it is still below it.

      In practice, factor this out as a lemma such as:

        lemma recSucc_app_bound (B S N : Ordinal) :
          omega0^3 * (S+1)
            + omega0^2 *
                (omega0^(N+S+6) + omega0*(B+1) + 1 + 1)
            + 1
          <
          omega0^(omega0^5*(N+1) + 1 + S + 6) := ...

      -/
      sorry

    calc
      omega0 ^ (3 : Ordinal) * (S + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (N + S + (6 : Ordinal))
                + omega0 * (B + 1) + 1 + 1)
          + 1
          < omega0 ^
              (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)) := hmain
      _ ≤ omega0 ^
              (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal))
            + omega0 * (B + 1) + 1 := by
          exact le_add_right _ _
```

The easy reflexivity case can be closed directly:

```lean
-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact add_pos_right _ one_pos
```

For `eq_diff`, use the same strategy: prove the target is below the leading `eqW` tower, then use `< x + 1`.

```lean
-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    /-
    After simplification, the goal is essentially:

      omega0^4 *
        (omega0^3 * (mu a + 1)
          + omega0^2 * (mu b + 1)
          + 1 + 1)
        + 1
      <
      omega0 ^ (mu a + mu b + 9) + 1

    It suffices to show the stronger inequality without the trailing `+ 1`
    on the RHS:

      omega0^4 *
        (omega0^3 * (A + 1)
          + omega0^2 * (B + 1)
          + 1 + 1)
        + 1
      <
      omega0 ^ (A + B + 9).
    -/

    set A : Ordinal := mu a
    set B : Ordinal := mu b

    have hmain :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (A + 1)
              + omega0 ^ (2 : Ordinal) * (B + 1)
              + 1 + 1)
          + 1
        <
        omega0 ^ (A + B + (9 : Ordinal)) := by
      /-
      Skeleton for this helper proof:

      1. First bound the `merge` measure plus one by a smaller tower:

           omega0^3 * (A + 1)
             + omega0^2 * (B + 1)
             + 1 + 1
           <
           omega0 ^ (A + B + 5).

         Reason:
         - `omega0^3 * (A + 1)` is below `omega0^(A + B + 4)`;
         - `omega0^2 * (B + 1)` is also below `omega0^(A + B + 4)`;
         - finite constants are below that same tower;
         - combine by additively-principal closure.

      2. Multiply by `omega0^4`:

           omega0^4 *
             (omega0^3 * (A + 1)
               + omega0^2 * (B + 1)
               + 1 + 1)
           <
           omega0^4 * omega0^(A + B + 5).

      3. Use the exponent law:

           omega0^4 * omega0^(A + B + 5)
             = omega0^(4 + (A + B + 5)).

      4. Show

           4 + (A + B + 5) ≤ A + B + 9.

         For finite/ordinal arithmetic this is the intended padding provided
         by the `+ 9` in the definition of `mu (eqW a b)`.

      5. By monotonicity of `omega0 ^ ·`:

           omega0^(4 + (A + B + 5))
             ≤ omega0^(A + B + 9).

      6. Finally, the trailing `+ 1` on the left is still below
         `omega0^(A + B + 9)` because that tower is a limit/additively
         principal ordinal large enough to absorb finite successors.

      This can be factored as:

        lemma eqDiff_integrate_merge_bound (A B : Ordinal) :
          omega0^4 *
              (omega0^3 * (A+1)
                + omega0^2 * (B+1)
                + 1 + 1)
            + 1
          <
          omega0^(A+B+9) := ...

      -/
      sorry

    calc
      omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (A + 1)
              + omega0 ^ (2 : Ordinal) * (B + 1)
              + 1 + 1)
          + 1
          < omega0 ^ (A + B + (9 : Ordinal)) := hmain
      _ < omega0 ^ (A + B + (9 : Ordinal)) + 1 := lt_add_one _
```

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    set B : Ordinal := mu b
    set S : Ordinal := mu s
    set N : Ordinal := mu n
    have h := recSucc_app_bound B S N
    calc
      omega0 ^ (3 : Ordinal) * (S + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (N + S + (6 : Ordinal))
                + omega0 * (B + 1) + 1 + 1)
          + 1
          < omega0 ^
              (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal)) := h
      _ ≤ omega0 ^
              (omega0 ^ (5 : Ordinal) * (N + 1) + 1 + S + (6 : Ordinal))
            + omega0 * (B + 1) + 1 := le_add_right _ _

| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact add_pos_right _ one_pos

| _, _, Step.R_eq_diff a b => by
    simp [mu]
    set A : Ordinal := mu a
    set B : Ordinal := mu b
    have h := eqDiff_integrate_merge_bound A B
    calc
      omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (A + 1)
              + omega0 ^ (2 : Ordinal) * (B + 1)
              + 1 + 1)
          + 1
          < omega0 ^ (A + B + (9 : Ordinal)) := h
      _ < omega0 ^ (A + B + (9 : Ordinal)) + 1 := lt_add_one _
```
