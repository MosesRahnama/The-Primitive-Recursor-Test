  --   < ω^(mu (delta n) + mu s + 6) + ω*(mu b + 1) + 1

  -- Expand mu (delta n):
  -- mu (delta n) = ω^5*(mu n + 1) + 1
  -- so RHS leading term is ω^(ω^5*(mu n + 1) + 1 + mu s + 6).

    -- * show each additive piece is < that huge ω^E
    -- * use that E is successor-ish, so ω^E is additively principal
    -- * key comparison:
    --     mu n + mu s + 6 < ω^5*(mu n + 1) + 1 + mu s + 6
    --   hence
    --     ω^(mu n + mu s + 6) < ω^(...)
    --   then multiplying by ω^2 and adding lower terms stays below ω^(...)

  -- Step 2: strictness survives adding tail `+ ω*(mu b+1)+1` on RHS.
  have hR :
      omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal))
      < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal))

| _, _, Step.R_eq_diff a b => by

  -- ω^4 * (ω^3*(mu a + 1) + ω^2*(mu b + 1) + 1 + 1) + 1
  --   < ω^(mu a + mu b + 9) + 1

  -- Step 1: drop the trailing +1 on RHS (enough to prove < ω^(...)).

            + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
      < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by

    -- * bound inside parentheses by some ω^(mu a + mu b + 4) (or similar)
    -- * then ω^4 * ω^(...) = ω^(...+4) (or ≤ that), still < ω^(mu a+mu b+9)
    -- * final strictness from exponent inequality (...+4 < ...+9).
    --
    -- Useful intermediate inequalities:
    --   mu a + 1 ≤ ω^(mu a + 1)
    --   mu b + 1 ≤ ω^(mu b + 1)
    --   ω^3*(mu a+1) < ω^(mu a + 4)

    -- then absorb finite sums under a larger ω^(mu a + mu b + 5), etc.

      omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (mu a + 1)
            + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1
      < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
