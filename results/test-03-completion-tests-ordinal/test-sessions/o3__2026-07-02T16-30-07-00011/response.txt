Here is a (non-compiling) “fill-in-the-blanks’’ style outline for the three missing cases of
`mu_decreases`.
Nothing below is *proved* – every hard ordinal-arithmetic step has been
replaced by an `admit`.  What ​is​ supplied is a clear sequence of the
individual lemmas that have to be invoked, together with the structure
(`simp`, `have`, `calc`, …) in which they have to be combined.
Dropping in the required ordinal-inequalities in place of each `admit`
will turn the outline into a complete proof.

```lean
/-!  ##  Proof skeleton for the remaining rule cases                -/

-- … `mu_decreases` up-to-date prefix omitted …

-- recDelta b s (delta n)  ⟶  app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- Expand both sides
    simp [mu]

    /-  We must show

          ω^(μ n + μ s + 6) + ω * (μ b + 1) + 1
        < ω^(ω^5 * (μ n + 1) + (μ s + 6)) + ω * (μ b + 1) + 1.

        Strategy:

        •  Show  μ n + μ s + 6
           < ω^5 * (μ n + 1) + (μ s + 6).

        •  Apply   opow_lt_opow_left  to lift this to the tower level.

        •  Add the common positive tail   ω * (μ b + 1) + 1.               -/

    have h_exp :
        (μ n + μ s + (6 : Ordinal)) <
          ω0 ^ (5 : Ordinal) * (μ n + 1) + (μ s + 6) := by
      --  The ω^5·(μ n + 1) part dwarfs the whole LHS.
      admit

    have h_tower :
        ω0 ^ (μ n + μ s + 6)
        < ω0 ^ (ω0 ^ (5 : Ordinal) * (μ n + 1) + (μ s + 6)) :=
      opow_lt_opow_left omega0_pos h_exp

    have h_goal :
        ω0 ^ (μ n + μ s + 6) + ω0 * (μ b + 1) + 1
      < ω0 ^ (ω0 ^ 5 * (μ n + 1) + (μ s + 6)) + ω0 * (μ b + 1) + 1 := by
      -- add_lt_add_of_lt_of_nonneg :  a<b  → 0≤c → a+c < b+c
      have h_tail_nonneg : (0 : Ordinal) ≤ ω0 * (μ b + 1) + 1 := by
        -- product-and-sum of positive ordinals is non-negative
        exact le_of_lt (add_pos (mul_pos omega0_pos (succ_pos _)) one_pos)
      exact
        add_lt_add_of_lt_of_nonneg
          (add_lt_add_of_lt_of_nonneg h_tower h_tail_nonneg)
          (by exact zero_le_one)

    exact h_goal


-- eqW a a  ⟶  void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    /-  Goal   0 < ω^(μ a + μ a + 9) + 1. -/

    have h_pos : (0 : Ordinal) < ω0 ^ (μ a + μ a + (9 : Ordinal)) :=
      opow_pos _ _

    have : (0 : Ordinal) < ω0 ^ (μ a + μ a + 9) + 1 :=
      lt_of_lt_of_le h_pos (le_add_of_nonneg_right _ zero_le_one)

    exact this


-- eqW a b ⟶ integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    /-  Unfold  μ  on both sides.                                      -/
    simp [mu]

    /-
      LHS = ω^4 * ( μ(merge a b) + 1) + 1
      RHS = ω^(μ a + μ b + 9)       + 1

      Plan:

      1.  Prove      4 < μ a + μ b + 9.
      2.  From this obtain   ω^4 < ω^(μ a + μ b + 9)
         using `opow_lt_opow_left`.
      3.  Since   μ(merge a b) + 1  is positive,
         we get   ω^4 * (μ(merge a b) + 1)
               < ω^(μ a + μ b + 9).
      4.  Finally add 1 to both sides.
    -/

    have h_exp : (4 : Ordinal) < μ a + μ b + (9 : Ordinal) := by
      --  trivial numeric inequality because  μ a + μ b  ≥ 0.
      admit

    have h_pow :
        ω0 ^ (4 : Ordinal)
        < ω0 ^ (μ a + μ b + 9) :=
      opow_lt_opow_left omega0_pos h_exp

    have h_mul :
        ω0 ^ (4 : Ordinal) * (μ (merge a b) + 1)
        < ω0 ^ (μ a + μ b + 9) := by
      --  Multiplication by the positive ordinal  μ(merge a b)+1
      --  keeps the strict inequality in the *left* coefficient slot.
      --  (Use  `mul_lt_of_lt_one_left`  + monotonicity wrt left factor.)
      admit

    have h_final :
        ω0 ^ (4 : Ordinal) * (μ (merge a b) + 1) + 1
      < ω0 ^ (μ a + μ b + 9)                     + 1 :=
      add_lt_add_of_lt_of_nonneg h_mul zero_le_one

    exact h_final
```

The `admit`s mark the only genuinely non-trivial parts:

•  the arithmetical comparison of the two exponents in the `recDelta`
   successor case,

•  the elementary but still explicit inequality
   `4 < μ a + μ b + 9` in the `eq_diff` case, and

•  the technical step showing that multiplication by the (positive)
   factor `μ (merge a b) + 1` preserves the strict ordering of the
   towers.

Filling in those three pieces of ordinal arithmetic (all routine once
the right lemmas from `Mathlib.SetTheory.Ordinal` are picked) completes
`mu_decreases`, and hence the whole strong-normalisation development.
