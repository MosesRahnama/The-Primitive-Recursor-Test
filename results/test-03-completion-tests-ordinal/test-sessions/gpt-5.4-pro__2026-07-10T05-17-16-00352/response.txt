Below is a Lean-ish skeleton that should be close to what you want. I have **not** compiled it, so a few library names may need tiny adjustment, but the shape is the important part.

The short version is:

- `R_eq_refl` is immediate.
- `R_eq_diff` should go through by bounding `mu (merge a b) + 1` by a single `ω`-power, then multiplying by `ω^4`.
- `R_rec_succ` has a natural skeleton, but I would **double-check the measure there**: the `ω * (mu b + 1)` tail looks like the genuinely delicate part.

---

## 1. `eqW a a --> void`

This one can be filled exactly like `R_int_delta`:

```lean
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)
```

## 2. `eqW a b --> integrate (merge a b)`

A good way to structure this case is to first prove a local helper of the form

```lean
have hshift :
    ∀ k x : Ordinal, omega0 ^ k * (x + 1) < omega0 ^ (x + (k + 1)) := by
  intro k x
  -- standard bound: x + 1 < ω^(x + 1)
  have hx : x + 1 < omega0 ^ (x + 1) := by
    -- use the usual “below the next ω-power” lemma here
    -- lemma name may be something like `lt_opow_succ` / similar
    sorry
  calc
    omega0 ^ k * (x + 1)
        < omega0 ^ k * omega0 ^ (x + 1) := by
            -- strict monotonicity of left-multiplication by a positive ordinal
            simpa using mul_lt_mul_of_pos_left hx (opow_pos _ omega0_pos)
    _ = omega0 ^ (k + (x + 1)) := by
          rw [← opow_add]
    _ ≤ omega0 ^ (x + (k + 1)) := by
          -- monotonicity of `ω^·`
          apply (Ordinal.isNormal_opow omega0).monotone
          -- finite reassociation
          simpa [add_assoc]
```

Then the `eq_diff` case can be organized like this:

```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    let C : Ordinal := mu a + mu b + (4 : Ordinal)

    have h₁ : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ C := by
      have hsmall : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + (4 : Ordinal)) := by
        simpa [add_assoc] using hshift (3 : Ordinal) (mu a)
      refine lt_of_lt_of_le hsmall ?_
      apply (Ordinal.isNormal_opow omega0).monotone
      -- μ a + 4 ≤ μ a + μ b + 4
      simpa [C, add_assoc] using add_le_add_left (le_add_right (mu a + (4 : Ordinal)) (mu b)) 0

    have h₂ : omega0 ^ (2 : Ordinal) * (mu b + 1) < omega0 ^ C := by
      have hsmall : omega0 ^ (2 : Ordinal) * (mu b + 1) < omega0 ^ (mu b + (3 : Ordinal)) := by
        simpa [add_assoc] using hshift (2 : Ordinal) (mu b)
      refine lt_of_lt_of_le hsmall ?_
      apply (Ordinal.isNormal_opow omega0).monotone
      -- μ b + 3 ≤ μ a + μ b + 4
      -- insert μ a on the left and one more finite step on the right
      -- this is just ordinal monotonicity / reassociation
      sorry

    have hOne : (1 : Ordinal) < omega0 ^ C := by
      -- since C ≥ 1 (indeed C ≥ 4), ω^C > 1
      sorry

    have hTwo : (2 : Ordinal) < omega0 ^ C := by
      -- e.g. from hOne and positivity, or by showing ω ≤ ω^C
      sorry

    have hmerge :
        omega0 ^ (3 : Ordinal) * (mu a + 1)
          + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1
        < omega0 ^ C := by
      -- Here use additive-principalness of `ω^C`
      -- (`Mathlib.SetTheory.Ordinal.Principal` is imported exactly for this)
      --
      -- First combine h₁ and h₂:
      --   h12 : ω^3*(μa+1) + ω^2*(μb+1) < ω^C
      -- Then add the two `1`s, still staying below `ω^C`.
      --
      -- The lemma name may be something like:
      --   `add_lt_of_lt_of_lt_of_principal`
      -- or a variant specialized to `ω^C`.
      sorry

    have hprod :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
        < omega0 ^ (mu a + mu b + (8 : Ordinal)) := by
      calc
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1)
            < omega0 ^ (4 : Ordinal) * omega0 ^ C := by
                simpa using mul_lt_mul_of_pos_left hmerge (opow_pos 4 omega0_pos)
        _ = omega0 ^ ((4 : Ordinal) + C) := by
              rw [← opow_add]
        _ = omega0 ^ (mu a + mu b + (8 : Ordinal)) := by
              simp [C, add_assoc]

    have hfinal :
        omega0 ^ (4 : Ordinal) *
            (omega0 ^ (3 : Ordinal) * (mu a + 1)
              + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1
        < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := by
      exact add_lt_add_right
        (lt_of_lt_of_le hprod <|
          by
            apply (Ordinal.isNormal_opow omega0).monotone
            simp [add_assoc]) 1

    simpa [C, add_assoc] using hfinal
```

### Core idea for `eq_diff`
The proof is:

1. show the inner `merge` payload is `< ω^(μ a + μ b + 4)`;
2. multiply by `ω^4` to get `< ω^(μ a + μ b + 8)`;
3. then `+ 1 < ω^(μ a + μ b + 9) + 1`.

## 3. `recDelta b s (delta n) --> app s (recDelta b s n)`

Here is the **intended skeleton**:

```lean
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    let A : Ordinal := mu (delta n) + mu s + (6 : Ordinal)

    have hs :
        omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ A := by
      -- same `hshift` helper as above
      have hsmall :
          omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (mu s + (4 : Ordinal)) := by
        simpa [add_assoc] using hshift (3 : Ordinal) (mu s)
      refine lt_of_lt_of_le hsmall ?_
      apply (Ordinal.isNormal_opow omega0).monotone
      -- μ s + 4 ≤ μ(delta n) + μ s + 6
      -- because μ(delta n) is positive and 4 ≤ 6
      sorry

    have hrec :
        omega0 ^ (2 : Ordinal) *
          (omega0 ^ (mu n + mu s + (6 : Ordinal))
            + omega0 * (mu b + 1) + 1 + 1)
        < omega0 ^ A := by
      -- Intended route:
      --   (1) prove the parenthesized term is < ω^(A - 2)  (or any exponent < A
      --       leaving room for the leading ω²),
      --   (2) then multiply by ω²,
      --   (3) compare exponents via strict monotonicity of `ω^·`.
      --
      -- Concretely, you'd want a local lemma like
      --
      --   have hinner :
      --     omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 1 + 1
      --       < omega0 ^ (mu (delta n) + mu s + 4) := ...
      --
      -- and then:
      --
      --   calc
      --     omega0 ^ 2 * (...) < omega0 ^ 2 * omega0 ^ (...) := ...
      --     _ = omega0 ^ (2 + ...) := by rw [← opow_add]
      --     _ ≤ omega0 ^ A := by
      --         apply (Ordinal.isNormal_opow omega0).monotone
      --         ...
      --
      sorry

    have hsum :
        omega0 ^ (3 : Ordinal) * (mu s + 1)
          + omega0 ^ (2 : Ordinal) *
              (omega0 ^ (mu n + mu s + (6 : Ordinal))
                + omega0 * (mu b + 1) + 1 + 1) + 1
        < omega0 ^ A := by
      -- combine `hs`, `hrec`, and the final `1`
      -- using add-principality of `ω^A`
      sorry

    exact lt_of_lt_of_le hsum (le_add_right _ _)
```

### Important caution on `R_rec_succ`
This is the one place where I’m **not confident the current measure is actually strong enough**.

The reason is structural: after expanding
```lean
mu (app s (recDelta b s n))
```
you get a term
```lean
omega0 ^ 2 * ( ... + omega0 * (mu b + 1) + ... )
```
which creates an `ω^3 * (mu b + 1)`-type contribution, while
```lean
mu (recDelta b s (delta n))
```
only has
```lean
... + omega0 * (mu b + 1) + 1
```
outside the big leading `ω^A`.

So the missing sublemma in `hrec` is the one to test first. If that sublemma does not close, the issue is likely with the measure itself, not with missing library lemmas.

## Practical recommendation

If you want the minimal actionable next step:

1. **Fill `R_eq_refl` exactly** with the 3-line positivity proof above.
2. For **`R_eq_diff`**, introduce a local helper `hshift` and then prove a local bound
   ```lean
   have hmerge :
     mu (merge a b) + 1 < omega0 ^ (mu a + mu b + 4) := ...
   ```
   After that, the rest is just one multiplication by `ω^4` and one monotonicity step.
3. For **`R_rec_succ`**, try to prove the analogue
   ```lean
   have hinner :
     mu (recDelta b s n) + 1 < omega0 ^ (mu (delta n) + mu s + 4) := ...
   ```
   If this is where it fails, that’s the real obstruction.
