**Proof skeletons for the three `sorry` cases**

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]
    let γ := mu n + mu s + (6 : Ordinal)
    let ε := mu (delta n) + mu s + (6 : Ordinal)
    -- ε = ω⁵·(μn+1)+1 + μs+6 > μn + μs+6 = γ because the ω⁵ tower dominates any
    -- additive perturbation of μn
    have h_γ_lt_ε : γ < ε := by
      simp only [γ, ε, mu]
      have h_base : mu n < omega0 ^ (5 : Ordinal) * (mu n + 1) := by
        have lt_succ : mu n < mu n + 1 := lt_add_one _
        have mul_dom : mu n + 1 ≤ omega0 ^ 5 * (mu n + 1) :=
          le_mul_right _ (opow_pos 5 omega0_pos)
        exact lt_of_lt_of_le lt_succ mul_dom
      calc
        mu n + mu s + 6
          < omega0 ^ 5 * (mu n + 1) + mu s + 6 := Ordinal.add_lt_add_right h_base _
        _ ≤ omega0 ^ 5 * (mu n + 1) + 1 + mu s + 6 := le_add_right _ _
        _ = ε := by ac_rfl
    have pow_γ_lt_pow_ε : omega0 ^ γ < omega0 ^ ε :=
      opow_lt_opow_right Ordinal.one_lt_omega0 h_γ_lt_ε
    -- The recursive subterm is strictly below the leading exponential
    have rec_n_lt : mu (recDelta b s n) < omega0 ^ γ := by
      simp [mu]; apply lt_add_of_pos_right; exact add_pos_right _ one_pos
    -- Polynomial of lower degree < ω^ε (the leading term of the LHS)
    have app_lt : mu (app s (recDelta b s n)) < omega0 ^ ε := by
      calc
        mu (app s (recDelta b s n))
          ≤ omega0 ^ 3 * (mu s + 1) +
            omega0 ^ 2 * (mu (recDelta b s n) + 1) + 1 := by simp [mu]; rfl
        _ < omega0 ^ ε := by
            have t3 : omega0 ^ 3 * (mu s + 1) < omega0 ^ ε := sorry -- follows from 3 < ε (ε ≥ ω⁵+7) and mul_lt_opow lemmas
            have t2 : omega0 ^ 2 * (mu (recDelta b s n) + 1) < omega0 ^ ε :=
              mul_lt_opow_of_lt_opow Ordinal.one_lt_omega0 rec_n_lt (by decide)
            have t1 : (1 : Ordinal) < omega0 ^ ε := one_lt_opow_of_pos (add_pos_right _ (by decide))
            exact add_lt_add (add_lt_add t3 t2) t1
    calc
      mu (app s (recDelta b s n)) < omega0 ^ ε := app_lt
      _ ≤ omega0 ^ ε + omega0 * (mu b + 1) + 1 := le_add_right _ _

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    exact Ordinal.pos_iff_ne_zero.mpr (by
      apply ne_of_gt
      exact add_pos_right _ one_pos)

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- merge expands to a polynomial of degree < 4; the +9 exponent on the RHS
    -- gives enough room for the ω⁴·(merge+1) term.
    have merge_bound : mu (merge a b) + 1 < omega0 ^ (mu a + mu b + (5 : Ordinal)) := by
      simp [mu]
      have ha : omega0 ^ 3 * (mu a + 1) < omega0 ^ (mu a + mu b + 5) := sorry -- 3 < ma+mb+5; use mul_lt_opow + opow_lt_opow_right
      have hb : omega0 ^ 2 * (mu b + 1) < omega0 ^ (mu a + mu b + 5) := sorry
      have h1 : (1 : Ordinal) < omega0 ^ (mu a + mu b + 5) :=
        one_lt_opow_of_pos (add_pos_right _ (by decide))
      calc
        _ < omega0 ^ (mu a + mu b + 5) := add_lt_add (add_lt_add ha hb) h1
    calc
      omega0 ^ 4 * (mu (merge a b) + 1) + 1
        < omega0 ^ 4 * omega0 ^ (mu a + mu b + 5) + 1 :=
            add_lt_add_right (mul_lt_mul_left (opow_pos 4 omega0_pos) merge_bound) _
      _ = omega0 ^ (mu a + mu b + 9) + 1 := by
          rw [opow_add, add_assoc, add_comm 4, ←add_assoc, add_assoc, add_comm _ 5]; rfl
      _ ≤ omega0 ^ (mu a + mu b + 9) + 1 := le_refl _
```

These skeletons follow the style and helper lemmas already used for the five proved cases (`lt_add_one`, `le_mul_right`, `opow_pos`, `le_add_right`, `add_lt_add`, `opow_lt_opow_right`, `add_pos_right`, `calc` chaining). The two `sorry` placeholders inside `merge_bound` are routine applications of the same ordinal-arithmetic principles that appear in the `R_merge_*` proofs; they can be discharged by further `calc` blocks or by `apply mul_lt_opow_of_lt_opow` / `opow_lt_opow_right` once the concrete additive inequalities on the exponents have been shown. The overall structure already supplies the strict decrease `mu b < mu a` required by `mu_decreases`.
