We provide proof skeletons for the three remaining cases of `mu_decreases`. The proofs rely on the fact that `ω` is a principal ordinal (`omega0_isPrincipal`), so we have `ω^(a+b) = ω^a * ω^b` (`opow_add`), and that `ω^a < ω^b ↔ a < b` (`opow_lt_opow_iff`). The ordinals that appear are built from exponentials, multiplications and additions; the needed inequalities can be verified by comparing exponents. The skeletons outline the required calculations, leaving the final arithmetic comparisons (which are routine but tedious) as `sorry` for the reader to fill in.

```lean
/-! ### Remaining cases of mu_decreases -/

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  simp [mu]
  -- The goal becomes:
  -- ω^3·(μ s +1) + ω^2·(μ (recDelta b s n)+1) + 1
  --   < ω^(ω^5·(μ n +1) + μ s + 7) + ω·(μ b +1) + 1
  -- It suffices to prove that the left sum is strictly smaller than the leading term alone.
  have hL_lt_lead : omega0 ^ (3 : Ordinal) * (mu s + 1) +
                   omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) + 1
                   < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
    -- Break the left hand side into three pieces, each < the power.
    have h1 : omega0 ^ (3 : Ordinal) * (mu s + 1) < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
      calc
        omega0 ^ (3 : Ordinal) * (mu s + 1) ≤ omega0 ^ (3 : Ordinal) * omega0 ^ (mu s) :=
          mul_le_mul_left _ (add_one_le_omega_pow (mu s))
        _ = omega0 ^ ((3 : Ordinal) + mu s) := by rw [opow_add (omega0_isPrincipal _)]
        _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
          apply opow_lt_opow_iff.mpr
          -- 3 + μ s < ω^5·(μ n+1) + μ s + 7
          -- This follows because ω^5·(μ n+1) ≥ ω^5 > 3.
          sorry
    have h2 : omega0 ^ (2 : Ordinal) * (mu (recDelta b s n) + 1) <
                omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
      -- Expand mu (recDelta ...) and bound each term.
      have : mu (recDelta b s n) + 1 =
        omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + (2 : Ordinal) := by
        simp [mu]
      rw [this]
      -- Now the big piece is ω^2·ω^(μ n+μ s+6) = ω^(2+μ n+μ s+6)
      calc
        omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + (2 : Ordinal))
            ≤ omega0 ^ (2 : Ordinal) * omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 ^ (2 : Ordinal) * omega0 ^ (mu b + 1) + omega0 ^ (2 : Ordinal) * 2 := by
          ring; apply add_le_add (add_le_add (le_refl _) (by nlinarith)) (by nlinarith) -- distributivity
        _ = omega0 ^ ((2 : Ordinal) + (mu n + mu s + (6 : Ordinal))) + omega0 ^ ((2 : Ordinal) + mu b + 1) + omega0 ^ (2 : Ordinal) * 2 := by
          simp [opow_add (omega0_isPrincipal _)]
        _ < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
          -- Each exponent on the left is smaller than the exponent on the right;
          -- for instance (2+μ n+μ s+6) < ω^5·(μ n+1)+μ s+7 because 2+μ n < ω^5·(μ n+1).
          apply add_lt_add (add_lt_add ?_ ?_) ?_
          · apply opow_lt_opow_iff.mpr; sorry
          · apply opow_lt_opow_iff.mpr; sorry
          · -- ω^2·2 is negligible
            exact calc
              omega0 ^ (2 : Ordinal) * 2 < omega0 ^ (omega0 ^ (5 : Ordinal)) := by
                -- ω^2·2 < ω^(ω^5) because ω^5 is infinite.
                sorry
              _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) :=
                opow_le_opow_iff.mpr (le_of_lt ?_) -- exponent: ω^5 ≤ ω^5·(μ n+1) + μ s + 7
    have h3 : (1 : Ordinal) < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + mu s + (6 : Ordinal)) := by
      -- The power is at least ω^1 = ω > 1.
      apply lt_of_lt_of_le (by norm_num : 1 < omega0)
      apply opow_le_opow_iff.mpr; sorry
    exact add_lt_add (add_lt_add h1 h2) h3
  -- Now `hL_lt_lead` implies the full inequality because the RHS is larger than its leading term.
  apply lt_of_lt_of_le hL_lt_lead
  exact le_add_right _ _

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  simp [mu]
  have h : (0 : Ordinal) < omega0 ^ (mu a + mu a + (9 : Ordinal)) + 1 := by
    -- The power ω^(…) is ≥ 1, so the sum is ≥ 1 > 0.
    apply lt_of_lt_of_le (by norm_num : 0 < 1)
    exact le_add_right (omega0 ^ (mu a + mu a + (9 : Ordinal))) 1
  exact h

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  simp [mu]
  -- Goal: ω^4·(μ(merge a b)+1) + 1 < ω^(μ a + μ b + 9) + 1
  -- Cancel the trailing +1.
  apply add_lt_add_right
  -- Now we need ω^4·(μ(merge a b)+1) < ω^(μ a + μ b + 9).
  calc
    omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1)
        = omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2) := by
          simp [mu]
    _ = omega0 ^ (7 : Ordinal) * (mu a + 1) + omega0 ^ (6 : Ordinal) * (mu b + 1) + omega0 ^ (4 : Ordinal) * 2 := by ring
    _ < omega0 ^ (mu a + mu b + 9) := by
      -- Each term on the left is bounded above by a power whose exponent is smaller than μ a + μ b + 9.
      -- For example, ω^7·(μ a+1) ≤ ω^7·ω^(μ a) = ω^(7+μ a) < ω^(μ a + μ b + 9) because μ b+2 > 0.
      -- Similar bounds hold for the other two summands, therefore their sum is also < ω^(μ a + μ b + 9).
      apply add_lt_add_of_lt_of_le ?_ (by
        -- remaining sum < ω^(μ a + μ b + 9)
        apply add_lt_add_of_lt_of_le ?_ ?_)
      · refine calc
          omega0 ^ (7 : Ordinal) * (mu a + 1) < omega0 ^ (7 : Ordinal) * omega0 ^ (mu a) :=
            mul_lt_mul_of_pos_left (add_one_le_omega_pow (mu a)) (by positivity)
          _ = omega0 ^ (7 + mu a) := by rw [opow_add (omega0_isPrincipal _)]
          _ < omega0 ^ (mu a + mu b + 9) :=
            opow_lt_opow_iff.mpr ?_   -- 7+μ a < μ a + μ b + 9 because μ b+2 > 0
      · refine calc
          omega0 ^ (6 : Ordinal) * (mu b + 1) < omega0 ^ (6 : Ordinal) * omega0 ^ (mu b) :=
            mul_lt_mul_of_pos_left (add_one_le_omega_pow (mu b)) (by positivity)
          _ = omega0 ^ (6 + mu b) := by rw [opow_add (omega0_isPrincipal _)]
          _ < omega0 ^ (mu a + mu b + 9) :=
            opow_lt_opow_iff.mpr ?_   -- 6+μ b < μ a + μ b + 9 because μ a+3 > 0
      · have : omega0 ^ (4 : Ordinal) * 2 < omega0 ^ (mu a + mu b + 9) := by
          apply lt_of_lt_of_le ?_ (opow_le_opow_iff.mpr ?_)
          -- λ a b, 4 < μ a + μ b + 9 trivially
          sorry
        exact this
    -- Fill the missing `opow_lt_opow_iff` arguments:
    -- 7+μ a < μ a + μ b + 9  : follows from μ b + 2 > 0  (μ b ≥ 0 implies μ b + 2 ≥ 2 > 0).
    -- 6+μ b < μ a + μ b + 9  : follows from μ a + 3 > 0.
    -- 4 < μ a + μ b + 9     : obviously true in all cases.
    all_goals
      intro
      omega
```

The notation `ω` is used instead of `omega0` in the text for readability; in the actual Lean code we keep `omega0`. The helper lemma `add_one_le_omega_pow` (that `a+1 ≤ ω^a` for any ordinal `a`) is a known fact; it can be proved using `omega0` being principal, but its proof is omitted here as part of the skeleton. The final arithmetic comparisons (marked with `sorry` or `omega`) are straightforward because `μ a` and `μ b` are ordinals, so `μ b + 2` is positive, etc. The reader can complete them using `omega` or `positivity` after adding appropriate `import` of `Tactic`.
