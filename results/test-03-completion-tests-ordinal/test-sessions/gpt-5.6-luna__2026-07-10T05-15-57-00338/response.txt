A useful way to organize the three remaining cases is to isolate the ordinal-arithmetic comparisons as local facts. After `simp [mu]`, the goals are purely ordinal inequalities.

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp [mu]

    let B : Ordinal := mu b
    let S : Ordinal := mu s
    let N : Ordinal := mu n

    -- The exponent occurring in the source measure dominates the
    -- exponent introduced by the recursive call.
    have hexp :
        N + S + 8 <
          (omega0 ^ 5 * (N + 1) + 1) + S + 6 := by
      -- Prove this using:
      --   N < omega0 ^ 5 * (N + 1)
      -- followed by monotonicity of ordinal addition.
      have hN : N < omega0 ^ 5 * (N + 1) := by
        calc
          N < N + 1 := lt_add_one N
          _ ≤ omega0 ^ 5 * (N + 1) :=
            le_mul_right _ (opow_pos 5 omega0_pos)
      -- The remaining finite constants are absorbed by the strict
      -- inequality above.
      sorry

    -- Bound the measure of the recursive call by an omega-power.
    have hrec :
        omega0 ^ 2 *
            (omega0 ^ (N + S + 6) + omega0 * (B + 1) + 1 + 1)
          <
        omega0 ^ (N + S + 8) := by
      -- Expand ordinal multiplication and use that the exponent on the
      -- right is strictly larger than every exponent on the left.
      -- This is a Cantor-normal-form / omega-power comparison.
      sorry

    -- The first summand of `mu (app s (recDelta b s n))` is handled
    -- similarly.
    have happ :
        omega0 ^ 3 * (S + 1) <
          omega0 ^ (N + S + 8) := by
      -- Use `S < N + S + 8` and monotonicity of `x ↦ omega0 ^ x`.
      sorry

    calc
      omega0 ^ 3 * (S + 1) +
          omega0 ^ 2 *
            (omega0 ^ (N + S + 6) + omega0 * (B + 1) + 1 + 1) + 1
          <
          omega0 ^ (N + S + 8) + 1 := by
            -- Combine `happ` and `hrec`, then use ordinal addition
            -- monotonicity.
            sorry
      _ ≤ omega0 ^
            ((omega0 ^ 5 * (N + 1) + 1) + S + 6) +
            omega0 * (B + 1) + 1 := by
            -- Apply `opow_le_opow` or the corresponding monotonicity
            -- lemma to `hexp`, then weaken by adding the remaining tail.
            sorry
```

The reflexivity rule only needs positivity:

```lean
-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp [mu]
    calc
      0 < 1 := zero_lt_one
      _ ≤ omega0 ^ (mu a + mu a + 9) + 1 :=
        le_add_left _ _
```

For the `eq_diff` case, first bound the expanded `integrate (merge a b)` measure by a single omega-power.

```lean
-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    simp [mu]

    let A : Ordinal := mu a
    let B : Ordinal := mu b

    have hmerge :
        omega0 ^ 4 *
            (omega0 ^ 3 * (A + 1) +
             omega0 ^ 2 * (B + 1) + 1 + 1)
          <
        omega0 ^ (A + B + 8) := by
      -- Expand the products and compare Cantor-normal-form exponents.
      --
      -- The relevant estimates are:
      --   4 + 3 < A + B + 8
      --   4 + 2 < A + B + 8
      --
      -- with the small cases A = B = 0 handled directly. In the
      -- nonzero cases, use positivity of A + 1 and B + 1 together
      -- with monotonicity of ordinal addition and exponentiation.
      sorry

    have hexp :
        A + B + 8 < A + B + 9 := by
      exact lt_add_one _

    calc
      omega0 ^ 4 *
          (omega0 ^ 3 * (A + 1) +
           omega0 ^ 2 * (B + 1) + 1 + 1) + 1
          <
          omega0 ^ (A + B + 8) + 1 := by
            exact add_lt_add_right hmerge _
      _ <
          omega0 ^ (A + B + 9) + 1 := by
            -- Strict monotonicity of `x ↦ omega0 ^ x`.
            exact add_lt_add_right
              (opow_lt_opow_right (by exact omega0_gt_one) hexp) _
```

In practice, the two `sorry` blocks in the `rec_succ` case and the first one in `eq_diff` are best discharged by proving a small collection of reusable lemmas about omega powers, for example:

```lean
have hpow_mono :
    ∀ {x y : Ordinal}, x < y → omega0 ^ x < omega0 ^ y := by
  intro x y hxy
  exact opow_lt_opow_right (by exact omega0_gt_one) hxy

have hlinear :
    ∀ x : Ordinal, x < omega0 ^ 5 * (x + 1) := by
  intro x
  calc
    x < x + 1 := lt_add_one x
    _ ≤ omega0 ^ 5 * (x + 1) :=
      le_mul_right _ (opow_pos 5 omega0_pos)
```

The remaining arithmetic is then a sequence of `calc` steps using `add_lt_add_left`, `add_lt_add_right`, monotonicity of ordinal multiplication, and `hpow_mono`; normalization of the displayed expressions should be done before those comparisons with `simp only [Ordinal.add_assoc, Ordinal.add_left_comm, Ordinal.add_comm]` as needed.
