```lean
| _, _, Step.R_eq_diff a b => by
    simp [mu]
    -- Key observation: ω^(μa + μb + 9) dominates ω^4 * (ω^3 * (μa + 1) + ω^2 * (μb + 1) + 2)
    have h1 : omega0 ^ (3 : Ordinal) * (mu a + 1) < omega0 ^ (mu a + mu b + 9) := by
      apply opow_lt_opow_of_lt (lt_add_of_pos_right _ (by simp))
      apply add_lt_add_right
      apply opow_lt_opow_of_lt (lt_add_one 5)
      exact Nat.lt_succ_self 5
    have h2 : omega0 ^ (2 : Ordinal) * (mu b + 1) < omega0 ^ (mu a + mu b + 9) := by
      apply opow_lt_opow_of_lt (lt_add_of_pos_right _ (by simp))
      apply add_lt_add_right
      apply opow_lt_opow_of_lt (lt_add_one 6)
      exact Nat.lt_succ_self 6
    have h3 : 2 < omega0 ^ (mu a + mu b + 9) := by
      apply lt_opow_of_lt (lt_add_of_pos_right _ (by simp))
      exact Nat.lt_succ_self 0
    calc mu (integrate (merge a b))
      _ = omega0 ^ (4 : Ordinal) * (mu (merge a b) + 1) + 1 := rfl
      _ = omega0 ^ (4 : Ordinal) *
          (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + 1 + 1) + 1 := rfl
      _ < omega0 ^ (4 : Ordinal) *
          (omega0 ^ (mu a + mu b + 9) + omega0 ^ (mu a + mu b + 9) + omega0 ^ (mu a + mu b + 9)) + 1 := by
          apply mul_lt_mul_left (opow_pos 4 omega0_pos)
          apply add_lt_add
          · exact h1
          apply add_lt_add
          · exact h2
          · exact h3
      _ = omega0 ^ (4 : Ordinal) * (3 * omega0 ^ (mu a + mu b + 9)) + 1 := by simp
      _ < omega0 ^ (4 : Ordinal) * (omega0 ^ (mu a + mu b + 9 + 1)) + 1 := by
          apply add_lt_add_left
          apply mul_lt_mul_left (opow_pos 4 omega0_pos)
          apply mul_lt_opow (by simp) (by simp)
      _ = omega0 ^ (4 + mu a + mu b + 9 + 1) + 1 := by rw [opow_add]

          apply add_lt_add_left
          apply opow_lt_opow_of_lt (by simp)
          exact Nat.lt_succ_self 14
```

These proofs rely on the fact that higher ω-powers dominate lower ones, and that ordinal addition and multiplication preserve order in specific ways. The key is to show that the exponent in the right-hand side's measure is sufficiently large to dominate all terms in the left-hand side's measure.
