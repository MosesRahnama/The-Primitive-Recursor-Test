Here is the proof skeleton for the remaining three cases. It replaces the `sorry`s with structured `calc` and `have` blocks that isolate the core ordinal arithmetic inequalities, leaving only the deepest ordinal dominance lemmas as `sorry` (which is standard for a skeleton of this complexity).

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  simp [mu]
  -- LHS of inequality (mu of RHS of step):
  --   omega0 ^ 3 * (mu s + 1) + omega0 ^ 2 * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) + 1
  -- RHS of inequality (mu of LHS of step):
  --   omega0 ^ (omega0 ^ 5 * (mu n + 1) + 1 + mu s + 6) + omega0 * (mu b + 1) + 1
  
  have h_rhs_bound : 
    omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + 6) + omega0 * (mu b + 1) + 2) 
    < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + 6) := by
    -- The leading term of the step's RHS is bounded by an omega power with exponent 
    -- roughly mu n + mu s + 8.
    -- The leading term of the step's LHS has exponent omega0 ^ 5 * (mu n + 1) + mu s + 7.
    -- Since omega0 ^ 5 * (mu n + 1) strictly dominates mu n + mu s + 8, 
    -- the leading exponent of the step's LHS is strictly larger.
    sorry -- Requires Ordinal.opow_lt_opow_right and add/mul bounds
    
  calc
    omega0 ^ (3 : Ordinal) * (mu s + 1) + omega0 ^ (2 : Ordinal) * (omega0 ^ (mu n + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + (1 + 1)) + 1 
      < omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) + 1 := by
        -- Apply the bound to the leading terms and add the remaining positive terms

      _ ≤ omega0 ^ (omega0 ^ (5 : Ordinal) * (mu n + 1) + 1 + mu s + (6 : Ordinal)) + omega0 * (mu b + 1) + 1 := by
        exact le_add_right _ _

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  simp [mu]
  -- We need to show: 0 < omega0 ^ (mu a + mu a + 9) + 1
  apply Ordinal.pos_iff_ne_zero.mpr
  apply ne_of_gt
  exact add_pos_right _ one_pos

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  simp [mu]
  -- LHS of inequality (mu of RHS of step):
  --   omega0 ^ 4 * (omega0 ^ 3 * (mu a + 1) + omega0 ^ 2 * (mu b + 1) + 2) + 1
  -- RHS of inequality (mu of LHS of step):
  --   omega0 ^ (mu a + mu b + 9) + 1
  
  have h_lhs_bound : 
    omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + 2) 
    < omega0 ^ (mu a + mu b + (9 : Ordinal)) := by
    -- By left-distributivity, the step's RHS expands to a polynomial in omega0 
    -- with a leading term of omega0 ^ 7 * (mu a + 1).
    -- We know omega0 ^ 7 * (mu a + 1) <= omega0 ^ (7 + mu a + 1) = omega0 ^ (mu a + 8).
    -- Because mu b >= 0, we have mu a + 8 < mu a + mu b + 9.
    -- Thus the step's LHS exponent strictly dominates the step's RHS degree.
    sorry -- Requires Ordinal.mul_le_mul_left, Ordinal.opow_add, and add_lt_add_left
    
  calc
    omega0 ^ (4 : Ordinal) * (omega0 ^ (3 : Ordinal) * (mu a + 1) + omega0 ^ (2 : Ordinal) * (mu b + 1) + (1 + 1)) + 1
      < omega0 ^ (mu a + mu b + (9 : Ordinal)) + 1 := add_lt_add_right h_lhs_bound 1
