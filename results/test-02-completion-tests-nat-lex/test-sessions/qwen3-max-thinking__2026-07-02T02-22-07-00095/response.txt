| _, _, Step.R_int_delta t => by
  by_cases h_pos : 0 < kappa t
  · -- kappa decreases: 0 < kappa t = kappa(integrate (delta t))
    have hk : kappa void < kappa (integrate (delta t)) := by
      simp [kappa_void, kappa_integrate, kappa_delta]; exact h_pos
    exact drop_left hk
  · -- kappa ties (both 0), mu decreases: 0 < 1 + mu t
    have hk : kappa void = kappa (integrate (delta t)) := by
      simp [kappa_void, kappa_integrate, kappa_delta]; omega
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu_void, mu_integrate, mu_delta]; omega
    exact drop_right hmu hk
