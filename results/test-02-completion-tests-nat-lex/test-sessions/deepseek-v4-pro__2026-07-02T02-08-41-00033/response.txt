Here are the proof skeletons for the five remaining `Step` cases. They follow the lexicographic measure-decrease pattern used in the already-proved cases.

-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
  have hmu : mu void < mu (integrate (delta t)) := by
    simp [mu]
    omega
  by_cases hk : kappa t = 0
  · have hk_eq : kappa void = kappa (integrate (delta t)) := by
      simpa [kappa, hk]
    exact drop_right hmu hk_eq
  · have hk_lt : kappa void < kappa (integrate (delta t)) := by
      simpa [kappa] using Nat.pos_of_ne_zero hk
    exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
  have hkappa_lhs_val : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by
    simp [kappa]
  have hmu_lt : mu b < mu (recDelta b s void) := by
    simp [mu]
    omega
  by_cases hle : kappa s ≤ kappa b
  · have hk_eq : kappa b = kappa (recDelta b s void) := by
      simpa [hkappa_lhs_val, Nat.max_eq_left hle]
    exact drop_right hmu_lt hk_eq
  · have hlt : kappa b < kappa s := Nat.lt_of_not_ge hle
    have hk_lt : kappa b < kappa (recDelta b s void) := by
      simpa [hkappa_lhs_val, Nat.max_eq_right (Nat.le_of_lt hlt)] using hlt
    exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  set M := Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) with hM
  have hM_ge_ks : kappa s ≤ M := by
    apply le_trans (le_max_right (kappa b) (kappa s)) (le_max_left _ _)
  have hLHS : kappa (recDelta b s (delta n)) = M + 1 := by
    simp [kappa, hM]
  have hRHS : kappa (app s (recDelta b s n)) = M := by
    simp [kappa, hM, hM_ge_ks]
  have hk_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
    rw [hRHS, hLHS]
    omega
  exact drop_left hk_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  have hmu : mu void < mu (eqW a a) := by
    simp [mu]
    omega
  by_cases hk : kappa a = 0
  · have hk_eq : kappa void = kappa (eqW a a) := by
      simpa [kappa, hk]
    exact drop_right hmu hk_eq
  · have hk_lt : kappa void < kappa (eqW a a) := by
      simpa [kappa] using Nat.pos_of_ne_zero hk
    exact drop_left hk_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
    simp [mu]
    omega
  have hk_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by
    simp [kappa]
  exact drop_right hmu hk_eq
