Here is a compact proof skeleton that can be pasted in place of the five remaining `sorry`s.
It follows the pattern already used for the three finished rules:
* show that `mu` always drops;
* analyse whether `kappa` drops strictly or only ties;
* dispatch to the appropriate `drop_left` / `drop_right` helper.

```lean
-- integrate (delta t)  ⟶  void
| _, _, Step.R_int_delta t => by
    -- mu always drops by at least 1
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu]
    -- kappa may drop strictly (if κ t > 0) or tie (if κ t = 0)
    have hk_le : kappa void ≤ kappa (integrate (delta t)) := by
      simp [kappa]
    have hk_cases := Nat.eq_or_lt_of_le hk_le
    cases hk_cases with
    | inl hk_eq  =>                         -- tie: use mu-drop branch
        exact drop_right hmu hk_eq
    | inr hk_lt  =>                         -- strict κ-drop branch
        exact drop_left hk_lt

-- recDelta b s void  ⟶  b
| _, _, Step.R_rec_zero b s => by
    -- mu drops by 3 + μ s
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]
    -- kappa either ties (κ b maximum) or drops (κ s the maximum)
    have hk_le : kappa b ≤ kappa (recDelta b s void) := by
      simp [kappa]
    have hk_cases := Nat.eq_or_lt_of_le hk_le
    cases hk_cases with
    | inl hk_eq  => exact drop_right hmu hk_eq   -- tie case
    | inr hk_lt  => exact drop_left hk_lt        -- strict drop case

-- recDelta b s (delta n)  ⟶  app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- kappa drops by the added “+1” of the δ-argument
    have hk : kappa (app s (recDelta b s n)) <
               kappa (recDelta b s (delta n)) := by
      simpa [kappa] using
        Nat.lt_succ_self (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n))
    exact drop_left hk                                   -- strict κ-drop

-- eqW a a  ⟶  void
| _, _, Step.R_eq_refl a => by
    have hmu : mu void < mu (eqW a a) := by
      simp [mu]
    have hk_le : kappa void ≤ kappa (eqW a a) := by
      simp [kappa]
    have hk_cases := Nat.eq_or_lt_of_le hk_le
    cases hk_cases with
    | inl hk_eq  => exact drop_right hmu hk_eq   -- κ ties
    | inr hk_lt  => exact drop_left hk_lt        -- κ drops

-- eqW a b  ⟶  integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa ties; mu drops by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```
