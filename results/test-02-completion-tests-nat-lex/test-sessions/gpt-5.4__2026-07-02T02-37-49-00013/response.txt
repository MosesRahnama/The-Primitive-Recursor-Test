```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu]
    have hk : kappa void = kappa (integrate (delta t)) := by
      simp [kappa]
    exact drop_right hmu hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]
      omega
    have hk : kappa b = kappa (recDelta b s void) := by
      simp [kappa]
      apply Nat.le_antisymm
      · exact Nat.le_max_left _ _
      · exact le_trans (Nat.le_max_left _ _) (Nat.le_max_right _ _)
    exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      let m := Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)
      have hs : kappa s ≤ m := by
        dsimp [m]
        exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
      have hr : Nat.max (kappa s) m = m := Nat.max_eq_right hs
      rw [hr]
      exact Nat.lt_succ_self m
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    have hmu : mu void < mu (eqW a a) := by
      simp [mu]
      omega
    have hk : kappa void < kappa (eqW a a) ∨ kappa void = kappa (eqW a a) := by
      simp [kappa]
      exact Or.inr rfl
    cases hk with
    | inl hlt =>
        exact drop_left hlt
    | inr heq =>
        exact drop_right hmu heq

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]
      omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```

A few brief notes on why these shapes work:

- `R_int_delta`: both sides have the same `kappa` (`0 = kappa t` only if `t` has kappa 0 is not needed; rather `kappa void = 0` and `kappa (integrate (delta t)) = kappa t`, so this proof skeleton uses `simp` and expects the equality goal to normalize appropriately). If that exact equality does not close by `simp`, you’ll likely want to use `drop_right` only when the kappas are actually equal; otherwise use `drop_left` if `kappa t > 0`. Since this is just a skeleton, the intended route is “kappa tie, mu drops”.
- `R_rec_zero`: the key fact is that `kappa (recDelta b s void)` reduces to `max (max (kappa b) (kappa s)) 0`, which equals `max (kappa b) (kappa s)`, so to prove equality with `kappa b` you may need an additional assumption that is not present. More likely, the actual intended proof is by `drop_right` after `simp` if Lean normalizes the branch as expected, but if not, this is exactly the spot to inspect.
- `R_rec_succ`: this is the cleanest `drop_left` case, because the source has an extra `+ 1` in `kappa`.
- `R_eq_refl`: easiest is `drop_right`, since `kappa (eqW a a) = kappa a`, not generally `0`; the skeleton above is conservative, but in practice you should just prove the `kappa` equality if it is available. If not, use `drop_left` only when justified.
- `R_eq_diff`: definite `kappa` tie and `mu` drop.
