The above shows why `R_rec_zero` should be done by **branching**: sometimes `kappa` ties, sometimes it drops. A clean complete skeleton is:

```lean
| _, _, Step.R_rec_zero b s => by
    by_cases hs : kappa s ≤ kappa b
    · have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_left hs]
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]
      exact drop_right hmu hk
    · have hs' : kappa b < kappa s := Nat.lt_of_not_ge hs
      have hk : kappa b < kappa (recDelta b s void) := by
        -- RHS = max (max kb ks) 0 = max kb ks, and hs' makes that = ks
        have hmax : Nat.max (kappa b) (kappa s) = kappa s := Nat.max_eq_right (Nat.le_of_lt hs')
        simp [kappa, hmax, hs']
      exact drop_left hk
```

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- kappa drops by exactly 1 on the recDelta spine
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      -- Let m := max (max kb ks) kn. LHS = max ks m = m, RHS = m+1.
      -- so LHS < RHS by Nat.lt_succ_self.
      simp [kappa]
      -- goal reduces to: max (kappa s) (max (max (kappa b) (kappa s)) (kappa n))
      --                  < max (max (kappa b) (kappa s)) (kappa n) + 1
      exact Nat.lt_succ_self _
    exact drop_left hk
