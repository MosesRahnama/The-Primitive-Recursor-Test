Here is a “fill-in” skeleton that shows how each of the remaining five cases can be discharged.
The idea is always the same as in the already proven cases:

•  first obtain the relevant `mu` / `kappa` (in-)equalities with `simp`,  
•  then decide whether we are in a “strict-κ-drop” situation (`drop_left`)
   or a “κ-tie / μ-drop” situation (`drop_right`),
•  finally apply the appropriate helper lemma.

`omega` is used for the elementary linear-arithmetic checks (as already done
in the provided proof for `R_merge_cancel`).

```
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- μ always drops by 1
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu]
    -- κ may drop (if κ t ≠ 0) or tie (if κ t = 0)
    by_cases hκ : kappa t = 0
    · -- κ ties
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hκ]
      exact drop_right hmu hk
    · -- κ strictly drops (0 < κ t)
      have hk_lt : kappa void < kappa (integrate (delta t)) := by
        have : 0 < kappa t := Nat.pos_of_ne_zero hκ
        simpa [kappa] using this
      exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- μ drops by the fixed overhead “3 + μ s”
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]; omega
    -- κ either ties or drops, depending on κ s
    by_cases hk_eq : kappa b = kappa (recDelta b s void)
    · exact drop_right hmu hk_eq
    · -- show κ b < κ (rec …)
      have hk_lt : kappa b < kappa (recDelta b s void) := by
        have hle : kappa b ≤ kappa (recDelta b s void) := by
          simp [kappa, Nat.le_max_left, Nat.le_max_left]
        exact Nat.lt_of_le_of_ne hle hk_eq
      exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- κ strictly drops by the “+1” that is removed
    have hk_lt :
        kappa (app s (recDelta b s n)) <
        kappa (recDelta b s (delta n)) := by
      simp [kappa]; omega
    exact drop_left hk_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- μ drops by the fixed overhead “4 + …”
    have hmu : mu void < mu (eqW a a) := by
      simp [mu]; omega
    -- κ may tie (if κ a = 0) or drop (otherwise)
    by_cases hκ : kappa a = 0
    · have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, hκ]
      exact drop_right hmu hk
    · have hk_lt : kappa void < kappa (eqW a a) := by
        have : 0 < kappa a := Nat.pos_of_ne_zero hκ
        simpa [kappa] using this
      exact drop_left hk_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- κ ties; μ drops by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```
