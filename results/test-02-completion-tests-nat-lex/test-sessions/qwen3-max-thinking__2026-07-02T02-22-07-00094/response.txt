-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
    have hk_void : kappa void = 0 := rfl
    have hk_left : kappa (integrate (delta t)) = kappa t := by simp [kappa]
    by_cases h_pos : 0 < kappa t
    · exact drop_left h_pos
    · have hk_eq : kappa void = kappa (integrate (delta t)) := by
          rw [hk_void, hk_left]
          have : kappa t = 0 := by omega
          rw [this]
          rfl
      exact drop_right hmu hk_eq

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
    have h_ge : kappa b ≤ kappa (recDelta b s void) := by
      simp [kappa]
      apply Nat.le_trans _ (Nat.le_max_left _ _)
      apply Nat.le_max_left
    by_cases h_lt : kappa b < kappa (recDelta b s void)
    · exact drop_left h_lt
    · have hk_eq : kappa b = kappa (recDelta b s void) := by
          have h_le : kappa (recDelta b s void) ≤ kappa b := by
            rw [Nat.not_lt_iff_le] at h_lt
            exact h_lt
          exact Nat.le_antisymm h_ge h_le
      exact drop_right hmu hk_eq

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk_left : kappa (recDelta b s (delta n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by simp
    have hk_rec : kappa (recDelta b s n) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by simp [kappa]
    have hk_right : kappa (app s (recDelta b s n)) = Nat.max (kappa s) (kappa (recDelta b s n)) := by simp
    rw [hk_rec] at hk_right
    have h_simp : Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) = 
      Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
      apply Nat.max_eq_right
      apply Nat.le_trans _ (Nat.le_max_left _ _)
      apply Nat.le_max_right
    rw [h_simp] at hk_right
    have h_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      rw [hk_left, hk_right]; omega
    exact drop_left h_lt
