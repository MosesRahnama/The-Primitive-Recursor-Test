Here is the proof skeleton for the five remaining cases of the `measure_decreases` theorem.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    have hk_le : kappa void ≤ kappa (integrate (delta t)) := by
      simp; exact Nat.zero_le (kappa t)
    rcases Nat.eq_or_lt_of_le hk_le with hk | hk
    · -- Case: kappa ties (kappa t = 0)
      have hmu : mu void < mu (integrate (delta t)) := by simp; omega
      exact drop_right hmu hk
    · -- Case: kappa drops (kappa t > 0)
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hk_le : kappa b ≤ kappa (recDelta b s void) := by
      simp [kappa]; exact Nat.le_max_left (kappa b) (kappa s)
    rcases Nat.eq_or_lt_of_le hk_le with hk | hk
    · -- Case: kappa ties (kappa s <= kappa b)
      have hmu : mu b < mu (recDelta b s void) := by simp; omega
      exact drop_right hmu hk
    · -- Case: kappa drops (kappa s > kappa b)
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      -- The goal simplifies to:
      -- max (kappa s) (max (max (kappa b) (kappa s)) (kappa n)) < max (max (kappa b) (kappa s)) (kappa n) + 1
      -- which is equivalent to:
      -- max (max (kappa b) (kappa s)) (kappa n) < max (max (kappa b) (kappa s)) (kappa n) + 1
      exact Nat.lt_succ_self _
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    have hk_le : kappa void ≤ kappa (eqW a a) := by
      simp; exact Nat.zero_le (kappa a)
    rcases Nat.eq_or_lt_of_le hk_le with hk | hk
    · -- Case: kappa ties (kappa a = 0)
      have hmu : mu void < mu (eqW a a) := by simp; omega
      exact drop_right hmu hk
    · -- Case: kappa drops (kappa a > 0)
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp
    exact drop_right hmu hk
```
