```lean
-- integrate (delta t) --> void : (0,0) <lex (k, 1+μt); either κ drops to 0 or μ drops
| _, _, Step.R_int_delta t => by
    simp only [measure, LexOrder]
    by_cases h : kappa t = 0
    · simp [h]; omega
    · simp [h]; apply Prod.Lex.left; omega

-- recDelta b s void --> b : κ(b) ≤ max(κb,κs), μb < 3+μb+μs; either κ drops or μ drops
| _, _, Step.R_rec_zero b s => by
    have hle : kappa b ≤ kappa (recDelta b s void) := by simp; apply Nat.le_max_left
    have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
    by_cases heq : kappa b = kappa (recDelta b s void)
    · exact drop_right hmu heq
    · have hlt : kappa b < kappa (recDelta b s void) := by omega
      exact drop_left hlt

-- recDelta b s (delta n) --> app s (recDelta b s n) : κ drops by 1
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp; omega
    exact drop_left hk

-- eqW a a --> void : (0,0) <lex (κa, 4+2·μa); either κ drops or μ drops
| _, _, Step.R_eq_refl a => by
    simp only [measure, LexOrder]
    by_cases h : kappa a = 0
    · simp [h]; omega
    · simp [h]; apply Prod.Lex.left; omega

-- eqW a b --> integrate (merge a b) : κ equal, μ drops by 1
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk
```

The skeleton follows the style of the already-proved cases (explicit `have` for the measure components, then `drop_left`/`drop_right`). For the three rules where `κ` may or may not drop we case on equality (the `by_cases` branches are exactly the two `drop_*` lemmas). All `omega` calls are purely arithmetic and succeed from the definitions of `kappa`/`mu`. This completes the `measure_decreases` theorem and therefore `strong_normalization`.
