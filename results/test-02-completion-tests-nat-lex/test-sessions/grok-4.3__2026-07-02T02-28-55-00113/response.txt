The remaining cases can be completed with the following proof skeleton (five replacement `| ... => by ...` arms). Each uses the same `drop_left`/`drop_right` helpers already present in the file; case splits on `kappa` are required only when that value may stay the same.

```lean
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · have hmu : mu void < mu (integrate (delta t)) := by simp [mu]
      have hk  : kappa void = kappa (integrate (delta t)) := by simp [kappa, h]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]; omega
      exact drop_left hk

| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b = kappa (recDelta b s void)
    · have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      exact drop_right hmu h
    · have hk : kappa b < kappa (recDelta b s void) := by simp [kappa]; omega
      exact drop_left hk

| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) <
              kappa (recDelta b s (delta n)) := by simp [kappa]; omega
    exact drop_left hk

| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · have hmu : mu void < mu (eqW a a) := by simp [mu]
      have hk  : kappa void = kappa (eqW a a) := by simp [kappa, h]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (eqW a a) := by simp [kappa]; omega
      exact drop_left hk

| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk
```
