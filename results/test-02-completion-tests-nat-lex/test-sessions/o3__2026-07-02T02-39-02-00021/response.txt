Lean-4 proof skeleton that fills the five remaining `sorry`s:

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- Either `kappa` strictly drops or it ties and `mu` drops.
    by_cases h : kappa t = 0
    · -- ❶  kappa ties, use μ–drop
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, h]
      have hμ : mu void < mu (integrate (delta t)) := by
        simp [mu]
      exact drop_right hμ hk
    · -- ❷  kappa drops strictly
      have hk : kappa void < kappa (integrate (delta t)) := by
        have : 0 < kappa t := Nat.pos_of_ne_zero h
        simpa [kappa] using this
      exact drop_left hk


-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- First look at the κ–components
    have hκ_le : kappa b ≤ kappa (recDelta b s void) := by
      simp [kappa]
    by_cases hκ_eq : kappa b = kappa (recDelta b s void)
    · -- ❶  κ ties, use μ–drop
      have hk : kappa b = kappa (recDelta b s void) := hκ_eq
      have hμ : mu b < mu (recDelta b s void) := by
        simp [mu]; omega
      exact drop_right hμ hk
    · -- ❷  κ drops strictly
      have hk : kappa b < kappa (recDelta b s void) :=
        Nat.lt_of_le_of_ne hκ_le hκ_eq
      exact drop_left hk


-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- κ-component drops by one
    have hk : kappa (app s (recDelta b s n))
            < kappa (recDelta b s (delta n)) := by
      simp [kappa]; omega
    exact drop_left hk


-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- Same pattern as the first case
    by_cases h : kappa a = 0
    · -- κ ties, μ drops
      have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, h]
      have hμ : mu void < mu (eqW a a) := by
        simp [mu]
      exact drop_right hμ hk
    · -- κ drops strictly
      have hk : kappa void < kappa (eqW a a) := by
        have : 0 < kappa a := Nat.pos_of_ne_zero h
        simpa [kappa] using this
      exact drop_left hk


-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- κ ties, but μ drops by one
    have hμ : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hμ hk
```

The pattern is always the same:

• decide whether `κ` strictly decreases or merely ties;  
• if it ties, show that `μ` strictly decreases;  
• then apply the corresponding helper lemma `drop_left` or `drop_right`.

All arithmetic facts are discharged by `simp` plus the linear-arithmetic tactic `omega`, exactly as already used in the proved cases.
