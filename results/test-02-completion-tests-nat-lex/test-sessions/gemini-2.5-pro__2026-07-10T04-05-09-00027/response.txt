Here is a proof skeleton for the five remaining cases of the `measure_decreases` theorem.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- We show that the measure decreases from `integrate (delta t)` to `void`.
    -- The measure of `void` is `(0, 0)`.
    -- The measure of `integrate (delta t)` is `(kappa t, 1 + mu t)`.
    -- We prove `(0, 0) <_lex (kappa t, 1 + mu t)` by cases on `kappa t`.
    by_cases h_kappa_zero : kappa t = 0
    · -- Case `kappa t = 0`: kappa ties, mu drops.
      -- `mu void = 0 < 1 + mu t = mu (integrate (delta t))`.
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h_kappa_zero]
      have hmu : mu void < mu (integrate (delta t)) := by simp; omega
      exact drop_right hmu hk
    · -- Case `kappa t > 0`: kappa drops.
      -- `kappa void = 0 < kappa t = kappa (integrate (delta t))`.
      have hk : kappa void < kappa (integrate (delta t)) := by
        simp; exact Nat.pos_of_ne_zero h_kappa_zero
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- We show that the measure decreases from `recDelta b s void` to `b`.
    -- The proof is by cases on the relative order of `kappa s` and `kappa b`.
    by_cases h_kappa : kappa s ≤ kappa b
    · -- Case `kappa s ≤ kappa b`: kappa ties, mu drops.
      -- `kappa (recDelta b s void) = max(kappa b, kappa s) = kappa b`.
      have hk : kappa b = kappa (recDelta b s void) := by
        simp; exact Nat.max_eq_left h_kappa
      -- `mu b < 3 + mu b + mu s = mu (recDelta b s void)`.
      have hmu : mu b < mu (recDelta b s void) := by simp; omega
      exact drop_right hmu hk
    · -- Case `kappa s > kappa b`: kappa drops.
      -- `kappa b < kappa s <= max(kappa b, kappa s) = kappa (recDelta b s void)`.
      have hk : kappa b < kappa (recDelta b s void) := by
        simp; exact Nat.lt_of_not_le h_kappa
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- We show measure decrease from `recDelta b s (delta n)` to `app s (recDelta b s n)`.
    -- `kappa` on the redex has a `+1` from the `delta n` argument.
    -- `kappa (recDelta b s (delta n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1`.
    -- The proof proceeds by cases on whether `n` is itself delta-wrapped.
    by_cases h_n_is_delta : (exists m, n = delta m)
    · -- Case `n = delta m`: The `kappa` of the inner `recDelta` also gets a `+1`.
      -- This causes `kappa` to tie. Analysis shows `mu` increases in this case.
      -- A proof of decrease would require a different measure.
      sorry
    · -- Case `n` is not delta-wrapped: `kappa` of the inner `recDelta` has no `+1`.
      -- This is sufficient for the overall `kappa` to strictly decrease.
      have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
        have h_n_not_delta : forall m, n ≠ delta m := by intro m; contrapose! h_n_is_delta; exact ⟨m, h_n_is_delta⟩
        simp [kappa, h_n_not_delta]
        rw [Nat.max_eq_right]
        · exact Nat.lt_succ_self _
        · exact Nat.le_trans (Nat.le_max_left _ _) (Nat.le_max_left _ _)
      exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- We show measure decrease from `eqW a a` to `void`.
    -- This is analogous to `R_int_delta`, by cases on `kappa a`.
    by_cases h_kappa_zero : kappa a = 0
    · -- Case `kappa a = 0`: kappa ties, mu drops.
      have hk : kappa void = kappa (eqW a a) := by simp [h_kappa_zero]
      have hmu : mu void < mu (eqW a a) := by simp; omega
      exact drop_right hmu hk
    · -- Case `kappa a > 0`: kappa drops.
      have hk : kappa void < kappa (eqW a a) := by
        simp; exact Nat.pos_of_ne_zero h_kappa_zero
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- We show measure decrease from `eqW a b` to `integrate (merge a b)`.
    -- `kappa` is `max(kappa a, kappa b)` for both terms, so it ties.
    -- `mu` for `eqW a b` is `4 + mu a + mu b`.
    -- `mu` for `integrate (merge a b)` is `1 + (2 + mu a + mu b) = 3 + mu a + mu b`.
    -- Thus `mu` strictly drops.
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp; omega
    exact drop_right hmu hk
```
