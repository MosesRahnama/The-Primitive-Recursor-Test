-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · -- Case: kappa t = 0, so kappa ties (both sides have kappa = 0)
      have hk : kappa void = kappa (integrate (delta t)) := by 
        simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by 
        simp [mu]
        omega
      exact drop_right hmu hk
    · -- Case: kappa t > 0, so kappa strictly decreases
      have hk : kappa void < kappa (integrate (delta t)) := by 
        simp [kappa]
        omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa s ≤ kappa b
    · -- Case: kappa s ≤ kappa b, so kappa ties (max (kappa b) (kappa s) = kappa b)
      have hk : kappa b = kappa (recDelta b s void) := by 
        simp [kappa]
        rw [Nat.max_eq_left h]
        rfl
      have hmu : mu b < mu (recDelta b s void) := by 
        simp [mu]
        omega
      exact drop_right hmu hk
    · -- Case: kappa s > kappa b, so kappa strictly decreases
      have hk : kappa b < kappa (recDelta b s void) := by 
        simp [kappa]
        have : kappa b < Nat.max (kappa b) (kappa s) := by 
          apply Nat.lt_max_of_lt_right
          omega
        linarith
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by

    -- When n is a delta, kappa ties but mu decreases.
    by_cases h : kappa (recDelta b s n) < Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1
    · -- Subcase: kappa strictly decreases
      have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by 
        simp [kappa] at h ⊢
        have : kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by 
          simp [Nat.le_max_left, Nat.le_max_of_le_left]
        have : Nat.max (kappa s) (kappa (recDelta b s n)) < Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by 
          apply Nat.max_lt_iff.mpr
          constructor
          · linarith [this]
          · linarith [h]
        linarith
      exact drop_left hk
    · -- Subcase: kappa ties (n must be a delta), mu strictly decreases
      have hk : kappa (app s (recDelta b s n)) = kappa (recDelta b s (delta n)) := by 
        simp [kappa] at h ⊢
        have : kappa (recDelta b s n) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by 
          omega
        simp [this, Nat.max_assoc]
        rw [Nat.max_eq_right]
        · rfl
        · -- Show kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)
          simp [Nat.le_max_left, Nat.le_max_of_le_left]
      have hmu : mu (app s (recDelta b s n)) < mu (recDelta b s (delta n)) := by 
        simp [mu]
        omega
      exact drop_right hmu hk

| _, _, Step.R_eq_refl a => by
