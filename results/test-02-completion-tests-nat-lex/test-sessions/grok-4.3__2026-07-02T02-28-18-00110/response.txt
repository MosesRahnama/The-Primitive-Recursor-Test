The remaining cases follow the same pattern as the proved merge rules (use `drop_left` on strict `kappa` decrease; otherwise use `drop_right` after establishing `kappa` equality + strict `mu` decrease). The skeletons are therefore:

```lean
| _, _, Step.R_int_delta t => by
    -- kappa ties or drops; mu always drops by ≥1
    cases (kappa t) with
    | zero =>
        have hmu : mu void < mu (integrate (delta t)) := by simp [mu]
        have hk  : kappa void = kappa (integrate (delta t)) := by simp
        exact drop_right hmu hk
    | succ _ =>
        have hk : kappa void < kappa (integrate (delta t)) := by simp
        exact drop_left hk

| _, _, Step.R_rec_zero b s => by
    -- kappa ties or drops; mu always drops by ≥3
    have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
    by_cases hk : kappa b = kappa (recDelta b s void)
    · exact drop_right hmu hk
    · have hk' : kappa b < kappa (recDelta b s void) := by
        simp [kappa]; omega
      exact drop_left hk'

| _, _, Step.R_rec_succ b s n => by
    -- kappa drops by exactly 1
    have hk : kappa (app s (recDelta b s n))
            < kappa (recDelta b s (delta n)) := by simp [kappa]
    exact drop_left hk

| _, _, Step.R_eq_refl a => by
    -- kappa ties or drops; mu always drops by ≥4
    have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
    by_cases hk : kappa void = kappa (eqW a a)
    · exact drop_right hmu hk
    · have hk' : kappa void < kappa (eqW a a) := by simp [kappa]; omega
      exact drop_left hk'

| _, _, Step.R_eq_diff a b => by
    -- kappa ties; mu drops by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk
```

(The `omega` steps are only needed where the `Nat.max` expressions are non-trivial; they can be replaced by explicit `Nat.le_max_*` reasoning if preferred.)

Final answer: see skeletons above (drop_left/right per kappa/mu)
