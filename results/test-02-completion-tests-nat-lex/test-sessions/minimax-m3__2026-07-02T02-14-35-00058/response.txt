Below is a proof skeleton for the five remaining `sorry` cases. The idea in every case is to identify whether `kappa` drops strictly (use `drop_left`) or merely ties while `mu` drops strictly (use `drop_right`), and to prove the relevant inequality with `simp` + `omega`.

```lean
-- integrate (delta t) --> void
--   kappa(integrate(delta t)) = kappa t
--   mu(integrate(delta t))   = 1 + mu t,  mu void = 0
| _, _, Step.R_int_delta t => by
    by_cases hk0 : kappa t = 0
    · -- kappa ties at 0; mu strictly decreases
      have hmu : mu void < mu (integrate (delta t)) := by
        simp [mu]; omega
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk0]
      exact drop_right hmu hk
    · -- kappa strictly drops to 0
      have hk : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]; omega
      exact drop_left hk
```

```lean
-- recDelta b s void --> b
--   kappa(recDelta b s void) = max (max (kappa b) (kappa s)) 0
--                           = max (kappa b) (kappa s)
--   mu  (recDelta b s void) = 3 + mu b + mu s
| _, _, Step.R_rec_zero b s => by
    by_cases hle : kappa s ≤ kappa b
    · -- max(kappa b)(kappa s) = kappa b  →  kappa ties
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]; omega
      have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa]
        -- goal: kappa b = max (max (kappa b) (kappa s)) 0
        rw [Nat.max_eq_right hle, Nat.max_zero_right]
      exact drop_right hmu hk
    · -- kappa b < kappa s, hence max(kappa b)(kappa s) > kappa b
      have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa]; omega
      exact drop_left hk
```

```lean
-- recDelta b s (delta n) --> app s (recDelta b s n)
--   LHS kappa = max (max (kappa b) (kappa s)) (kappa n) + 1
--   RHS kappa = kappa(app s (recDelta b s n))
--            = max (kappa s) (kappa (recDelta b s n))
--            = max (max (kappa b) (kappa s)) (kappa n)        (since s ≤ that max)
--   So RHS kappa = LHS kappa − 1  ⇒  strict kappa decrease.
| _, _, Step.R_rec_succ b s n => by
    -- Use the generated equation lemmas to unfold the non-`delta` `recDelta` case
    -- and the `Nat.max` laws to absorb the duplicate `kappa s`.
    have hk : kappa (app s (recDelta b s n))
              < kappa (recDelta b s (delta n)) := by
      have eL : kappa (app s (recDelta b s n))
              = Nat.max (kappa s) (kappa (recDelta b s n)) := rfl
      have eR : kappa (recDelta b s n)
              = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := rfl
      have eD : kappa (recDelta b s (delta n))
              = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := rfl
      rw [eL, eR, eD]
      -- max(kappa s) M = M  because kappa s ≤ M
      rw [Nat.max_eq_right
            (le_trans (Nat.le_max_left _ _) (Nat.le_max_left _ _))]
      -- now: M < M + 1
      exact Nat.lt_succ_self _
    exact drop_left hk
```
(If the generated `rfl` equations for `kappa` don't match exactly in your toolchain, replace `eL`/`eR`/`eD` with the corresponding `@[simp]` forms, e.g. `kappa_app` for `eL` and `kappa_rec_delta` for `eD`, and `simp [kappa]` (or `kappa.eq_1`/…) for `eR`. The `omega` shortcut `rw [eL, eR, eD]; omega` also closes the goal.)

```lean
-- eqW a a --> void
--   kappa(eqW a a) = max (kappa a) (kappa a) = kappa a
--   mu  (eqW a a) = 4 + mu a + mu a
| _, _, Step.R_eq_refl a => by
    by_cases hk0 : kappa a = 0
    · -- kappa ties at 0; mu strictly decreases
      have hmu : mu void < mu (eqW a a) := by
        simp [mu]; omega
      have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, hk0]
      exact drop_right hmu hk
    · -- kappa strictly drops to 0
      have hk : kappa void < kappa (eqW a a) := by
        simp [kappa]; omega
      exact drop_left hk
```

```lean
-- eqW a b --> integrate (merge a b)
--   kappa(integrate(merge a b)) = kappa(merge a b)
--                              = max (kappa a) (kappa b)         (ties with LHS)
--   mu  (integrate(merge a b)) = 1 + (2 + mu a + mu b)
--                              = 3 + mu a + mu b
--   mu  (eqW a b)             = 4 + mu a + mu b                  (strict drop of 1)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```

### Summary of the measure behaviour used

| rule | kappa LHS | kappa RHS | mu LHS | mu RHS | helper |
|---|---|---|---|---|---|
| `R_int_delta t` | `kappa t` | `0` | `1 + mu t` | `0` | case split on `kappa t = 0` |
| `R_rec_zero b s` | `max(kb, ks)` | `kb` | `3 + mb + ms` | `mb` | case split on `ks ≤ kb` |
| `R_rec_succ b s n` | `M + 1` | `M` | `4 + mb + ms + mn` | `1 + (mb + ms + mn + 3) = …` | `drop_left` (kappa ↓ by 1) |
| `R_eq_refl a` | `ka` | `0` | `4 + 2·ma` | `0` | case split on `ka = 0` |
| `R_eq_diff a b` | `max(ka, kb)` | `max(ka, kb)` | `4 + ma + mb` | `3 + ma + mb` | `drop_right` (mu ↓ by 1) |

In every case the chosen inequality (`Nat.lt_succ_self`, `Nat.le_max_left`, `Nat.max_eq_right`, or `omega`) is elementary arithmetic, so the `sorry`s can be filled in by the patterns above.
