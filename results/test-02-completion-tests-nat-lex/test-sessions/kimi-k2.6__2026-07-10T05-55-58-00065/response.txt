Here is a complete proof skeleton for the five remaining `sorry` cases, using the same style as the three proved cases above.

The main patterns are:
* **When `kappa` strictly decreases:** use `drop_left`.
* **When `kappa` is equal and `mu` strictly decreases:** use `drop_right`.
* For `R_rec_succ`, note that because `n` is a *variable* in the proof context, `kappa (recDelta b s n)` uses the fallback equation (without the `+1`), whereas the left-hand side `recDelta b s (delta n)` matches the first equation (with the `+1`). Hence `kappa` drops by one unconditionally.
* The arithmetic goals after `simp` are discharged with `omega` (or with explicit `Nat.max` lemmas where needed).

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · -- kappa ties (both sides are 0), mu drops from 1 + mu t to 0
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, h]
      have hmu : mu void < mu (integrate (delta t)) := by
        simp [mu]
        omega
      exact drop_right hmu hk
    · -- kappa drops strictly because kappa t > 0
      have hk : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]
        omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b < kappa s
    · -- kappa drops because max(kappa b, kappa s) = kappa s
      have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa]
        omega
      exact drop_left hk
    · -- kappa ties (max equals kappa b), mu drops by 3 + mu s
      have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa]
        omega
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]
        omega
      exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- Because n is a variable, kappa(recDelta b s n) uses the fallback
    -- (no +1), while the lhs has the +1. So kappa drops by one.
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      have h : Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n))
             = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
        apply Nat.max_eq_right
        apply le_trans (Nat.le_max_right (kappa b) (kappa s))
        apply Nat.le_max_left (Nat.max (kappa b) (kappa s)) (kappa n)
      rw [h]
      omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · -- kappa ties (both sides are 0), mu drops from 4 + 2*mu a to 0
      have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, h]
      have hmu : mu void < mu (eqW a a) := by
        simp [mu]
        omega
      exact drop_right hmu hk
    · -- kappa drops strictly because kappa a > 0
      have hk : kappa void < kappa (eqW a a) := by
        simp [kappa]
        omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa is max(kappa a, kappa b) on both sides
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    -- mu drops from 4 + mu a + mu b to 3 + mu a + mu b
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]
      omega
    exact drop_right hmu hk
```
