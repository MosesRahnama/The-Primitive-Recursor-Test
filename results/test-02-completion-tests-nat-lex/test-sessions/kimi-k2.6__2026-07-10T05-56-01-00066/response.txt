Here are proof skeletons for the five remaining cases. They follow the same structure as the existing proofs: establish either a strict drop in `kappa` (via `drop_left`) or a tie in `kappa` with a strict drop in `mu` (via `drop_right`). The `by_cases` tactic is used when the behavior of `kappa` depends on a comparison that is not statically known.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by simp at h ⊢; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b < kappa s
    · have hk : kappa b < kappa (recDelta b s void) := by simp [kappa]; omega
      exact drop_left hk
    · have hk : kappa b = kappa (recDelta b s void) := by simp [kappa]; omega
      have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      omega
    exact drop_left hk

    · have hk : kappa void < kappa (eqW a a) := by simp at h ⊢; omega

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by

    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
```
