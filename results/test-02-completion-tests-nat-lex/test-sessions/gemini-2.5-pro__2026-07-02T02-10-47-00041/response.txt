Here is a proof skeleton for the five remaining cases of the `measure_decreases` theorem.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- We proceed by cases on whether `kappa t` is zero or positive.
    cases h : kappa t
    · -- Case `kappa t = 0`: kappa ties at 0, and mu strictly decreases.
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by simp
      exact drop_right hmu hk
    · -- Case `kappa t > 0`: kappa strictly decreases.
      have hk : kappa void < kappa (integrate (delta t)) := by
        simp [h]
        apply Nat.zero_lt_succ
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- We proceed by cases on the comparison of `kappa b` and `kappa s`.
    cases Nat.lt_or_ge (kappa b) (kappa s)
    · next h_lt => -- Case `kappa b < kappa s`: kappa strictly decreases.
      -- In this case, `kappa (recDelta b s void)` is `kappa s`.
      have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_right (Nat.le_of_lt h_lt), h_lt]
      exact drop_left hk
    · next h_ge => -- Case `kappa b >= kappa s`: kappa ties, and mu strictly decreases.
      -- In this case, `kappa (recDelta b s void)` is `kappa b`.
      have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_left h_ge]
      have hmu : mu b < mu (recDelta b s void) := by simp; omega
      exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- The key idea is that `kappa` strictly decreases. The `delta` constructor
    -- is peeled off the third argument of `recDelta`, which is what `kappa` is
    -- designed to measure as a "recursion depth".
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      -- A full proof requires a case distinction on the structure of `n`
      -- to correctly apply the definition of `kappa` for `recDelta`.
      -- If `n` is not of the form `delta _`, the `+ 1` in the `kappa`
      -- definition for `recDelta`'s `delta` case is absent for the subterm
      -- `recDelta b s n`, leading to a strict decrease.
      sorry
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- We proceed by cases on whether `kappa a` is zero or positive.
    cases h : kappa a
    · -- Case `kappa a = 0`: kappa ties at 0, and mu strictly decreases.
      have hk : kappa void = kappa (eqW a a) := by simp [h]
      have hmu : mu void < mu (eqW a a) := by simp; omega
      exact drop_right hmu hk
    · -- Case `kappa a > 0`: kappa strictly decreases.
      have hk : kappa void < kappa (eqW a a) := by
        simp [h]
        apply Nat.zero_lt_succ
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- In this case, kappa ties and mu strictly decreases.
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp; omega
    exact drop_right hmu hk
```
