Here are the proof skeletons for the five remaining cases. Each follows the pattern of establishing either a strict decrease in `kappa` (using `drop_left`) or a tie in `kappa` with strict decrease in `mu` (using `drop_right`).

```tactics
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · -- kappa ties: kappa void = 0 = kappa t = kappa (integrate (delta t))
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      exact drop_right hmu hk
    · -- kappa decreases: 0 < kappa t
      have hk : kappa void < kappa (integrate (delta t)) := by simp; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa s ≤ kappa b
    · -- kappa ties: max (max (kappa b) (kappa s)) 0 = kappa b
      have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_left, h]
      have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      exact drop_right hmu hk
    · -- kappa decreases: kappa b < kappa s implies kappa b < max (max (kappa b) (kappa s)) 0
      have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_right, Nat.max_eq_left, h]
        all_goals omega
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- kappa decreases by exactly 1 (the +1 from the delta wrapper disappears)
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · -- kappa ties: kappa void = 0 = kappa a = kappa (eqW a a)
      have hk : kappa void = kappa (eqW a a) := by simp [h]
      have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
      exact drop_right hmu hk
    · -- kappa decreases: 0 < kappa a
      have hk : kappa void < kappa (eqW a a) := by simp; omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa ties: max (kappa a) (kappa b) = max (kappa a) (kappa b)
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    -- mu decreases: 3 + mu a + mu b < 4 + mu a + mu b
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    exact drop_right hmu hk
```
