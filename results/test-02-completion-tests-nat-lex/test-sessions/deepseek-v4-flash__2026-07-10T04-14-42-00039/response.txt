-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases hzero : kappa t = 0
    · have hmu : mu void < mu (integrate (delta t)) := by simp [mu]
      have hk : kappa void = kappa (integrate (delta t)) := by simp [kappa, hzero]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]
        omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b < kappa s
    · have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa, h]
      exact drop_left hk
    · have hle : kappa s ≤ kappa b := by omega
      have hmax : kappa (recDelta b s void) = kappa b := by
        simp [kappa, hle]
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]
      exact drop_right hmu hmax

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hM : kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
      apply le_trans (le_max_right _ _) (le_max_left _ _)
    have hM' : Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) :=
      Nat.max_eq_right hM
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa, hM']
      exact Nat.lt_succ_self _
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases hzero : kappa a = 0
    · have hmu : mu void < mu (eqW a a) := by simp [mu]

      exact drop_right hmu hk
    · have hk : kappa void < kappa (eqW a a) := by
        simp [kappa]
        omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]
    exact drop_right hmu hk
