For R_rec_succ b s n:
```lean
| _, _, Step.R_rec_succ b s n => by
    -- Compute kappa of the original: recDelta b s (delta n)
    have hk_orig : kappa (recDelta b s (delta n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by simp
    -- Compute kappa of the result: app s (recDelta b s n)
    have hk_res : kappa (app s (recDelta b s n)) = Nat.max (kappa s) (kappa (recDelta b s n)) := by simp
    -- Compute kappa of recDelta b s n (non-delta case)
    have hk_rec : kappa (recDelta b s n) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by simp [kappa]
    rw [hk_rec] at hk_res
    -- Simplify hk_res: max(kappa s, max(max(kappa b, kappa s), kappa n)) = max(max(kappa b, kappa s), kappa n) because the inner max is at least kappa s.
    have : Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
        apply Nat.max_eq_right
        apply Nat.le_trans (Nat.le_max_right _ _)
        apply Nat.le_max_left
    rw [this] at hk_res
    -- Now, hk_res: kappa (app ...) = M, where M = max(max(kappa b, kappa s), kappa n)
    -- hk_orig: kappa (recDelta ...) = M + 1
    have h_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
        rw [hk_res, hk_orig]
        omega
    exact drop_left h_lt
```
