Below is the skeleton I would use for the five `sorry`s.

Four of them should go through with the current file. The one exception is `R_rec_succ`: the natural proof skeleton is a `cases n`, but the `n = delta n'` branch is a real obstruction for the current `(kappa, mu)` measure.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases hk0 : kappa t = 0
    · have hmu : mu void < mu (integrate (delta t)) := by
        simp [mu]
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk0]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by
        have hpos : 0 < kappa t := Nat.pos_of_ne_zero hk0
        simpa [kappa] using hpos
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hle : kappa b ≤ kappa (recDelta b s void) := by
      -- If `simpa` does not discharge this directly in your environment,
      -- replace it by the explicit two-step `le_trans` proof.
      simpa [kappa] using (Nat.le_max_left (kappa b) (kappa s))
    by_cases hk : kappa b = kappa (recDelta b s void)
    · have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]
        omega
      exact drop_right hmu hk
    · exact drop_left (lt_of_le_of_ne hle hk)
      -- or: exact drop_left (Nat.lt_of_le_of_ne hle hk)

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- Intended skeleton:
    --   `cases n`
    --   * in every non-`delta` branch, prove strict kappa drop with `drop_left`
    --   * in the `delta` branch, the current measure does NOT decrease
    cases n with
    | delta n' =>
        -- This branch is the problem:
        --   kappa (app s (recDelta b s (delta n')))
        --     = kappa (recDelta b s (delta (delta n')))
        -- and the RHS mu is actually larger.
        --
        -- Concretely, with b = void, s = void, n' = void:
        --   measure (recDelta void void (delta (delta void))) = (1, 3)
        --   measure (app void (recDelta void void (delta void))) = (1, 4)
        --
        -- So `LexOrder (measure rhs) (measure lhs)` is false here.
        --
        -- You will need to revise the measure (most likely `kappa`) to finish this case.
        sorry

    | void =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa void) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s void)) <
              kappa (recDelta b s (delta void)) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self (Nat.max (Nat.max (kappa b) (kappa s)) (kappa void)))
        exact drop_left hk

    | integrate t =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa (integrate t)) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s (integrate t))) <
              kappa (recDelta b s (delta (integrate t))) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self
              (Nat.max (Nat.max (kappa b) (kappa s)) (kappa (integrate t))))
        exact drop_left hk

    | merge t u =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa (merge t u)) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s (merge t u))) <
              kappa (recDelta b s (delta (merge t u))) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self
              (Nat.max (Nat.max (kappa b) (kappa s)) (kappa (merge t u))))
        exact drop_left hk

    | app t u =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa (app t u)) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s (app t u))) <
              kappa (recDelta b s (delta (app t u))) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self
              (Nat.max (Nat.max (kappa b) (kappa s)) (kappa (app t u))))
        exact drop_left hk

    | recDelta b' s' t =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa (recDelta b' s' t)) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s (recDelta b' s' t))) <
              kappa (recDelta b s (delta (recDelta b' s' t))) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self
              (Nat.max (Nat.max (kappa b) (kappa s)) (kappa (recDelta b' s' t))))
        exact drop_left hk

    | eqW t u =>
        have hs :
            kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa (eqW t u)) := by
          exact le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
        have hk :
            kappa (app s (recDelta b s (eqW t u))) <
              kappa (recDelta b s (delta (eqW t u))) := by
          simpa [kappa, Nat.max_eq_right hs] using
            (Nat.lt_succ_self
              (Nat.max (Nat.max (kappa b) (kappa s)) (kappa (eqW t u))))
        exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases hk0 : kappa a = 0
    · have hmu : mu void < mu (eqW a a) := by
        simp [mu]
        omega
      have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, hk0]

        simpa [kappa] using hpos
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]
      omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```

Important point: `R_rec_succ` is not just “hard”; with the current definitions it is actually false in the `n = delta n'` subcase. So the skeleton above is enough to finish 4/5 missing cases, and it isolates the exact branch where the measure must be changed.
