-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
  by_cases hk : kappa t > 0
  · have h : kappa void < kappa (integrate (delta t)) := by
      simpa [kappa] using hk
    exact drop_left h
  · have hk0 : kappa t = 0 := by omega
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu, hk0] ; omega
    have hk_eq : kappa void = kappa (integrate (delta t)) := by
      simp [kappa, hk0]
    exact drop_right hmu hk_eq

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
  by_cases hks : kappa s > kappa b
  · have hk : kappa b < kappa (recDelta b s void) := by
      simp [kappa] ; omega
    exact drop_left hk
  · have hk_eq : kappa b = kappa (recDelta b s void) := by
      simp [kappa] ; omega
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu] ; omega
    exact drop_right hmu hk_eq

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  cases' n with
  | void => 
    have hk : kappa (app s (recDelta b s void)) < kappa (recDelta b s (delta void)) := by
      simp [kappa] ; omega
    exact drop_left hk
  | delta n' =>
    -- When the argument is itself a delta, kappa ties and mu increases.
    -- The given measure does not decrease for this subcase; a refined measure
    -- would be required to complete the proof.
    sorry
  | integrate n' =>
    have hk : kappa (app s (recDelta b s (integrate n'))) < kappa (recDelta b s (delta (integrate n'))) := by
      simp [kappa] ; omega
    exact drop_left hk
  | merge a b' =>
    have hk : kappa (app s (recDelta b s (merge a b'))) < kappa (recDelta b s (delta (merge a b'))) := by
      simp [kappa] ; omega
    exact drop_left hk
  | app a b' =>
    have hk : kappa (app s (recDelta b s (app a b'))) < kappa (recDelta b s (delta (app a b'))) := by
      simp [kappa] ; omega
    exact drop_left hk
  | recDelta r r' n' =>
    have hk : kappa (app s (recDelta b s (recDelta r r' n'))) < kappa (recDelta b s (delta (recDelta r r' n'))) := by
      simp [kappa] ; omega
    exact drop_left hk
  | eqW a b' =>
    have hk : kappa (app s (recDelta b s (eqW a b'))) < kappa (recDelta b s (delta (eqW a b'))) := by
      simp [kappa] ; omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  have hmu : mu void < mu (eqW a a) := by
    simp [mu] ; omega
  have hk_eq : kappa void = kappa (eqW a a) := by
    simp [kappa]
  exact drop_right hmu hk_eq

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
    simp [mu] ; omega
  have hk_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by
    simp [kappa]
  exact drop_right hmu hk_eq
