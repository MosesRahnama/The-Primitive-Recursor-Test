Here are the proof sketches for the five remaining `Step` rules, each reducing the lexicographic measure `(kappa, mu)` either by a strict drop in `kappa` or by tied `kappa` and a strict drop in `mu`.

```lean4
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
  by_cases hk : kappa t = 0
  · -- kappa ties: both 0
    have hkappa_eq : kappa void = kappa (integrate (delta t)) := by
      simp [hk, kappa]
    have hmu_lt : mu void < mu (integrate (delta t)) := by
      simp [mu]; omega
    exact drop_right hmu_lt hkappa_eq
  · -- kappa void = 0 < kappa t = kappa (integrate (delta t))
    have hk_lt : kappa void < kappa (integrate (delta t)) := by
      simp [kappa]; exact Nat.pos_of_ne_zero hk
    exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
  by_cases hle : kappa s ≤ kappa b
  · -- kappa ties: max(b,s) = b
    have hkappa_eq : kappa b = kappa (recDelta b s void) := by
      simp [kappa, hle]
    have hmu_lt : mu b < mu (recDelta b s void) := by
      simp [mu]; omega
    exact drop_right hmu_lt hkappa_eq
  · -- kappa b < kappa s, so kappa strictly drops
    have hk_lt : kappa b < kappa (recDelta b s void) := by
      simp [kappa]; exact Nat.gt_of_not_le hle
    exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  have hk_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
    simp [kappa]; omega
  exact drop_left hk_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  by_cases hk : kappa a = 0
  · -- kappa ties: both 0
    have hkappa_eq : kappa void = kappa (eqW a a) := by
      simp [hk, kappa]
    have hmu_lt : mu void < mu (eqW a a) := by
      simp [mu]; omega
    exact drop_right hmu_lt hkappa_eq
  · -- kappa void = 0 < kappa a = kappa (eqW a a)
    have hk_lt : kappa void < kappa (eqW a a) := by
      simp [kappa]; exact Nat.pos_of_ne_zero hk
    exact drop_left hk_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  have hkappa_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by
    simp [kappa]
  have hmu_lt : mu (integrate (merge a b)) < mu (eqW a b) := by
    simp [mu]; omega
  exact drop_right hmu_lt hkappa_eq
```

Each case uses the helper lemmas `drop_left` (strict kappa decrease) or `drop_right` (equal kappa and strict mu decrease) after simplifying the measures with `simp` and, where needed, `omega` for natural number arithmetic.
