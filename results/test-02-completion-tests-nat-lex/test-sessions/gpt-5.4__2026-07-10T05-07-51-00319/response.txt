-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      -- after simp the goal is essentially
      -- max (kappa s) (max (max (kappa b) (kappa s)) (kappa n))
      --   < max (max (kappa b) (kappa s)) (kappa n) + 1
      -- show lhs ≤ M, then use Nat.lt_succ_of_le

      change Nat.max (kappa s) M < M + 1
      have hle : Nat.max (kappa s) M ≤ M := by
        apply Nat.max_le
        · exact le_trans (Nat.le_max_right _ _) (Nat.le_refl _)
          -- or:
          -- exact le_trans (Nat.le_max_right (kappa b) (kappa s)) (Nat.le_max_left _ _)
        · exact le_rfl
