Here is a compact proof skeleton that completes the five remaining rule–cases.  
Each case follows the same pattern as the three already proved:

• decide first on whether the primary component `kappa` really drops;  
• if it does, use `drop_left`;  
• otherwise use `drop_right`, having established strict decrease of `mu`.

``lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- μ always drops by 1
    have hμ : mu void < mu (integrate (delta t)) := by
      simp [mu]
    -- κ is either 0 (tie) or strictly positive
    by_cases hκ₀ : kappa t = 0
    · -- κ ties
      have hκ_eq : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hκ₀]
      exact drop_right hμ hκ_eq
    · -- κ strictly drops
      have hκ_lt : kappa void < kappa (integrate (delta t)) := by
        have : 0 < kappa t := Nat.pos_of_ne_zero hκ₀
        simpa [kappa] using this
      exact drop_left hκ_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- μ drops by 3 + μ s
    have hμ : mu b < mu (recDelta b s void) := by
      simp [mu]; omega
    -- analyse κ–behaviour
    have hκ_form : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by
      simp [kappa]
    by_cases hle : kappa s ≤ kappa b
    · -- κ ties
      have hκ_eq : kappa b = kappa (recDelta b s void) := by
        have : Nat.max (kappa b) (kappa s) = kappa b := Nat.max_eq_left hle
        simpa [hκ_form, this]
      exact drop_right hμ hκ_eq
    · -- κ strictly drops
      have hκ_lt : kappa b < kappa (recDelta b s void) := by
        have hlt : kappa b < kappa s := lt_of_not_ge hle
        have : Nat.max (kappa b) (kappa s) = kappa s :=
          Nat.max_eq_right (le_of_lt hlt)
        simpa [hκ_form, this] using hlt
      exact drop_left hκ_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- κ surely drops by one
    have hκ_lt : kappa (app s (recDelta b s n)) <
                 kappa (recDelta b s (delta n)) := by
      simp [kappa]; omega
    exact drop_left hκ_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- μ drops by 4 + 2 μ a
    have hμ : mu void < mu (eqW a a) := by
      simp [mu]; omega
    -- κ may drop or tie
    by_cases hκ₀ : kappa a = 0
    · -- tie
      have hκ_eq : kappa void = kappa (eqW a a) := by
        simp [kappa, hκ₀]
      exact drop_right hμ hκ_eq
    · -- strict drop
      have hκ_lt : kappa void < kappa (eqW a a) := by
        have : 0 < kappa a := Nat.pos_of_ne_zero hκ₀
        simpa [kappa] using this
      exact drop_left hκ_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- κ ties; μ drops by 1
    have hμ : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hκ_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hμ hκ_eq
```

The `omega` tactic (already used earlier in the file) discharges the simple linear-arithmetic goals that arise after `simp`.  Apart from these minor calculations, the proofs only rely on the previously supplied helper lemmas `drop_left` and `drop_right`, together with routine `Nat.max` reasoning.
