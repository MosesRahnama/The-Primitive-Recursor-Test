-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- The measure for `integrate (delta t)` is `(kappa t, 1 + mu t)`.
    -- The measure for `void` is `(0, 0)`.
    -- We need to show `(0,0)` is lexicographically smaller than `(kappa t, 1 + mu t)`.
    -- This can be shown by cases on `kappa t`.
    -- If `kappa t > 0`, then `0 < kappa t`, so the measure strictly decreases on the first component.
    -- If `kappa t = 0`, the first component is equal, so we check the second.
    -- `0 < 1 + mu t`, so the measure strictly decreases on the second component.
    -- The logic can be implemented with a `by_cases` or `cases` tactic.
    by_cases h_kappa : kappa t = 0
    · -- Case: kappa ties. We show mu drops.
      apply drop_right
      · -- Goal: mu void < mu (integrate (delta t))
        simp; omega
      · -- Goal: kappa void = kappa (integrate (delta t))
        simp [h_kappa]
    · -- Case: kappa drops.
      apply drop_left
      · -- Goal: kappa void < kappa (integrate (delta t))
        simp; exact Nat.pos_of_ne_zero h_kappa

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- The measure for `recDelta b s void` is `(max (kappa b) (kappa s), 3 + mu b + mu s)`.
    -- The measure for `b` is `(kappa b, mu b)`.
    -- We do a case analysis on the relationship between `kappa b` and `kappa s`.
    by_cases h_le : kappa s <= kappa b
    · -- Case `kappa s <= kappa b`:
      -- `max (kappa b) (kappa s) = kappa b`, so kappa ties.
      -- `mu b < 3 + mu b + mu s`, so mu drops.
      apply drop_right
      · -- Goal: mu b < mu (recDelta b s void)
        simp; omega
      · -- Goal: kappa b = kappa (recDelta b s void)
        simp [Nat.max_eq_left h_le]
    · -- Case `kappa b < kappa s`:
      -- `max (kappa b) (kappa s) = kappa s`, so `kappa b < kappa s`. Kappa drops.
      apply drop_left
      · -- Goal: kappa b < kappa (recDelta b s void)
        have h_lt : kappa b < kappa s := Nat.lt_of_not_le h_le
        simp [Nat.max_eq_right_of_lt h_lt]

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- For this rule, `mu` increases:
    -- `mu (app s (recDelta b s n))` is `1 + mu s + (3 + mu b + mu s + mu n)`.
    -- `mu (recDelta b s (delta n))` is `3 + mu b + mu s + mu n`.
    -- The result has a larger `mu` by `1 + mu s`.
    -- Therefore, `kappa` must strictly decrease.
    apply drop_left
    -- The proof obligation is:
    -- `kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n))`
    simp
    -- After simplification, the goal becomes:
    -- `max (kappa s) (kappa (recDelta b s n)) < max (max (kappa b) (kappa s)) (kappa n) + 1`
    -- This inequality holds by analysis of `kappa (recDelta b s n)`, which requires
    -- a case distinction on whether `n` is a `delta`-term.
    sorry

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- The measure for `eqW a a` is `(kappa a, 4 + 2 * mu a)`.
    -- The measure for `void` is `(0, 0)`.
    -- The argument is identical to the R_int_delta case.
    by_cases h_kappa : kappa a = 0
    · -- Case: kappa ties. We show mu drops.
      apply drop_right
      · -- Goal: mu void < mu (eqW a a)
        simp; omega
      · -- Goal: kappa void = kappa (eqW a a)
        simp [h_kappa]
    · -- Case: kappa drops.
      apply drop_left
      · -- Goal: kappa void < kappa (eqW a a)
        simp; exact Nat.pos_of_ne_zero h_kappa

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- The `kappa` measure is `max (kappa a) (kappa b)` for both terms, so it ties.
    -- We show that `mu` strictly decreases.
    apply drop_right
    · -- Goal: mu (integrate (merge a b)) < mu (eqW a b)
      -- This simplifies to `3 + mu a + mu b < 4 + mu a + mu b`.
      simp; omega
    · -- Goal: kappa (integrate (merge a b)) = kappa (eqW a b)
      simp
