Below is a **proof‑skeleton** that fills the five `sorry` holes in `measure_decreases`.  
The idea for every rule is the same:

1. Compute `kappa` of the left‑ and right‑hand side (using the `@[simp]` lemmas).  
2. Compare the two `kappa` values.  
   * If the left‑hand side has a **strictly larger** `kappa`, use `drop_left`.  
   * If the `kappa` values are **equal**, compare `mu`; a strict drop in `mu` gives `drop_right`.  
3. Finish the proof with `exact` (or `apply`) the appropriate lemma.

The scripts use the already‑proved lemmas `drop_left`, `drop_right`, and the `@[simp]` lemmas for `kappa` and `mu`.  The only additional arithmetic is discharged by `omega`.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
  -- kappa(integrate (delta t)) = kappa t   (delta is transparent)
  have hkL : kappa (integrate (delta t)) = kappa t := by simp
  have hkR : kappa void = 0 := by simp
  -- case split on whether kappa t = 0
  by_cases hk0 : kappa t = 0
  · -- kappa ties, mu drops
    have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
    have hk : kappa void = kappa (integrate (delta t)) := by simp [hk0, hkL, hkR]
    exact drop_right hmu hk
  · -- kappa strictly decreases
    have hk : kappa void < kappa (integrate (delta t)) := by
      simp [hkL, hkR, hk0]  -- because hk0 gives kappa t > 0
    exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
  -- kappa(recDelta b s void) = max (kappa b) (kappa s)
  have hkL : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by simp
  have hkR : kappa b = kappa b := rfl
  -- compare the two kappas
  by_cases hk : kappa b < kappa s
  · -- left side has larger kappa
    have hk' : kappa b < kappa (recDelta b s void) := by simp [hkL, hk]
    exact drop_left hk'
  · -- kappa are equal (or b larger); in any case ≤, and mu drops
    have hk_eq : kappa b = kappa (recDelta b s void) := by
      simp [hkL]; omega   -- because hk is false, i.e. kappa s ≤ kappa b
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]; omega    -- 3 + mu b + mu s  >  mu b
    exact drop_right hmu hk_eq

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  -- kappa(left) = max(kappa b, kappa s, kappa n) + 1
  have hkL : kappa (recDelta b s (delta n)) =
              Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by simp
  -- kappa(right) = max(kappa s, kappa(recDelta b s n))
  have hkR : kappa (app s (recDelta b s n)) =
              Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) := by simp
  -- The +1 on the left makes kappa strictly larger
  have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
    simp [hkL, hkR]; omega
  exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
  -- kappa(eqW a a) = kappa a
  have hkL : kappa (eqW a a) = kappa a := by simp
  have hkR : kappa void = 0 := by simp
  by_cases hk0 : kappa a = 0
  · -- kappa ties, mu drops
    have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
    have hk : kappa void = kappa (eqW a a) := by simp [hk0, hkL, hkR]
    exact drop_right hmu hk
  · -- kappa strictly decreases
    have hk : kappa void < kappa (eqW a a) := by simp [hkL, hkR, hk0]
    exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
  -- kappa is the same on both sides
  have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp
  -- compare mu:  mu(integrate(merge a b)) = 1 + (2 + mu a + mu b) = 3 + mu a + mu b
  have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
    simp [mu]; omega   -- 3 + mu a + mu b  < 4 + mu a + mu b
  exact drop_right hmu hk
```

These five blocks replace the `sorry` placeholders and complete the proof of `measure_decreases`.  Once inserted, the whole theorem `strong_normalization` follows automatically.
