Here is a complete proof skeleton for the five remaining cases.

```tactics
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · -- kappa is 0 on both sides, so we drop on mu
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      exact drop_right hmu hk
    · -- kappa void = 0 < kappa t = kappa (integrate (delta t))
      have hk : kappa void < kappa (integrate (delta t)) := by simp; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hkappa : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by
      simp [kappa]
    by_cases h : kappa b < kappa s
    · -- kappa recDelta = kappa s > kappa b
      have hk : kappa b < kappa (recDelta b s void) := by
        rw [hkappa]
        rw [Nat.max_eq_right (by omega)]
        omega
      exact drop_left hk
    · -- kappa b >= kappa s, so kappa ties and mu drops by 3 + mu s
      have hk : kappa b = kappa (recDelta b s void) := by
        rw [hkappa]
        rw [Nat.max_eq_left (by omega)]
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]; omega
      exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- the +1 in kappa for the delta-wrapped argument is lost
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      rw [← Nat.max_assoc]
      rw [Nat.max_eq_right (le_max_right (kappa b) (kappa s))]
      omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · -- kappa ties at 0, mu drops from 4 + 2*mu a to 0
      have hk : kappa void = kappa (eqW a a) := by simp [h]
      have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
      exact drop_right hmu hk
    · -- kappa void = 0 < kappa a = kappa (eqW a a)
      have hk : kappa void < kappa (eqW a a) := by simp; omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa is max(kappa a, kappa b) on both sides
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    -- mu drops from 4 + mu a + mu b to 3 + mu a + mu b
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    exact drop_right hmu hk
```
