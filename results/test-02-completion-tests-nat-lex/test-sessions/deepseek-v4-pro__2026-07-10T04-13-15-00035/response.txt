-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- mu always decreases
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu] ; omega
    -- kappa ties iff kappa t = 0
    by_cases hk : kappa t = 0
    · -- kappa ties
      have hk_eq : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk]
      exact drop_right hmu hk_eq
    · -- kappa strictly drops (kappa void = 0 < kappa t)
      have hk_lt : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]
        exact Nat.pos_of_ne_zero hk
      exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- mu always decreases
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu] ; omega
    by_cases hmax : Nat.max (kappa b) (kappa s) = kappa b
    · -- kappa ties
      have hk_eq : kappa b = kappa (recDelta b s void) := by
        simp [kappa, kappa_void, hmax]
      exact drop_right hmu hk_eq
    · -- kappa strictly drops
      have hk_lt : kappa b < kappa (recDelta b s void) := by
        -- kappa (recDelta b s void) = max (kappa b) (kappa s) > kappa b
        have h_lt : kappa b < kappa s := by
          apply Nat.lt_of_not_ge
          intro hle
          apply hmax
          exact Nat.max_eq_left hle
        -- then max is kappa s
        simp [kappa, kappa_void, Nat.max_eq_right (Nat.le_of_lt h_lt)]
        exact h_lt
      exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- The crucial observation: kappa s ≤ max (max (kappa b) (kappa s)) (kappa n)
    have hbase : kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) :=
      calc
        kappa s ≤ Nat.max (kappa b) (kappa s)       := le_max_right
        _ ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := le_max_left
    -- Hence kappa of RHS simplifies to that max, and kappa of LHS is max+1
    have hR : kappa (app s (recDelta b s n)) =
               Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
      simp [kappa, kappa_rec_delta]
      exact Nat.max_eq_right hbase
    have hL : kappa (recDelta b s (delta n)) =
               (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) + 1 := by
      simp [kappa_rec_delta]
    have hk_lt : kappa (app s (recDelta b s n)) <
                 kappa (recDelta b s (delta n)) := by
      rw [hR, hL]
      omega
    exact drop_left hk_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    have hmu : mu void < mu (eqW a a) := by
      simp [mu] ; omega
    by_cases hk : kappa a = 0
    · -- kappa ties
      have hk_eq : kappa void = kappa (eqW a a) := by
        simp [kappa, hk]
      exact drop_right hmu hk_eq
    · -- kappa strictly drops
      have hk_lt : kappa void < kappa (eqW a a) := by
        simp [kappa]
        exact Nat.pos_of_ne_zero hk
      exact drop_left hk_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by

      simp [mu] ; omega
    have hk_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by
