-- R_int_delta
| _, _, Step.R_int_delta t => by
    by_cases hk : kappa t = 0
    · -- kappa ties, mu drops
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]
      have hk_eq : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk]
      exact drop_right hmu hk_eq
    · -- kappa drops
      have hk_lt : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]
        exact Nat.pos_of_ne_zero hk
      exact drop_left hk_lt

-- R_rec_zero
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b ≥ kappa s
    · -- kappa equal (since max = kappa b)
      have hk_eq : kappa b = kappa (recDelta b s void) := by
        simp [kappa, Nat.max_eq_left h]
      have hmu : mu b < mu (recDelta b s void) := by simp [mu]
      exact drop_right hmu hk_eq
    · -- kappa drops
      have h_lt : kappa b < kappa (recDelta b s void) := by
        simp [kappa]
        have h' : kappa b < kappa s := by omega
        omega
      exact drop_left h_lt

-- R_rec_succ
| _, _, Step.R_rec_succ b s n => by
    have hk_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      calc
        kappa (app s (recDelta b s n)) = Nat.max (kappa s) (kappa (recDelta b s n)) := by simp
        _ = Nat.max (kappa s) (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) := by simp
        _ = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
          have h : kappa s ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
            calc
              kappa s ≤ Nat.max (kappa b) (kappa s) := Nat.le_max_right _ _
              _ ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := Nat.le_max_left _ _
          exact Nat.max_eq_right h
        _ < (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1) := by omega
        _ = kappa (recDelta b s (delta n)) := by simp
    exact drop_left hk_lt

    · -- kappa drops

| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      calc
        mu (integrate (merge a b)) = 1 + (2 + mu a + mu b) := by simp
        _ = 3 + mu a + mu b := by omega
        _ < 4 + mu a + mu b := by omega
        _ = mu (eqW a b) := by simp
    have hk_eq : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk_eq
