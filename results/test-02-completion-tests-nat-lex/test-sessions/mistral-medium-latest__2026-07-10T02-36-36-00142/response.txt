```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- kappa: kappa void = 0 < kappa (integrate (delta t)) = kappa t
    -- So we need to show kappa t > 0, but this isn't always true (if t = void)
    -- Instead, we can use mu: mu void = 0 < mu (integrate (delta t)) = 1 + mu t
    have hmu : mu void < mu (integrate (delta t)) := by simp [mu]
    -- kappa void = 0 ≤ kappa (integrate (delta t)) = kappa t
    -- So we need to handle cases where kappa t = 0
    by_cases hk : kappa t = 0
    · -- If kappa t = 0, then kappa void = kappa (integrate (delta t))
      have hk' : kappa void = kappa (integrate (delta t)) := by simp [kappa, hk]
      exact drop_right hmu hk'
    · -- If kappa t > 0, then kappa void < kappa (integrate (delta t))
      have hk' : kappa void < kappa (integrate (delta t)) := by simp [kappa]; omega
      exact drop_left hk'

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- kappa: kappa b ≤ kappa (recDelta b s void) = max (max (kappa b) (kappa s)) (kappa void)
    --        = max (kappa b) (kappa s)
    -- So kappa b ≤ kappa (recDelta b s void)
    -- mu: mu b < mu (recDelta b s void) = 3 + mu b + mu s + mu void
    have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
    by_cases hk : kappa b < kappa (recDelta b s void)
    · exact drop_left hk
    · -- If kappa b = kappa (recDelta b s void)
      have hk' : kappa b = kappa (recDelta b s void) := by
        simp [kappa] at hk ⊢
        omega
      exact drop_right hmu hk'

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- kappa: kappa (app s (recDelta b s n)) = max (kappa s) (kappa (recDelta b s n))
    --        = max (kappa s) (max (max (kappa b) (kappa s)) (kappa n))
    --        = max (kappa b) (kappa s) (kappa n)
    -- kappa (recDelta b s (delta n)) = max (max (kappa b) (kappa s)) (kappa n) + 1
    -- So kappa decreases by at least 1
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- kappa: kappa void = 0 ≤ kappa (eqW a a) = max (kappa a) (kappa a) = kappa a
    -- mu: mu void = 0 < mu (eqW a a) = 4 + mu a + mu a
    have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
    by_cases hk : kappa a = 0
    · -- If kappa a = 0, then kappa void = kappa (eqW a a)
      have hk' : kappa void = kappa (eqW a a) := by simp [kappa, hk]
      exact drop_right hmu hk'
    · -- If kappa a > 0, then kappa void < kappa (eqW a a)
      have hk' : kappa void < kappa (eqW a a) := by simp [kappa]; omega
      exact drop_left hk'

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa: kappa (integrate (merge a b)) = kappa (merge a b) = max (kappa a) (kappa b)
    --        = kappa (eqW a b)
    -- So kappa stays the same
    -- mu: mu (integrate (merge a b)) = 1 + mu (merge a b) = 1 + 2 + mu a + mu b = 3 + mu a + mu b
    --     mu (eqW a b) = 4 + mu a + mu b
    -- So mu decreases by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk
```

This skeleton provides the key insights for each case:
1. For `R_int_delta`: The measure decreases either through kappa (if t has positive kappa) or through mu (always)
2. For `R_rec_zero`: The measure decreases through mu (always), and possibly through kappa
3. For `R_rec_succ`: The measure always decreases through kappa (by at least 1)
4. For `R_eq_refl`: Similar to `R_int_delta`, decreases through mu or kappa
5. For `R_eq_diff`: The measure decreases through mu (by exactly 1) while kappa stays the same

Each case uses either `drop_left` (when kappa decreases) or `drop_right` (when kappa stays the same but mu decreases) to establish the lexicographic decrease.
