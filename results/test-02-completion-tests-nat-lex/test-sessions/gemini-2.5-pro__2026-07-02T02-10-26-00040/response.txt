-- ── Remaining cases ───────────────────────────────────────────

-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- We show either kappa drops, or kappa ties and mu drops.
    -- This depends on whether `kappa t` is zero or not.
    have hmu : mu void < mu (integrate (delta t)) := by simp; omega
    have hk_le : kappa void <= kappa (integrate (delta t)) := by simp; exact Nat.zero_le _
    cases Nat.eq_or_lt_of_le hk_le with
    | inl hk_eq => -- kappa ties
      exact drop_right hmu hk_eq.symm
    | inr hk_lt => -- kappa drops
      exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- We show either kappa drops, or kappa ties and mu drops.
    -- This depends on the relationship between `kappa b` and `kappa s`.
    have hmu : mu b < mu (recDelta b s void) := by simp; omega
    have hk_le : kappa b <= kappa (recDelta b s void) := by simp; apply Nat.le_max_left
    cases Nat.eq_or_lt_of_le hk_le with
    | inl hk_eq => -- kappa ties
      exact drop_right hmu hk_eq.symm
    | inr hk_lt => -- kappa drops
      exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- For this rule, the measure change depends on the structure of `n`.
    -- A case split is required. The proof appears to fail for n of the form `delta _`
    -- with the given measures, as `kappa` ties and `mu` increases.
    -- The skeleton below demonstrates this.
    cases n with
    | delta n' => -- Case: n is of the form `delta n'`.
      -- In this case, `kappa` can be shown to be equal.
      have hk_eq : kappa (app s (recDelta b s (delta n'))) = kappa (recDelta b s (delta (delta n'))) := by
        simp
        -- Proof requires showing max (kappa s) (max (max (kappa b) (kappa s)) (kappa n') + 1) =
        -- max (max (kappa b) (kappa s)) (kappa n') + 1, which holds.
        sorry
      -- However, `mu` strictly increases.
      have hmu_inc : mu (recDelta b s (delta (delta n'))) < mu (app s (recDelta b s (delta n'))) := by
        simp; omega
      -- This contradicts the goal, as we would need `mu` to decrease when `kappa` ties.
      sorry
    | _ => -- Case: n is not of the form `delta _`.
      -- In this case, `kappa` strictly decreases.
      have hk_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
        simp [kappa]
        -- Proof requires showing max (kappa s) (max (max ...)) < max (max ...) + 1, which holds.
        sorry
      exact drop_left hk_lt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- We show either kappa drops, or kappa ties and mu drops.
    -- This depends on whether `kappa a` is zero or not.
    have hmu : mu void < mu (eqW a a) := by simp; omega
    have hk_le : kappa void <= kappa (eqW a a) := by simp; exact Nat.zero_le _
    cases Nat.eq_or_lt_of_le hk_le with
    | inl hk_eq => -- kappa ties
      exact drop_right hmu hk_eq.symm
    | inr hk_lt => -- kappa drops
      exact drop_left hk_lt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- For this rule, kappa always ties and mu always drops.
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp; omega
    exact drop_right hmu hk
