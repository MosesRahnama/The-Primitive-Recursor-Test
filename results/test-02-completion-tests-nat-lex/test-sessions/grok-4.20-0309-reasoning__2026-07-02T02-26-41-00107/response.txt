-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      have hk : kappa void = kappa (integrate (delta t)) := by simp [kappa, h]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by simp [kappa]; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa s ≤ kappa b
    · have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      have hk : kappa b = kappa (recDelta b s void) := by
        simp only [kappa]; rw [Nat.max_comm]; simp [Nat.max, h]
      exact drop_right hmu hk
    · have hk : kappa b < kappa (recDelta b s void) := by
        simp only [kappa]; rw [Nat.max_comm]; simp [Nat.max, h]; omega
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    by_cases h : kappa (recDelta b s n) < Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1
    · have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
        simp only [kappa]; apply Nat.lt_of_le_of_lt _ h
        simp only [kappa_app]; apply Nat.le_max_right
      exact drop_left hk
    · have hk : kappa (app s (recDelta b s n)) = kappa (recDelta b s (delta n)) := by
        simp only [kappa] at h ⊢; omega
      have hmu : mu (app s (recDelta b s n)) < mu (recDelta b s (delta n)) := by simp [mu]; omega
      exact drop_right hmu hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by

    · have hk : kappa void < kappa (eqW a a) := by simp [kappa]; omega

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]

The skeleton uses `by_cases` on whether `kappa` drops (to select `drop_left` vs. `drop_right`) for rules where the decrease is not uniform. For `R_rec_succ` the `by_cases` splits on whether the inner recursive call preserves the `kappa` bound (corresponding to whether the predecessor `n` itself triggers the `delta` case in `kappa`); the arithmetic follows from the definitions of `kappa`/`mu` and is discharged by `omega` after `simp`. The `R_eq_diff` and merge cases are uniform (`kappa` equal, `mu` drops), so no split is needed. All cases rely only on the provided `simp` lemmas, `drop_*` helpers, and basic `Nat` arithmetic.
