-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases h : kappa t = 0
    · -- Case kappa t = 0: kappa equal (both 0), mu decreases strictly
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      exact drop_right hmu hk.symm
    · -- Case kappa t > 0: kappa decreases strictly from kappa t to 0
      have hk : kappa void < kappa (integrate (delta t)) := by simp [h]; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases h : kappa b = kappa (recDelta b s void)
    · -- Case kappa b = max(kappa b, kappa s): kappa equal, mu decreases strictly
      have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      exact drop_right hmu h.symm
    · -- Case kappa b < max(kappa b, kappa s): kappa decreases strictly
      have hk : kappa b < kappa (recDelta b s void) := by
        simp [kappa] at h ⊢
        omega
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- Kappa decreases strictly because the +1 from the delta case is dropped
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · -- Case kappa a = 0: kappa equal (both 0), mu decreases strictly
      have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
      have hk : kappa void = kappa (eqW a a) := by simp [h]
      exact drop_right hmu hk.symm
    · -- Case kappa a > 0: kappa decreases strictly
      have hk : kappa void < kappa (eqW a a) := by simp [h]; omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- Kappa unchanged: max(kappa a, kappa b) = max(kappa a, kappa b)
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    -- Mu decreases: 3 + mu a + mu b < 4 + mu a + mu b
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]
      omega
    exact drop_right hmu hk.symm
