Below is a proof‑skeleton that replaces the five `sorry`s.  
Each case follows the same pattern as the already‑proved `R_merge_*` cases: we compare the two components of the measure `(κ,μ)`, prove that either κ drops strictly (using `drop_left`) or that κ stays the same while μ drops (using `drop_right`), and then apply the appropriate lemma.

```lean
/-- integrate (delta t) --> void -/
| _, _, Step.R_int_delta t => by
  have hk : kappa (integrate (delta t)) = kappa t := by simp [kappa]
  have hk0 : kappa void = 0 := by simp [kappa]
  cases Nat.lt_or_eq_of_le (Nat.zero_le (kappa t)) with
  | inl hlt =>               -- κ drops
    have hklt : kappa void < kappa (integrate (delta t)) :=
      by simp [hk, hk0] at hlt
    exact drop_left hklt
  | inr heq =>               -- κ ties, μ drops
    have hkeq : kappa void = kappa (integrate (delta t)) :=
      by simp [hk, hk0, heq]
    have hmu : mu void < mu (integrate (delta t)) :=
      by simp [mu]; omega
    exact drop_right hmu hkeq

/-- recDelta b s void --> b -/
| _, _, Step.R_rec_zero b s => by
  have hk : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by simp [kappa]
  have hk' : kappa b ≤ kappa (recDelta b s void) := by simp [hk]; exact Nat.le_max_left _ _
  cases Nat.lt_or_eq_of_le hk' with
  | inl hlt =>               -- κ drops
    exact drop_left hlt
  | inr heq =>               -- κ ties, μ drops
    have hmu : mu b < mu (recDelta b s void) :=
      by simp [mu]; omega
    exact drop_right hmu heq

/-- recDelta b s (delta n) --> app s (recDelta b s n) -/
| _, _, Step.R_rec_succ b s n => by
  -- κ on the left‑hand side (the term being reduced)
  have hkL : kappa (recDelta b s (delta n)) =
             Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 :=
    by simp [kappa]
  -- κ of the right‑hand side
  have hkR1 : kappa (app s (recDelta b s n)) =
              Nat.max (kappa s) (kappa (recDelta b s n)) :=
    by simp [kappa]
  have hkR2 : kappa (recDelta b s n) =
              Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) :=
    by simp [kappa]
  -- combine the two equalities for the RHS
  have hkR : kappa (app s (recDelta b s n)) =
             Nat.max (kappa b) (kappa s) (kappa n) :=
    calc
      _ = Nat.max (kappa s) (kappa (recDelta b s n))   := by rw [hkR1]
      _ = Nat.max (kappa s)
            (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n)) := by rw [hkR2]
      _ = Nat.max (kappa b) (kappa s) (kappa n) := by simp [Nat.max_assoc, Nat.max_comm]
  -- strict decrease of κ
  have hklt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) :=
    by rw [hkR, hkL]; simp
  exact drop_left hklt

/-- eqW a a --> void -/
| _, _, Step.R_eq_refl a => by
  have hk : kappa (eqW a a) = kappa a := by simp [kappa]
  have hk0 : kappa void = 0 := by simp [kappa]
  cases Nat.lt_or_eq_of_le (Nat.zero_le (kappa a)) with
  | inl hlt =>               -- κ drops
    have hklt : kappa void < kappa (eqW a a) :=
      by simp [hk, hk0] at hlt
    exact drop_left hklt
  | inr heq =>               -- κ ties, μ drops
    have hkeq : kappa void = kappa (eqW a a) :=
      by simp [hk, hk0, heq]
    have hmu : mu void < mu (eqW a a) :=
      by simp [mu]; omega
    exact drop_right hmu hkeq

/-- eqW a b --> integrate (merge a b) -/
| _, _, Step.R_eq_diff a b => by
  have hk : kappa (integrate (merge a b)) = Nat.max (kappa a) (kappa b) := by simp [kappa]
  have hk' : kappa (eqW a b) = Nat.max (kappa a) (kappa b) := by simp [kappa]
  have hkeq : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [hk, hk']
  have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
  exact drop_right hmu hkeq
```

These five blocks give a complete proof that each of the remaining rewrite rules strictly decreases the lexicographic measure `(κ, μ)`. Consequently the overall proof of `measure_decreases` goes through, and the final well‑foundedness theorem follows.
