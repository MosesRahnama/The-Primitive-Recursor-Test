-- integrate (delta t) --> void : kappa may drop to 0 or tie (if kappa t = 0), mu always drops to 0
| _, _, Step.R_int_delta t => by
    simp only [LexOrder, measure, kappa, mu]
    by_cases h : kappa t = 0
    · simp [h]; apply Prod.Lex.right; omega
    · apply Prod.Lex.left; omega

-- recDelta b s void --> b : kappa of b ≤ kappa of lhs (may drop or tie), mu of b drops by ≥ 3 + mu s
| _, _, Step.R_rec_zero b s => by
    simp only [LexOrder, measure, kappa, mu]
    by_cases h : kappa b < kappa s
    · apply Prod.Lex.left
      simp; omega [h]
    · apply Prod.Lex.right
      · simp; omega [h]
      · omega

-- recDelta b s (delta n) --> app s (recDelta b s n) : kappa drops by 1 (or ties if n headed by delta, but mu may rise; needs case analysis on n for full skeleton)
| _, _, Step.R_rec_succ b s n => by
    simp only [LexOrder, measure, kappa, mu]
    by_cases h : (match n with | delta _ => true | _ => false)
    · -- kappa ties, mu rises by 1 + mu s: requires refined measure or context (skeleton only)
      sorry
    · have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by simp [kappa]; omega
      exact drop_left hk

-- eqW a a --> void : kappa may drop to 0 or tie, mu always drops (to 0)
| _, _, Step.R_eq_refl a => by
    simp only [LexOrder, measure, kappa, mu]
    by_cases h : kappa a = 0
    · simp [h]; apply Prod.Lex.right; omega
    · apply Prod.Lex.left; omega

-- eqW a b --> integrate (merge a b) : kappa ties (max(ka, kb)), mu drops by 1
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk

The skeleton follows the style of the proved merge cases (using `have` for `kappa`/`mu` relations + `drop_left`/`drop_right` where a component strictly decreases). For mixed cases (kappa may tie or drop), `by_cases` on the `kappa` relation dispatches to the appropriate `Prod.Lex` constructor (or `drop_*` helper). The `R_rec_succ` case requires a structural case on the shape of `n` (to separate kappa-drop vs. kappa-tie subcases) because `kappa` definition is sensitive to whether the third argument is literally `delta`-headed; the tie subcase needs a refined argument (e.g., a lexicographic triple or multiset measure) to handle mu increase from duplication of `s`. All simp lemmas already in scope are used; `omega` closes the resulting Nat inequalities.
