Here is a compact “proof-skeleton’’ that can be pasted in place of the five remaining `sorry`s.
Each clause follows the same pattern as in the three already-proved
cases: 

• first show a suitable `mu`‐drop (`hmu`) or a strict `kappa`‐drop  
  (`hk_lt`),  

• then decide – by cases when necessary – whether the argument
  for the lexicographic decrease is `drop_left` (strict
  `kappa`) or `drop_right` (equal `kappa`, strict `mu`).

```
-- integrate (delta t)  -->  void
| _, _, Step.R_int_delta t => by
    -- μ(void) < μ(integrate (delta t))
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu]
    -- either κ drops strictly, or it ties at 0
    by_cases h : kappa t = 0
    · -- κ ties, rely on μ
      have hk : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, h]
      exact drop_right hmu hk
    · -- κ drops strictly
      have hk_lt : kappa void < kappa (integrate (delta t)) := by
        have : (0 : Nat) < kappa t := Nat.pos_of_ne_zero h
        simpa [kappa] using this
      exact drop_left hk_lt


-- recDelta b s void  -->  b
| _, _, Step.R_rec_zero b s => by
    -- μ-drop is always available
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]; omega
    -- decide whether κ ties or drops
    by_cases h : kappa s ≤ kappa b
    · -- κ ties
      have hk : kappa b = kappa (recDelta b s void) := by
        simp [kappa, h, Nat.max_eq_left h]
      exact drop_right hmu hk
    · -- κ drops because κ s dominates the max
      have hb_lt : kappa b < kappa s := Nat.lt_of_not_ge h
      have hk_lt : kappa b < kappa (recDelta b s void) := by
        -- κ(rec …) = max (max κb κs) 0, hence ≥ κs
        have : kappa s ≤ kappa (recDelta b s void) := by
          simp [kappa] using Nat.le_max_left_of_le
            (Nat.le_max_right _ _)
        exact Nat.lt_of_lt_of_le hb_lt this
      exact drop_left hk_lt


-- recDelta b s (delta n)  -->  app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- κ drops by the “+1’’ added on the delta branch
    have hk_lt : kappa (app s (recDelta b s n))
                 < kappa (recDelta b s (delta n)) := by
      -- lhs κ  =  κ(rec … n)  <  κ(rec … (delta n)) = … + 1
      have : kappa (app s (recDelta b s n)) =
            Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
        -- the max already contains κ s, hence the outer max collapses
        simp [kappa,
              (Nat.max_eq_right (Nat.le_max_left _ _))]
      -- use `Nat.lt_succ_self m : m < m.succ`
      simpa [kappa, this] using
        Nat.lt_succ_self (Nat.max (Nat.max (kappa b) (kappa s)) (kappa n))
    exact drop_left hk_lt


-- eqW a a  -->  void
| _, _, Step.R_eq_refl a => by
    have hmu : mu void < mu (eqW a a) := by
      simp [mu]; omega
    by_cases h : kappa a = 0
    · -- κ ties, fall back to μ
      have hk : kappa void = kappa (eqW a a) := by
        simp [kappa, h]
      exact drop_right hmu hk
    · -- κ drops strictly
      have hk_lt : kappa void < kappa (eqW a a) := by
        have : (0 : Nat) < kappa a := Nat.pos_of_ne_zero h
        simpa [kappa] using this
      exact drop_left hk_lt


-- eqW a b  -->  integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- κ stays the same, μ goes down by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk : kappa (integrate (merge a b)) =
              kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```

All five clauses follow exactly the same ideas as the three
already implemented proofs, and together they complete the
`measure_decreases` theorem.
