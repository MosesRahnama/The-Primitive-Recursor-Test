-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases hk0 : kappa t = 0
    · apply drop_right
      · simp [mu]; omega
      · simp [kappa, hk0]
    · apply drop_left
      · have hlt : kappa void < kappa (integrate (delta t)) := by
          simp [kappa]; omega
        exact hlt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]; omega
    by_cases h : kappa b ≥ kappa s
    · have hk_eq : kappa b = kappa (recDelta b s void) := by
        simp [kappa, h]
      exact drop_right hmu hk_eq
    · have hk_lt : kappa b < kappa (recDelta b s void) := by
        simp [kappa]; omega
      exact drop_left hk_lt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    apply drop_left
    simp [kappa, kappa_rec_delta]; omega

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases hk0 : kappa a = 0
    · apply drop_right
      · simp [mu]; omega
      · simp [kappa, hk0]
    · apply drop_left
      · have hlt : kappa void < kappa (eqW a a) := by
          simp [kappa]; omega
        exact hlt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    apply drop_right
    · simp [mu, mu_merge, mu_integrate]; omega
    · simp [kappa, kappa_merge, kappa_integrate]
