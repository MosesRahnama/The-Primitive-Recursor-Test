Here is the proof skeleton for the five remaining cases of the `measure_decreases` theorem.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- The measure of `void` is (0, 0).
    -- The measure of `integrate (delta t)` is (kappa t, 1 + mu t).
    -- If kappa t > 0, then kappa drops (0 < kappa t).
    -- If kappa t = 0, then kappa ties, and mu drops (0 < 1 + mu t).
    -- This covers all cases.
    by_cases h : kappa t = 0
    · have hk : kappa void = kappa (integrate (delta t)) := by simp [h]
      have hmu : mu void < mu (integrate (delta t)) := by simp
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by
        simp; exact Nat.pos_of_ne_zero h
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- The measure of `b` is (kappa b, mu b).
    -- The measure of `recDelta b s void` is (max(kappa b, kappa s), 3 + mu b + mu s).
    -- If kappa s <= kappa b, kappa ties at kappa b, and mu drops (mu b < 3+mu b+mu s).
    -- If kappa s > kappa b, kappa drops (kappa b < kappa s = max(kappa b, kappa s)).
    by_cases h : kappa s ≤ kappa b
    · have hk : kappa b = kappa (recDelta b s void) := by
        simp [Nat.max_eq_left h]
      have hmu : mu b < mu (recDelta b s void) := by simp; omega
      exact drop_right hmu hk
    · have hk : kappa b < kappa (recDelta b s void) := by
        simp
        rw [Nat.max_eq_right_of_lt (Nat.not_le.mp h)]
        exact Nat.not_le.mp h
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- This rule is the only one where kappa can increase.
    -- The `delta` wrapper on the third argument of `recDelta` causes a +1 bump to kappa.
    -- The RHS `app s (recDelta b s n)` does not have this bump.
    -- Let K = max(max(kappa b, kappa s), kappa n).
    -- kappa(recDelta b s (delta n)) = K + 1
    -- kappa(app s (recDelta b s n)) = max(kappa s, K) = K
    -- So kappa strictly decreases.
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      simp [kappa]
      apply Nat.lt_succ_of_le
      apply Nat.max_le_right
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    -- The measure of `void` is (0, 0).
    -- The measure of `eqW a a` is (kappa a, 4 + 2 * mu a).
    -- If kappa a > 0, then kappa drops (0 < kappa a).
    -- If kappa a = 0, then kappa ties, and mu drops (0 < 4 + 2 * mu a).
    by_cases h : kappa a = 0
    · have hk : kappa void = kappa (eqW a a) := by simp [h]
      have hmu : mu void < mu (eqW a a) := by simp; omega
      exact drop_right hmu hk
    · have hk : kappa void < kappa (eqW a a) := by
        simp; exact Nat.pos_of_ne_zero h
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- For this rule, kappa ties and mu drops.
    -- kappa(eqW a b) = max(kappa a, kappa b)
    -- kappa(integrate(merge a b)) = kappa(merge a b) = max(kappa a, kappa b)
    -- mu(eqW a b) = 4 + mu a + mu b
    -- mu(integrate(merge a b)) = 1 + mu(merge a b) = 1 + (2 + mu a + mu b) = 3 + mu a + mu b
    -- Mu drops by 1.
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp; omega
    exact drop_right hmu hk
```
