Here are the proof skeletons for the five unfinished cases, following the existing pattern of using `drop_left` when kappa strictly decreases and `drop_right` when kappa stays the same but mu decreases. The calculations are done with `simp` for measure expansions and `omega` for arithmetic.

-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases hk : kappa t = 0
    · -- kappa t = 0 => kappa ties, mu decreases
      have hk_eq : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk]
      have hmu_lt : mu void < mu (integrate (delta t)) := by
        simp [mu]
        omega
      exact drop_right hmu_lt hk_eq
    · -- kappa t > 0 => kappa void = 0 < kappa t = kappa (integrate (delta t))
      have hk_lt : kappa void < kappa (integrate (delta t)) := by
        simp [kappa]
        omega
      exact drop_left hk_lt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases hksle : kappa s ≤ kappa b
    · -- kappa s ≤ kappa b => kappa ties, mu decreases
      have hk_eq : kappa b = kappa (recDelta b s void) := by
        -- kappa (recDelta b s void) = max (kappa b) (kappa s) = kappa b
        simp [kappa, hksle]
      have hmu_lt : mu b < mu (recDelta b s void) := by
        simp [mu]
        omega
      exact drop_right hmu_lt hk_eq
    · -- kappa s > kappa b => kappa b < kappa (recDelta b s void)
      have hk_lt : kappa b < kappa (recDelta b s void) := by
        -- kappa (recDelta b s void) = max (kappa b) (kappa s) > kappa b
        have h_rec_kappa : kappa (recDelta b s void) = max (kappa b) (kappa s) := by
          simp [kappa]
        rw [h_rec_kappa]
        have h_lt : kappa b < kappa s := by omega
        exact lt_of_lt_of_le h_lt (Nat.le_max_right _ _)
      exact drop_left hk_lt

      -- kappa (app s (recDelta b s n)) = max (max (kappa b) (kappa s)) (kappa n)
      -- kappa (recDelta b s (delta n)) = max (max (kappa b) (kappa s)) (kappa n) + 1

        calc
          kappa s ≤ max (kappa b) (kappa s) := Nat.le_max_right _ _
          _ ≤ max (max (kappa b) (kappa s)) (kappa n) := Nat.le_max_left _ _
      simp [kappa]
      rw [Nat.max_eq_right h_le]
      omega
    exact drop_left hk_lt
