-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
  have h_kappa : kappa (integrate (delta t)) = kappa t := by simp
  have h_mu : mu (integrate (delta t)) = 1 + mu t := by simp
  by_cases h : kappa t = 0
  · have hk : kappa void = kappa (integrate (delta t)) := by simp [h_kappa, h]
    have hmu : mu void < mu (integrate (delta t)) := by simp [h_mu]; omega
    exact drop_right hmu hk
  · have hk_drop : kappa void < kappa (integrate (delta t)) := by
      simp [kappa_void, h_kappa]
      exact Nat.pos_of_ne_zero h
    exact drop_left hk_drop

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
  have h_kappa : kappa (recDelta b s void) = Nat.max (kappa b) (kappa s) := by simp
  have h_mu_drop : mu b < mu (recDelta b s void) := by simp; omega
  have h_le : kappa b ≤ kappa (recDelta b s void) := by
    rw [h_kappa]; apply Nat.le_max_left
  by_cases h_lt : kappa b < kappa (recDelta b s void)
  · exact drop_left h_lt
  · have hk_eq : kappa b = kappa (recDelta b s void) := Nat.eq_of_le_of_not_lt h_le h_lt
    exact drop_right h_mu_drop hk_eq

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
  have h_rec : kappa (recDelta b s n) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by simp
  have h_app : kappa (app s (recDelta b s n)) = Nat.max (kappa s) (kappa (recDelta b s n)) := by simp
  rw [h_rec] at h_app
  have h_app_simp : kappa (app s (recDelta b s n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
    rw [h_app]
    apply Nat.max_eq_right
    apply Nat.le_trans (Nat.le_max_right _ _) (Nat.le_max_left _ _)
  have h_lhs : kappa (recDelta b s (delta n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by simp
  have h_lt : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
    rw [h_app_simp, h_lhs]; omega
  exact drop_left h_lt
