A possible “skeleton” (no new definitions, only the five `sorry`s filled in) looks like this:

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- mu strictly drops, kappa may drop or tie
    have hmu : mu void < mu (integrate (delta t)) := by
      simp [mu]
    by_cases hk : kappa t = 0
    · -- kappa ties
      have hkappa : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, hk]
      exact drop_right hmu hkappa
    · -- kappa drops
      have hlt : kappa void < kappa (integrate (delta t)) := by
        have : 0 < kappa t := Nat.pos_of_ne_zero hk
        simpa [kappa] using this
      exact drop_left hlt

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- mu strictly drops
    have hmu : mu b < mu (recDelta b s void) := by
      simp [mu]
    -- compare kappas
    by_cases hk : kappa b = kappa (recDelta b s void)
    · exact drop_right hmu hk
    · have hlt : kappa b < kappa (recDelta b s void) := by
        have hle : kappa b ≤ kappa (recDelta b s void) := by
          -- kappa(b) ≤ max(kappa b, kappa s) ≤ ...
          have : kappa b ≤ Nat.max (kappa b) (kappa s) :=
            Nat.le_max_left _ _
          have : kappa b ≤ Nat.max (Nat.max (kappa b) (kappa s)) 0 :=
            le_trans this (Nat.le_max_left _ _)
          simpa [kappa] using this
        exact lt_of_le_of_ne hle hk
      exact drop_left hlt

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- kappa always drops by 1
    have hlt : kappa (app s (recDelta b s n)) <
               kappa (recDelta b s (delta n)) := by
      simp [kappa]
    exact drop_left hlt

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    have hmu : mu void < mu (eqW a a) := by
      simp [mu]
    by_cases hk : kappa a = 0
    · have hkappa : kappa void = kappa (eqW a a) := by
        simp [kappa, hk]
      exact drop_right hmu hkappa
    · have hlt : kappa void < kappa (eqW a a) := by
        have : 0 < kappa a := Nat.pos_of_ne_zero hk
        simpa [kappa] using this
      exact drop_left hlt

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- kappa ties, mu drops by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu] ; omega
    have hkappa : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hkappa
```

These blocks follow the same pattern used in the three already-proved rules:  

• establish the strict decrease (either on `kappa` or on `mu` when `kappa` ties),  
• call the suitable helper (`drop_left` or `drop_right`) to build the desired `LexOrder` proof.

Some minor arithmetic facts are discharged with `simp`, `Nat.pos_of_ne_zero`, `lt_of_le_of_ne`, or `omega` (mirroring the style already present in the file).
