Here is the proof skeleton to complete the remaining cases. 

For the rules where the change in `kappa` depends on the subterms (e.g., `0 < kappa t` vs `0 = kappa t`), we branch using `by_cases` to apply `drop_left` when `kappa` strictly drops, and `drop_right` when `kappa` ties. 

Note that `Step.R_rec_succ` contains a known mathematical edge case in this calculus: when `n` is of the form `delta n'`, both `kappa` ties and `mu` strictly increases, failing the well-founded lexicographic descent. Thus, its strict drop in `kappa` is cleanly isolated into an explicit `sorry`.

```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    by_cases hk : kappa void < kappa (integrate (delta t))
    · exact drop_left hk
    · have heq : kappa void = kappa (integrate (delta t)) := by simp [kappa] at hk ⊢; omega
      have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      exact drop_right hmu heq

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    by_cases hk : kappa b < kappa (recDelta b s void)
    · exact drop_left hk
    · have heq : kappa b = kappa (recDelta b s void) := by simp [kappa] at hk ⊢; omega
      have hmu : mu b < mu (recDelta b s void) := by simp [mu]; omega
      exact drop_right hmu heq

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
      sorry
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by

    · exact drop_left hk
    · have heq : kappa void = kappa (eqW a a) := by simp [kappa] at hk ⊢; omega
      have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
      exact drop_right hmu heq

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have heq : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    exact drop_right hmu heq
```
