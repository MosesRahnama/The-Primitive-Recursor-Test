      -- because M = max(max(kappa b, kappa s), kappa n) ≥ max(kappa b, kappa s) ≥ kappa s
      apply Nat.max_eq_right
      apply Nat.le_trans (Nat.le_max_left _ _) (Nat.le_max_left _ _)
  rw [h4] at h3
  have hk : kappa (app s (recDelta b s n)) = M := by rw [h3, h4]
  have hk' : M < M + 1 := by omega
  rw [hk, h1] at hk'
  exact drop_left hk'

But to keep it simple, we can rely on simp and then use omega for the inequality.

Actually, a simpler way:
  have : kappa (app s (recDelta b s n)) ≤ Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
    simp [kappa]
    -- This simplifies to: max(kappa s, max(max(kappa b, kappa s), kappa n)) 
    -- and we know that max(kappa s, X) ≤ X if X ≥ kappa s, but actually it's equal to X because X = max(...) ≥ kappa s.
    -- So we can prove equality, but for the inequality we only need ≤.
  -- But we need strict inequality: kappa(RHS) < kappa(LHS) = M+1.
  -- Since kappa(RHS) = M (as argued) and M < M+1, we can do:
  have h_kappa_rhs_val : kappa (app s (recDelta b s n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) := by
    simp [kappa]
    -- Now, we need to show that max(kappa s, max(max(kappa b, kappa s), kappa n)) = max(max(kappa b, kappa s), kappa n)
    -- This is true because max(max(kappa b, kappa s), kappa n) ≥ kappa s.
    rw [Nat.max_eq_right_of_le (by 
      apply Nat.le_trans (Nat.le_max_left _ _) (Nat.le_max_left _ _))]
  have h_kappa_lhs_val : kappa (recDelta b s (delta n)) = Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1 := by simp
  have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
    rw [h_kappa_rhs_val, h_kappa_lhs_val]
    omega
  exact drop_left hk

This is cleaner.
