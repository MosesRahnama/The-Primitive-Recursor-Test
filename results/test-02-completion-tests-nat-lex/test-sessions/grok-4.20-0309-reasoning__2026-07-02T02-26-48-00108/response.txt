```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    simp only [measure]
    by_cases h : kappa t = 0
    · have hmu : mu void < mu (integrate (delta t)) := by simp [mu]; omega
      have hk : kappa void = kappa (integrate (delta t)) := by simp [h, kappa]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (integrate (delta t)) := by simp [h]; omega
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    simp only [measure, kappa, mu]
    by_cases h : kappa b < kappa s
    · have hk : kappa b < kappa (recDelta b s void) := by simp [h]; omega
      exact drop_left hk
    · have hk : kappa b = kappa (recDelta b s void) := by simp [h]; omega
      have hmu : mu b < mu (recDelta b s void) := by simp; omega
      exact drop_right hmu hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    simp only [measure, kappa, mu]
    by_cases h : kappa (recDelta b s n) < Nat.max (Nat.max (kappa b) (kappa s)) (kappa n) + 1
    · have hk : kappa (app s (recDelta b s n)) < kappa (recDelta b s (delta n)) := by
        simp [h]; omega
      exact drop_left hk
    · have hk : kappa (app s (recDelta b s n)) = kappa (recDelta b s (delta n)) := by simp [h]; omega
      have hmu : mu (app s (recDelta b s n)) < mu (recDelta b s (delta n)) := by simp; omega
      exact drop_right hmu hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    simp only [measure]
    by_cases h : kappa a = 0
    · have hmu : mu void < mu (eqW a a) := by simp [mu]; omega
      have hk : kappa void = kappa (eqW a a) := by simp [h, kappa]
      exact drop_right hmu hk
    · have hk : kappa void < kappa (eqW a a) := by simp [h]; omega
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by simp [mu]; omega
    have hk : kappa (integrate (merge a b)) = kappa (eqW a b) := by simp [kappa]
    exact drop_right hmu hk
```

The skeletons use `by_cases` on whether `kappa` drops (for mixed cases) or directly use `drop_right` (for pure `mu`-decrease cases). All use the provided `drop_left`/`drop_right` helpers after establishing the required `kappa`/`mu` relations via `simp` and `omega`. The `rec_succ` case uses an analogous (but stronger) `by_cases` on an explicit `kappa` comparison derived from the `recDelta` definition.
