```lean
-- integrate (delta t) --> void
| _, _, Step.R_int_delta t => by
    -- Either κ strictly drops, or it ties and μ drops.
    by_cases h : kappa t = 0
    · -- κ ties, use μ–decrease
      have hmu : mu void < mu (integrate (delta t)) := by
        simp [mu]
      have hk  : kappa void = kappa (integrate (delta t)) := by
        simp [kappa, h]
      exact drop_right hmu hk
    · -- κ drops strictly
      have hk : kappa void < kappa (integrate (delta t)) := by
        have : 0 < kappa t := Nat.pos_of_ne_zero h
        simpa [kappa] using this
      exact drop_left hk

-- recDelta b s void --> b
| _, _, Step.R_rec_zero b s => by
    -- Split on whether κ(b) already dominates κ(s)
    by_cases hle : kappa s ≤ kappa b
    · -- κ ties, rely on μ–decrease
      have hmu : mu b < mu (recDelta b s void) := by
        simp [mu]; omega
      have hk  : kappa b = kappa (recDelta b s void) := by
        have hmax : Nat.max (kappa b) (kappa s) = kappa b :=
          Nat.max_eq_left hle
        simp [kappa, hmax]
      exact drop_right hmu hk
    · -- κ drops strictly
      have hlt : kappa b < kappa s := Nat.lt_of_not_ge hle
      have hrecd : kappa (recDelta b s void) =
          Nat.max (kappa b) (kappa s) := by
        simp [kappa]
      have hk : kappa b < kappa (recDelta b s void) := by
        have : kappa b < Nat.max (kappa b) (kappa s) :=
          Nat.lt_max_of_lt_right hlt
        simpa [hrecd] using this
      exact drop_left hk

-- recDelta b s (delta n) --> app s (recDelta b s n)
| _, _, Step.R_rec_succ b s n => by
    -- κ drops by exactly one
    have hk :
        kappa (app s (recDelta b s n)) <
        kappa (recDelta b s (delta n)) := by
      simp [kappa]; omega
    exact drop_left hk

-- eqW a a --> void
| _, _, Step.R_eq_refl a => by
    by_cases h : kappa a = 0
    · -- κ ties, μ drops
      have hmu : mu void < mu (eqW a a) := by
        simp [mu]; omega
      have hk  : kappa void = kappa (eqW a a) := by
        simp [kappa, h]
      exact drop_right hmu hk
    · -- κ drops
      have hk : kappa void < kappa (eqW a a) := by
        have : 0 < kappa a := Nat.pos_of_ne_zero h
        simpa [kappa_void, kappa_eqW] using this
      exact drop_left hk

-- eqW a b --> integrate (merge a b)
| _, _, Step.R_eq_diff a b => by
    -- κ ties, μ decreases by 1
    have hmu : mu (integrate (merge a b)) < mu (eqW a b) := by
      simp [mu]; omega
    have hk  : kappa (integrate (merge a b)) = kappa (eqW a b) := by
      simp [kappa]
    exact drop_right hmu hk
```

Each case now supplies  
• the required κ (recursion-depth) comparison,  
• the μ (size) comparison when κ ties, and  
• the appropriate lexicographic proof (`drop_left` or `drop_right`).  
Only basic `simp`, arithmetic lemmas and the `omega` tactic are used, so the file’s original imports suffice.
